// D2.4.2: C# owns ONE laboratory validation + staging session.
// Python TUF source is embedded in this assembly; no caller-supplied verdict,

internal static partial class Program

    static readonly string[] ReceiptFields = {
        "schema","session","manifest_sha256","tag","installed",
        "tuf_verified_test_only","production_authority","can_install",
        "operational_ready","automatic_recovery_authorized"
    };
    static readonly string[] ManifestFields = {
        "schema","repository","channel","tag","version","architecture",
        "asset","length","sha256","test_only","operational_installable"
    };

    static bool ValidChildReceipt(string raw,string session,string digest)
    {
        try
        {
            using var parsed=JsonDocument.Parse(raw,
                new JsonDocumentOptions {AllowTrailingCommas=false,
                    CommentHandling=JsonCommentHandling.Disallow,MaxDepth=5});
            if(parsed.RootElement.ValueKind!=JsonValueKind.Object) return false;
            var items=new Dictionary<string,JsonElement>(StringComparer.Ordinal);
            foreach(var p in parsed.RootElement.EnumerateObject())
                if(!items.TryAdd(p.Name,p.Value)) return false;
            if(items.Count!=ReceiptFields.Length ||
                ReceiptFields.Any(k=>!items.ContainsKey(k))) return false;
            bool S(string k,string expected) => items[k].ValueKind==JsonValueKind.String &&
                string.Equals(items[k].GetString(),expected,StringComparison.Ordinal);
            return S("schema","gf-d242-tuf-child/1") &&
                S("session",session) && S("manifest_sha256",digest) &&
                S("tag","v5.13.11.66") && S("installed","5.13.11.65") &&
                items["tuf_verified_test_only"].ValueKind==JsonValueKind.True &&
                items["production_authority"].ValueKind==JsonValueKind.False &&
                items["can_install"].ValueKind==JsonValueKind.False &&
                items["operational_ready"].ValueKind==JsonValueKind.False &&
                items["automatic_recovery_authorized"].ValueKind==JsonValueKind.False;
        }
        catch(JsonException) {return false;}
    }

    static bool ValidManifestBytes(byte[] bytes)
    {
        if(bytes.Length is < 1 or > 65536) return false;
        try
        {
            using var parsed=JsonDocument.Parse(bytes,
                new JsonDocumentOptions {AllowTrailingCommas=false,
                    CommentHandling=JsonCommentHandling.Disallow,MaxDepth=8});
            if(parsed.RootElement.ValueKind!=JsonValueKind.Object) return false;
            var items=new Dictionary<string,JsonElement>(StringComparer.Ordinal);
            foreach(var p in parsed.RootElement.EnumerateObject())
                if(!items.TryAdd(p.Name,p.Value)) return false;
            if(items.Count!=ManifestFields.Length ||
                ManifestFields.Any(k=>!items.ContainsKey(k))) return false;
            bool S(string k,string expected) => items[k].ValueKind==JsonValueKind.String &&
                string.Equals(items[k].GetString(),expected,StringComparison.Ordinal);
            return S("schema","gf-github-release-auth-lab/1") &&
                S("repository","walissonminel-1991/gerenciador-final") &&
                S("channel","stable") && S("tag","v5.13.11.66") &&
                S("version","5.13.11.66") &&
                S("architecture","win-x64") &&
                S("asset","GerenciadorFinal-portable-win-x64.zip") &&
                S("sha256",Convert.ToHexString(Digest).ToLowerInvariant()) &&
                items["length"].ValueKind==JsonValueKind.Number &&
                items["length"].TryGetInt32(out var length) &&
                length==Payload.Length &&
                items["test_only"].ValueKind==JsonValueKind.True &&
                items["operational_installable"].ValueKind==JsonValueKind.False;
        }
        catch(JsonException) {return false;}
    }

    static string ValidatorSource()
    {
        using var stream=typeof(Program).Assembly.GetManifestResourceStream(
            "GF.D242.TufChild") ?? throw new InvalidDataException("EMBEDDED_TUF_ABSENT");
        using var reader=new StreamReader(stream,Encoding.UTF8);
        var code=reader.ReadToEnd();
        if(code.Length is < 300 or > 16000 ||
            !code.Contains("verify_d0_tuf(",StringComparison.Ordinal))
            throw new InvalidDataException("EMBEDDED_TUF_INVALID");
        return code;
    }

    static async Task<(int Exit,string Stdout)> RunTufChild(
        string directory,string session,string d0,CancellationToken ct)
    {
        var psi=new ProcessStartInfo {
            FileName="python",
            UseShellExecute=false,
            CreateNoWindow=true,
            RedirectStandardOutput=true,
            RedirectStandardError=true,
            WorkingDirectory=directory
        };
        // No external verifier path or caller-controlled signature receipt.
        psi.ArgumentList.Add("-c");
        psi.ArgumentList.Add(ValidatorSource());
        psi.ArgumentList.Add("--d242-child");
        psi.ArgumentList.Add(directory);
        psi.ArgumentList.Add(session);
        psi.ArgumentList.Add(d0);
        using var child=Process.Start(psi)
            ?? throw new InvalidDataException("TUF_CHILD_NOT_STARTED");
        using var timeout=CancellationTokenSource.CreateLinkedTokenSource(ct);
        timeout.CancelAfter(TimeSpan.FromSeconds(15));
        try
        {
            var standardOut=child.StandardOutput.ReadToEndAsync(timeout.Token);
            var standardErr=child.StandardError.ReadToEndAsync(timeout.Token);
            await child.WaitForExitAsync(timeout.Token);
            var output=await standardOut;
            var errors=await standardErr;
            if(output.Length>4096 || errors.Length>4096)
                throw new InvalidDataException("CHILD_OUTPUT_TOO_LONG");
            return (child.ExitCode,output);
        }
        catch(OperationCanceledException)
        {
            if(!child.HasExited) child.Kill(entireProcessTree:true);
            throw;
        }
    }

    static async Task<int> RunD242(string[] args)
    {
        if(!OperatingSystem.IsWindows() ||
           Environment.GetEnvironmentVariable("GITHUB_ACTIONS")!="true")
            return 3;
        var root=Environment.GetEnvironmentVariable("RUNNER_TEMP");
        if(string.IsNullOrWhiteSpace(root) || !Directory.Exists(root))
            return 3;

        var outcome="rejected";
        var cause="NO_VALIDATION";
        var childStarted=false;
        var childSucceeded=false;
        var prepared=false;
        var connections=0;
        string? lab=null;
        try
        {
            if(args.Length!=2) throw new InvalidDataException("EXTERNAL_AUTH_ARGS_DENIED");
            var runner=Path.GetFullPath(root);
            var fixtures=Path.GetFullPath(args[1]);
            if(!fixtures.StartsWith(runner.TrimEnd(Path.DirectorySeparatorChar)+
                Path.DirectorySeparatorChar,StringComparison.OrdinalIgnoreCase) ||
               !Path.GetFileName(fixtures).StartsWith("gf-d242-",StringComparison.Ordinal))
                throw new InvalidDataException("FIXTURE_DIRECTORY_NOT_ALLOWED");
            var dirInfo=new DirectoryInfo(fixtures);
            if(!dirInfo.Exists ||
               (dirInfo.Attributes & FileAttributes.ReparsePoint)!=0)
                throw new InvalidDataException("REPARSE_FIXTURE_DIRECTORY");
            var d0=Path.Combine(runner,"gf-d24-d0","GF.D11.D0Probe.exe");
            if(!File.Exists(d0)) throw new InvalidDataException("D0_PROBE_MISSING");
            var manifestPath=Path.Combine(fixtures,"GF-RELEASE-MANIFEST.json");
            var manifestInfo=new FileInfo(manifestPath);
            if(!manifestInfo.Exists || manifestInfo.Length is < 1 or > 65536 ||
                (manifestInfo.Attributes & FileAttributes.ReparsePoint)!=0)
                throw new InvalidDataException("MANIFEST_NOT_PRESENT");

            // Random session correlates one child output with this invocation,
            // not a secret or same-user authentication token.
            var session=Convert.ToHexString(RandomNumberGenerator.GetBytes(16))
                .ToLowerInvariant();
            childStarted=true;
            var (exit,stdout)=await RunTufChild(fixtures,session,d0,
                                                  CancellationToken.None);
            if(exit!=0) throw new InvalidDataException("TUF_CHILD_REJECTED");
            var lines=stdout.Split('\n',StringSplitOptions.RemoveEmptyEntries);
            if(lines.Length!=1 ||
               !lines[0].StartsWith("GF_D242_VERIFIED ",StringComparison.Ordinal))
                throw new InvalidDataException("TUF_RECEIPT_MISSING_OR_DUPLICATE");

            // Snapshot exactly the bytes later parsed and used for staging.
            var bytes=await File.ReadAllBytesAsync(manifestPath);
            var hash=Convert.ToHexString(SHA256.HashData(bytes)).ToLowerInvariant();
            if(!ValidChildReceipt(lines[0]["GF_D242_VERIFIED ".Length..].TrimEnd('\r'),
                                  session,hash) || !ValidManifestBytes(bytes))
                throw new InvalidDataException("TUF_RECEIPT_OR_MANIFEST_MISMATCH");
            childSucceeded=true;

            // No second downloader. D2.3 shared Stage is reached exclusively
            // after the child has completed the in-session TUF/D0 checks.
            lab=Path.Combine(runner,"gf-d242-stage-"+Guid.NewGuid().ToString("N"));
            Directory.CreateDirectory(lab);
            var stage=Path.Combine(lab,"stage");
            Directory.CreateDirectory(stage);
            await using(var server=new LoopbackServer(new Reply(Fragment:7)))
            {
                try
                {
                    await Stage(server.Port,ExpectedRelease,stage,
                        CancellationToken.None,Payload.Length,Digest);
                }
                finally { connections=server.Accepts; }
            }
            await VerifyAsync(Path.Combine(stage,FinalName),Payload.Length,Digest,
                              CancellationToken.None);
            if(connections!=1) throw new InvalidDataException("INERT_SOCKET_COUNT");
            prepared=true;
            outcome="accepted_inert_test_only";
            cause="IN_SESSION_D0_TUF_PASSED";
        }
        catch(Exception e) when (e is InvalidDataException or IOException or
            JsonException or OperationCanceledException or Win32Exception or
            UnauthorizedAccessException or ArgumentException)
        {
            cause=e.GetType().Name;
        }
        finally
        {
            // Lab stage is created by this coordinator; NEVER use Stage to
            // delete arbitrary preexisting names on failures.
            if(lab!=null && Directory.Exists(lab)) Directory.Delete(lab,true);
        }
        Console.WriteLine("GF_D242_COORDINATOR "+JsonSerializer.Serialize(new {
            schema="gf-d242-session-coordinator/1",
            status=outcome,reason=cause,
            tuf_child_launched=childStarted,
            tuf_in_session_validated=childSucceeded,
            staged_inert_bytes=prepared,
            loopback_connections=connections,
            reused_dotnet_d23_stage=true,
            old_d24_direct_handoff_allowed=false,
            same_user_process_isolation_proven=false,
            production_authority=false,
            can_install=false,operational_ready=false,
            automatic_recovery_authorized=false
        }));
        return 0;
    }

    static int RunD242SelfTest()
    {
        var session="a1b2c3d4e5f60123456789abcdef0123";
        var hash=new string('0',64);
        var sample=new Dictionary<string,object> {
            ["schema"]="gf-d242-tuf-child/1",["session"]=session,
            ["manifest_sha256"]=hash,["tag"]="v5.13.11.66",
            ["installed"]="5.13.11.65",["tuf_verified_test_only"]=true,
            ["production_authority"]=false,["can_install"]=false,
            ["operational_ready"]=false,["automatic_recovery_authorized"]=false
        };
        var valid=JsonSerializer.Serialize(sample);
        var rejected=0;
        if(!ValidChildReceipt(valid,session,hash)) return 19;
        if(!ValidChildReceipt(valid,"b"+session[1..],hash)) rejected++;
        if(!ValidChildReceipt(valid,session,new string('1',64))) rejected++;
        var extra=new Dictionary<string,object>(sample){["extra"]=true};
        if(!ValidChildReceipt(JsonSerializer.Serialize(extra),session,hash)) rejected++;
        var fake=new Dictionary<string,object>(sample){["tuf_verified_test_only"]=false};
        if(!ValidChildReceipt(JsonSerializer.Serialize(fake),session,hash)) rejected++;
        var missing=new Dictionary<string,object>(sample);missing.Remove("session");
        if(!ValidChildReceipt(JsonSerializer.Serialize(missing),session,hash)) rejected++;
        if(!ValidChildReceipt("{\"schema\":1}",session,hash)) rejected++;
        if(!ValidChildReceipt(valid.Replace("\"can_install\":false",
             "\"can_install\":false,\"can_install\":false"),session,hash)) rejected++;
        Console.WriteLine("GF_D242_RECEIPT_SELFTEST "+JsonSerializer.Serialize(new {
            schema="gf-d242-receipt-selftest/1",
            status=rejected==7?"passed":"failed",
            synthetic_receipts_rejected=rejected,
            stages_started=0,automatic_recovery_authorized=false
        }));
        return rejected==7?0:20;
    }
