internal sealed class ManualClock(DateTimeOffset initial) : TimeProvider
{
    internal DateTimeOffset Wall=initial;
    internal long Ticks;
    public override DateTimeOffset GetUtcNow()=>Wall;
    public override long GetTimestamp()=>Ticks;
    public override long TimestampFrequency=>TimeSpan.TicksPerSecond;
    internal void Advance(TimeSpan duration) { Wall+=duration; Ticks+=duration.Ticks; }
}

internal sealed class StateCase : IDisposable
{
    internal readonly string Root;
    internal readonly BundleRef Old,Candidate,Later;
    internal readonly string InstallationId=Guid.NewGuid().ToString("N");
    internal StateCase(string suiteRoot)
    {
        Root=Path.Combine(suiteRoot,Guid.NewGuid().ToString("N"));
        Directory.CreateDirectory(Root); File.WriteAllText(Path.Combine(Root,".gf-u2-fixture"),FixtureSandbox.Marker);
        Directory.CreateDirectory(Path.Combine(Root,"ledger")); Directory.CreateDirectory(Path.Combine(Root,"versions"));
        File.WriteAllText(Path.Combine(Root,"lease.fixture"),"persistent-lock-file-do-not-unlink\n");
        File.WriteAllText(Path.Combine(Root,"preferences.fixture.txt"),"CONSENT=NO;PERSONAL-DATA-INERT\n");
        File.WriteAllText(Path.Combine(Root,"legacy66.fixture.txt"),"LEGACY66-MARKER-NOT-A-REAL-BETA\n");
        File.WriteAllText(Path.Combine(Root,"unknown.fixture.txt"),"UNKNOWN-FILE-PRESERVE\n");
        Old=MakeBundle(1); Candidate=MakeBundle(2); Later=MakeBundle(3);
        var trust=new TrustFloor(1,1,1,1,1,Old.Version,CycleClock.Wire(DateTimeOffset.UtcNow-TimeSpan.FromMinutes(1)),[]);
        new PersistentState(Root).BootstrapFixture(new("gf-state-fixture/2",InstallationId,1,new string('0',64),trust,Old,null));
        File.WriteAllBytes(Path.Combine(Root,"active.fixture.json"),StateCodec.Encode(new FixtureSelector("gf-selector-fixture/2",InstallationId,Old)));
    }
    private BundleRef MakeBundle(int sequence)
    {
        byte[] bytes=Encoding.UTF8.GetBytes("GF-U2-INERT-BUNDLE-"+sequence+"-NO-EXE-NO-DLL\n");
        var bundle=new BundleRef(sequence,"0.0.0."+sequence,Hash(bytes),"linux-fixture-model");
        string folder=Path.Combine(Root,"versions",bundle.DirectoryName); Directory.CreateDirectory(folder);
        string path=Path.Combine(folder,"bundle.fixture.txt"); File.WriteAllBytes(path,bytes);
        if(OperatingSystem.IsWindows())
        { using var pinned=WindowsFileGuard.Open(Root,path); bundle=bundle with {Identity=pinned.Identity}; }
        return bundle;
    }
    internal CycleClock Clock()=>new(TimeProvider.System,StateCodec.Observed(new PersistentState(Root).Load().Record.Trust));
    internal StateRecord Begin(Action<string>? cut=null)
    { using var session=new FixtureSession(Root); return session.Begin(Candidate,Clock(),DateTimeOffset.UtcNow+TimeSpan.FromHours(1),true,cut); }
    internal void Preserved()
    {
        Require(File.ReadAllText(Path.Combine(Root,"preferences.fixture.txt"))=="CONSENT=NO;PERSONAL-DATA-INERT\n"
            && File.ReadAllText(Path.Combine(Root,"legacy66.fixture.txt"))=="LEGACY66-MARKER-NOT-A-REAL-BETA\n"
            && File.ReadAllText(Path.Combine(Root,"unknown.fixture.txt"))=="UNKNOWN-FILE-PRESERVE\n","USER_DATA_CHANGED");
    }
    public void Dispose() { Preserved(); Directory.Delete(Root,true); }
}

    private static async Task Test(string id,string description,Func<Task> test)
    {
        try { await test(); Results.Add(new{id,description,status="passed"}); Console.WriteLine("PASS "+id+" "+description); }
        catch(Exception error) { Results.Add(new{id,description,status="failed",error=error.GetType().Name+": "+error.Message}); Console.WriteLine("FAIL "+id+" "+error); }
    }
    private static Task Sync(Action test) { test(); return Task.CompletedTask; }
    private static void Refuses(string code,Action action)
    {
        try { action(); } catch(Rejected e) { Require(e.Code==code,"WRONG_REJECTION:"+e.Code+" expected "+code); return; }
        throw new Rejected("UNSAFE_ACCEPTANCE:"+code);
    }
    private static void IoRefuses(Action action)
    {
        try { action(); } catch(Exception e) when(e is IOException or UnauthorizedAccessException or System.ComponentModel.Win32Exception) { return; }
        throw new Rejected("KERNEL_OPERATION_UNEXPECTEDLY_ALLOWED");
    }
    private static StateRecord Next((StateRecord Record,string Digest) loaded)=>loaded.Record with {Generation=loaded.Record.Generation+1,PreviousDigest=loaded.Digest};
    private static Process Start(params string[] arguments)
    {
        var info=new ProcessStartInfo(Environment.ProcessPath!) {
            RedirectStandardOutput=true,RedirectStandardError=true,RedirectStandardInput=true,UseShellExecute=false,
            WorkingDirectory=Path.GetTempPath() };
        info.ArgumentList.Add(Assembly.GetExecutingAssembly().Location);
        info.ArgumentList.Add("--state-child"); foreach(string a in arguments) info.ArgumentList.Add(a);
        return Process.Start(info) ?? throw new Rejected("CHILD_START");
    }
    private static async Task<int> Probe(params string[] arguments)
    {
        using var process=Start(arguments);
        using var budget=new CancellationTokenSource(TimeSpan.FromSeconds(20));
        try { await process.WaitForExitAsync(budget.Token); }
        catch { if(!process.HasExited) process.Kill(true); throw; }
        string output=await process.StandardOutput.ReadToEndAsync(), error=await process.StandardError.ReadToEndAsync();
        Require(error.Length==0,"CHILD_ERROR:"+error); return process.ExitCode;
    }
    private static async Task KillAt(StateCase c,string point)
    {
        using var child=Start("cut",c.Root,point);
        try
        {
            string? line=await child.StandardOutput.ReadLineAsync().WaitAsync(TimeSpan.FromSeconds(20));
            Require(line=="CUT:"+point,"CUT_NOT_REACHED:"+line+":"+await ReadErrorIfExited(child));
            child.Kill(entireProcessTree:true); await child.WaitForExitAsync();
        }
        finally { if(!child.HasExited) { child.Kill(true); await child.WaitForExitAsync(); } }
    }
    private static async Task<string> ReadErrorIfExited(Process p)=>p.HasExited ? await p.StandardError.ReadToEndAsync() : "";
    internal static async Task<int> Entry(string[] args)
    {
        if(args[0]=="--state-child") return await Child(args[1..]);
        Require(args.Length==3,"STAGE2_USAGE"); fixtures=Path.GetFullPath(args[1]);
        suiteRoot=Path.Combine(Path.GetTempPath(),"gf-u2-"+Guid.NewGuid().ToString("N")); Directory.CreateDirectory(suiteRoot);
        try { await Cases(); }
        finally { Directory.Delete(suiteRoot,true); }
        int failed=Results.Count(r=>JsonSerializer.SerializeToElement(r).GetProperty("status").GetString()=="failed");
        var report=new {schema=1,stage=2,contract="gf-self-update/1",fixture_only=true,package_installable=false,
            product_code_executed=false,network_transport="none; signed fixtures in memory",real_process_crash_tests=6,
            platform=OperatingSystem.IsWindows()?"Windows NTFS kernel + fixture transaction":"Linux fixture persistence + transaction; no NTFS claim",
            runtime=RuntimeInformation.FrameworkDescription,os=RuntimeInformation.OSDescription,
            total=Results.Count,passed=Results.Count-failed,failed,windows_only_tests_not_run=OperatingSystem.IsWindows()?0:14,tests=Results};
        await File.WriteAllBytesAsync(Path.GetFullPath(args[2]),StateCodec.Encode(report));
        Console.WriteLine($"Stage 2 fixture security tests: {Results.Count-failed}/{Results.Count}; no GF executable or update package.");
        Console.WriteLine("GF_U2_REPORT "+Encoding.UTF8.GetString(StateCodec.Encode(report)));
        return failed==0?0:1;
    }

    private static Dictionary<string,byte[]> ReadFixture(string name)=>Directory.GetFiles(Path.Combine(fixtures,name))
        .ToDictionary(p=>Path.GetFileName(p)!,File.ReadAllBytes,StringComparer.Ordinal);
    private static async Task<int> Child(string[] args)
    {
        Require(args.Length>=2,"CHILD_USAGE"); string mode=args[0],root=args[1]; FixtureSandbox.RequireRoot(root);
        if(mode is "lease" or "inspect" or "cleanup-probe")
        {
            try {using var session=new FixtureSession(root); if(mode=="inspect")_=session.SelectionForFixtureInspectionOnly(); return 0;}
            catch(Rejected e) when(e.Code=="INSTALLATION_BUSY_OR_LOCK_MISSING") {return 10;}
        }
        if(mode=="cut")
        {
            Require(args.Length==3,"CHILD_USAGE"); string target=args[2];
            using var session=new FixtureSession(root); var old=session.Store.Load().Record.Confirmed;
            byte[] bytes=Encoding.UTF8.GetBytes("GF-U2-INERT-BUNDLE-2-NO-EXE-NO-DLL\n");
            var candidate=new BundleRef(2,"0.0.0.2",Hash(bytes),"linux-fixture-model");
            if(OperatingSystem.IsWindows()) {using var guard=WindowsFileGuard.Open(root,session.Files.BundlePath(candidate)); candidate=candidate with {Identity=guard.Identity};}
            void Cut(string phase,string point)
            {if(target==phase+":"+point) {Console.WriteLine("CUT:"+target); Console.Out.Flush(); _=Console.ReadLine(); throw new Rejected("PARENT_DID_NOT_KILL");}}
            session.Begin(candidate,new(TimeProvider.System,StateCodec.Observed(session.Store.Load().Record.Trust)),
                DateTimeOffset.UtcNow+TimeSpan.FromHours(1),true,p=>Cut("begin",p));
            if(target.StartsWith("recover:",StringComparison.Ordinal))session.Recover(p=>Cut("recover",p));
            return 12;
        }
        if(mode=="cache-replay")
        {
            Require(args.Length==3,"CHILD_USAGE"); fixtures=args[2]; using var s=new FixtureSession(root); var state=s.Store.Load().Record;
            Require(state.Trust.Timestamp==2 && state.Trust.Metadata.Count==4,"CACHE_NOT_PERSISTED");
            var cache=state.Trust.Metadata.ToDictionary(p=>p.Key,p=>Convert.FromBase64String(p.Value));
            try {await FixtureValidator.Evaluate(File.ReadAllBytes(Path.Combine(fixtures,"trusted-root-v1.json")),ReadFixture("valid"),
                new("5.13.11.66",66,new string('c',64),StateCodec.Observed(state.Trust)),cache);}
            catch(Rejected e) when(e.Code=="TUF_REJECTED") {return 0;} return 13;
        }
        throw new Rejected("CHILD_MODE");
    }
