using System.ComponentModel;
using System.Runtime.InteropServices;
using System.Runtime.Versioning;
using System.Security.AccessControl;
using System.Security.Principal;
using System.Text;
using Microsoft.Win32.SafeHandles;
using static GfUpdaterFixture.StrictMetadata;

namespace GfUpdaterFixture;

// Experimental NTFS capability. Only inert fixture directories use this adapter.
// No ACL, owner, token privileges, AV settings or OS configuration are modified.
[SupportedOSPlatform("windows")]
internal sealed class WindowsFileGuard : IDisposable
{
    private const uint GenericRead=0x80000000, OpenExisting=3, OpenReparse=0x00200000, BackupSemantics=0x02000000;
    private readonly List<SafeFileHandle> ancestors=[];
    private SafeFileHandle? file;
    private readonly string path;
    private readonly string sandbox;
    internal string Identity { get; private set; }="";
    private WindowsFileGuard(string sandbox,string path) { this.sandbox=sandbox; this.path=path; }

    internal static WindowsFileGuard Open(string sandbox,string path,bool directory=false)
    {
        FixtureSandbox.RequireRoot(sandbox);
        string full=Path.GetFullPath(path), root=Path.GetFullPath(sandbox);
        Require(full==root || full.StartsWith(root+Path.DirectorySeparatorChar,StringComparison.OrdinalIgnoreCase),"NTFS_SCOPE");
        Require(full.Length<=240 && !full.StartsWith("\\\\",StringComparison.Ordinal) && !full[3..].Contains(':'),"NTFS_PATH");
        var drive=new DriveInfo(Path.GetPathRoot(full)!);
        Require(drive.DriveType==DriveType.Fixed && drive.DriveFormat=="NTFS","NTFS_FIXED_REQUIRED");
        var guard=new WindowsFileGuard(root,full);
        try
        {
            string current=Path.GetPathRoot(full)!;
            string[] components=full[current.Length..].Split(Path.DirectorySeparatorChar,StringSplitOptions.RemoveEmptyEntries);
            // Pin every ancestor before opening the next component: final-component NOFOLLOW alone is insufficient.
            guard.ancestors.Add(OpenChecked(current,true));
            for(int i=0;i<components.Length;i++)
            {
                current=Path.Combine(current,components[i]);
                bool isDirectory=i<components.Length-1 || directory;
                var handle=OpenChecked(current,isDirectory);
                try
                {
                    if(current.Equals(root,StringComparison.OrdinalIgnoreCase)
                        || current.StartsWith(root+Path.DirectorySeparatorChar,StringComparison.OrdinalIgnoreCase)) CheckAcl(handle);
                    Require(Volume(handle)==Volume(guard.ancestors[0]),"NTFS_VOLUME_CHANGED");
                }
                catch { handle.Dispose(); throw; }
                if(i==components.Length-1 && !directory) guard.file=handle;
                else guard.ancestors.Add(handle);
            }
            guard.Identity=Id(guard.file ?? guard.ancestors[^1]);
            return guard;
        }
        catch { guard.Dispose(); throw; }
    }
    internal static SafeFileHandle OpenExclusiveLease(string sandbox)
    {
        // Caller pins the ancestor chain first. Open the lock object itself without following a link.
        FixtureSandbox.RequireRoot(sandbox);
        var handle=CreateFileW(Path.Combine(sandbox,"lease.fixture"),GenericRead|0x40000000,0,IntPtr.Zero,
            OpenExisting,OpenReparse,IntPtr.Zero);
        if(handle.IsInvalid) {int error=Marshal.GetLastWin32Error(); handle.Dispose(); throw new Win32Exception(error,"NTFS_LEASE_OPEN");}
        try {CheckType(handle,false); CheckAcl(handle); return handle;}
        catch {handle.Dispose(); throw;}
    }
    private static SafeFileHandle OpenChecked(string path,bool directory)
    {
        // Directories must share WRITE for child rename/ReplaceFile. Deny DELETE to pin their names.
        // ACL + NOFOLLOW + identity/reparse rechecks remain mandatory; file bytes deny WRITE and DELETE.
        uint sharing=directory?3u:1u;
        var handle=CreateFileW(path,GenericRead,sharing,IntPtr.Zero,OpenExisting,OpenReparse|BackupSemantics,IntPtr.Zero);
        if(handle.IsInvalid) { handle.Dispose(); throw new Win32Exception(Marshal.GetLastWin32Error(),"NTFS_OPEN_FAILED"); }
        try
        {
            CheckType(handle,directory);
            var final=new StringBuilder(1024);
            uint count=GetFinalPathNameByHandleW(handle,final,(uint)final.Capacity,0);
            Require(count>0 && count<final.Capacity,"NTFS_FINAL_PATH");
            Require(final.ToString().TrimEnd('\\').Equals(("\\\\?\\"+path).TrimEnd('\\'),StringComparison.OrdinalIgnoreCase),"NTFS_PATH_ALIAS");
            return handle;
        }
        catch { handle.Dispose(); throw; }
    }
    private static void CheckType(SafeFileHandle handle,bool directory)
    {
        if(!GetFileInformationByHandle(handle,out var info)) throw new Win32Exception(Marshal.GetLastWin32Error());
        Require((info.Attributes&0x400)==0 && ((info.Attributes&0x10)!=0)==directory,"NTFS_REPARSE_OR_TYPE");
        if(!directory) Require(info.Links==1,"NTFS_HARDLINK");
    }
    private static ulong Volume(SafeFileHandle handle)
    {
        if(!GetFileInformationByHandleEx(handle,18,out var id,(uint)Marshal.SizeOf<FileIdInfo>()))
            throw new Win32Exception(Marshal.GetLastWin32Error());
        return id.Volume;
    }
    private static string Id(SafeFileHandle handle)
    {
        if(!GetFileInformationByHandleEx(handle,18,out var id,(uint)Marshal.SizeOf<FileIdInfo>()))
            throw new Win32Exception(Marshal.GetLastWin32Error());
        return $"{id.Volume:x16}-{id.Low:x16}{id.High:x16}";
    }
    internal void Recheck()
    {
        foreach(var handle in ancestors) CheckType(handle,true);
        var target=file ?? ancestors[^1];
        CheckType(target,file is null); CheckAcl(target);
        Require(Id(target)==Identity,"NTFS_ID_CHANGED");
    }
    internal byte[] Read(int maxBytes)
    {
        Require(file is not null,"NTFS_NOT_FILE"); Recheck();
        long length=RandomAccess.GetLength(file);
        Require(length is >0 && length<=maxBytes,"NTFS_SIZE");
        var bytes=new byte[(int)length]; int offset=0;
        while(offset<bytes.Length)
        {
            int n=RandomAccess.Read(file,bytes.AsSpan(offset),offset);
            Require(n>0,"NTFS_SHORT_READ"); offset+=n;
        }
        Require(RandomAccess.GetLength(file)==length,"NTFS_SIZE_CHANGED"); Recheck(); return bytes;
    }
    private static void CheckAcl(SafeFileHandle handle)
    {
        uint error=GetSecurityInfo(handle,1,5,out _,out _,out _,IntPtr.Zero,out var descriptor);
        if(error!=0) throw new Win32Exception((int)error);
        try
        {
            uint length=GetSecurityDescriptorLength(descriptor);
            Require(length is >0 and <=65536,"NTFS_SECURITY_SIZE");
            byte[] data=new byte[length]; Marshal.Copy(descriptor,data,0,data.Length);
            using var identity=WindowsIdentity.GetCurrent();
            string user=identity.User?.Value ?? throw new Rejected("NTFS_USER");
            ValidateDescriptor(new RawSecurityDescriptor(data,0),user);
        }
        finally { LocalFree(descriptor); }
    }
    internal static void ValidateDescriptor(RawSecurityDescriptor security,string user)
    {
        bool Trusted(SecurityIdentifier sid)=>sid.Value==user || sid.Value is "S-1-5-18" or "S-1-5-32-544";
        Require(security.Owner is not null && Trusted(security.Owner) && security.DiscretionaryAcl is not null,"NTFS_SECURITY");
        const uint writes=0x40000000|0x10000000|0x000d0156;
        foreach(GenericAce ace in security.DiscretionaryAcl)
        {
            if((ace.AceFlags&AceFlags.InheritOnly)!=0) continue;
            Require(ace is CommonAce,"NTFS_ACL_UNSUPPORTED");
            var common=(CommonAce)ace;
            if(common.AceQualifier==AceQualifier.AccessAllowed && ((uint)common.AccessMask&writes)!=0)
                Require(Trusted(common.SecurityIdentifier),"NTFS_BROAD_WRITE");
        }
    }
    public void Dispose()
    {
        file?.Dispose();
        for(int i=ancestors.Count-1;i>=0;i--) ancestors[i].Dispose();
    }
    [StructLayout(LayoutKind.Sequential)] private struct FileIdInfo { public ulong Volume,Low,High; }
    [StructLayout(LayoutKind.Sequential)] private struct FileInformation
    {
        public uint Attributes,CreationLow,CreationHigh,AccessLow,AccessHigh,WriteLow,WriteHigh,Volume,SizeHigh,SizeLow,Links,IndexHigh,IndexLow;
    }
    [DllImport("kernel32.dll",CharSet=CharSet.Unicode,SetLastError=true)]
    private static extern SafeFileHandle CreateFileW(string name,uint access,uint share,IntPtr security,uint creation,uint flags,IntPtr template);
    [DllImport("kernel32.dll",SetLastError=true)] [return:MarshalAs(UnmanagedType.Bool)]
    private static extern bool GetFileInformationByHandle(SafeFileHandle file,out FileInformation info);
    [DllImport("kernel32.dll",SetLastError=true)] [return:MarshalAs(UnmanagedType.Bool)]
    private static extern bool GetFileInformationByHandleEx(SafeFileHandle file,int infoClass,out FileIdInfo info,uint size);
    [DllImport("kernel32.dll",CharSet=CharSet.Unicode,SetLastError=true)]
    private static extern uint GetFinalPathNameByHandleW(SafeFileHandle file,StringBuilder path,uint size,uint flags);
    [DllImport("advapi32.dll",SetLastError=false)]
    private static extern uint GetSecurityInfo(SafeFileHandle handle,uint type,uint information,out IntPtr owner,out IntPtr group,
        out IntPtr dacl,IntPtr sacl,out IntPtr descriptor);
    [DllImport("advapi32.dll")] private static extern uint GetSecurityDescriptorLength(IntPtr descriptor);
    [DllImport("kernel32.dll")] private static extern IntPtr LocalFree(IntPtr value);
}
