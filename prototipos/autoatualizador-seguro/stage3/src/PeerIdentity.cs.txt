using System.ComponentModel;
using System.Diagnostics;
using System.IO.Pipes;
using System.Runtime.InteropServices;
using System.Runtime.Versioning;
using System.Security.Principal;
using System.Text;
using Microsoft.Win32.SafeHandles;

namespace GfUpdaterFixture;

[SupportedOSPlatform("windows")]
internal static class PeerIdentity
{
    internal static int PipePid(PipeStream pipe,bool serverEnd)
    {
        bool ok=serverEnd ? GetNamedPipeClientProcessId(pipe.SafePipeHandle,out uint id) : GetNamedPipeServerProcessId(pipe.SafePipeHandle,out id);
        PipeFrames.Check(ok && id>0,"IPC_KERNEL_PID"); return checked((int)id);
    }
    internal static (string Sid,bool Elevated,int Session,string Image) Facts(Process process)
    {
        if(!OpenProcessToken(process.SafeHandle,8,out var token)) throw new Win32Exception(Marshal.GetLastWin32Error());
        using(token)
        using(var owner=new WindowsIdentity(token.DangerousGetHandle()))
        {
            if(!GetTokenInformation(token,20,out int elevation,4,out _))throw new Win32Exception(Marshal.GetLastWin32Error());
            var image=new StringBuilder(1024); int capacity=image.Capacity;
            if(!QueryFullProcessImageNameW(process.SafeHandle,0,image,ref capacity))throw new Win32Exception(Marshal.GetLastWin32Error());
            return (owner.User?.Value ?? throw new InvalidDataException("IPC_SID"),elevation!=0,process.SessionId,image.ToString());
        }
    }
    internal static void Verify(PipeStream pipe,bool serverEnd,Process expected,string expectedImage)
    {
        PipeFrames.Check(!expected.HasExited && PipePid(pipe,serverEnd)==expected.Id,"IPC_PEER_PID");
        var peer=Facts(expected); using var current=Process.GetCurrentProcess(); var self=Facts(current);
        PipeFrames.Check(peer.Sid==self.Sid && peer.Session==self.Session && peer.Elevated==self.Elevated,"IPC_ACCOUNT_OR_ELEVATION");
        PipeFrames.Check(Path.GetFullPath(peer.Image).Equals(Path.GetFullPath(expectedImage),StringComparison.OrdinalIgnoreCase)
            && !Path.GetFileNameWithoutExtension(peer.Image).Equals("dotnet",StringComparison.OrdinalIgnoreCase),"IPC_IMAGE");
    }
    internal static void RequireNonElevated()
    {using var current=Process.GetCurrentProcess(); PipeFrames.Check(!Facts(current).Elevated,"ELEVATED_CONTEXT_BLOCKED");}
    internal static int OwnParentPid()
    {
        using var snapshot=CreateToolhelp32Snapshot(2,0); if(snapshot.IsInvalid)throw new Win32Exception(Marshal.GetLastWin32Error());
        var entry=new ProcessEntry{Size=(uint)Marshal.SizeOf<ProcessEntry>()};
        if(Process32FirstW(snapshot,ref entry))do {if(entry.Id==(uint)Environment.ProcessId)return checked((int)entry.Parent);}
            while(Process32NextW(snapshot,ref entry));
        throw new InvalidDataException("IPC_PARENT_UNKNOWN");
    }
    [StructLayout(LayoutKind.Sequential,CharSet=CharSet.Unicode)] private struct ProcessEntry
    {public uint Size,Usage,Id; public UIntPtr DefaultHeap; public uint Module,Threads,Parent; public int Priority; public uint Flags; [MarshalAs(UnmanagedType.ByValTStr,SizeConst=260)]public string Exe;}
    [DllImport("kernel32.dll",SetLastError=true)] [return:MarshalAs(UnmanagedType.Bool)] private static extern bool GetNamedPipeClientProcessId(SafePipeHandle p,out uint id);
    [DllImport("kernel32.dll",SetLastError=true)] [return:MarshalAs(UnmanagedType.Bool)] private static extern bool GetNamedPipeServerProcessId(SafePipeHandle p,out uint id);
    [DllImport("advapi32.dll",SetLastError=true)] [return:MarshalAs(UnmanagedType.Bool)] private static extern bool OpenProcessToken(SafeProcessHandle p,uint access,out SafeAccessTokenHandle token);
    [DllImport("advapi32.dll",SetLastError=true)] [return:MarshalAs(UnmanagedType.Bool)] private static extern bool GetTokenInformation(SafeAccessTokenHandle t,int kind,out int value,uint size,out uint returned);
    [DllImport("kernel32.dll",CharSet=CharSet.Unicode,SetLastError=true)] [return:MarshalAs(UnmanagedType.Bool)] private static extern bool QueryFullProcessImageNameW(SafeProcessHandle p,int flags,StringBuilder name,ref int size);
    [DllImport("kernel32.dll",SetLastError=true)] private static extern SafeFileHandle CreateToolhelp32Snapshot(uint flags,uint process);
    [DllImport("kernel32.dll",CharSet=CharSet.Unicode,SetLastError=true)] [return:MarshalAs(UnmanagedType.Bool)] private static extern bool Process32FirstW(SafeFileHandle snapshot,ref ProcessEntry entry);
    [DllImport("kernel32.dll",CharSet=CharSet.Unicode,SetLastError=true)] [return:MarshalAs(UnmanagedType.Bool)] private static extern bool Process32NextW(SafeFileHandle snapshot,ref ProcessEntry entry);
}
