using System.ComponentModel;
using System.Runtime.InteropServices;
using System.Runtime.Versioning;
using static GfUpdaterFixture.StrictMetadata;

namespace GfUpdaterFixture;

[SupportedOSPlatform("windows")]
internal sealed class PreparedTextExtraction : IDisposable
{
    private readonly List<IDisposable> pins=[];
    internal string DirectoryPath {get;private set;}="";
    internal bool Installable=>false;
    internal static PreparedTextExtraction Prepare(string root,byte[] archive,string authenticatedSha,
        Dictionary<string,string> authenticatedFiles,CancellationToken cancellation,Action<string,string>? hook=null,long? simulatedFreeBytes=null)
    {
        FixtureSandbox.RequireRoot(root);
        Require(archive.Length<=ZipInspection.MaxBytes,"ZIP_SIZE"); byte[] snapshot=archive.ToArray();
        Require(Hash(snapshot)==authenticatedSha,"ZIP_AUTHENTICATION");
        var files=ZipInspection.Inspect(snapshot,16,cancellation);
        Require(files.Keys.ToHashSet(StringComparer.Ordinal).SetEquals(authenticatedFiles.Keys),"EXTRACTION_SET");
        foreach(var p in files) Require(Hash(p.Value)==authenticatedFiles[p.Key],"EXTRACTION_HASH");
        long free=simulatedFreeBytes ?? new DriveInfo(Path.GetPathRoot(root)!).AvailableFreeSpace;
        Require(free>=files.Values.Sum(b=>(long)b.Length)+64L*1024*1024,"EXTRACTION_SPACE");
        var result=new PreparedTextExtraction();
        try
        {
            result.pins.Add(WindowsFileGuard.Open(root,root,true));
            string stage=Path.Combine(root,"u3-stage-"+Guid.NewGuid().ToString("N"));
            ExclusiveDirectory(stage); result.DirectoryPath=stage;
            result.pins.Add(WindowsFileGuard.Open(root,stage,true));
            var created=new HashSet<string>(StringComparer.OrdinalIgnoreCase){stage};
            foreach(var item in files)
            {
                cancellation.ThrowIfCancellationRequested();
                string target=Path.Combine(stage,item.Key.Replace('/',Path.DirectorySeparatorChar));
                Require(target.Length<=240,"EXTRACTION_PATH_LENGTH");
                string parent=stage;
                foreach(string part in item.Key.Split('/')[..^1])
                {
                    parent=Path.Combine(parent,part);
                    if(created.Add(parent))
                    {
                        ExclusiveDirectory(parent); hook?.Invoke("directory-created",parent);
                        result.pins.Add(WindowsFileGuard.Open(root,parent,true));
                    }
                }
                hook?.Invoke("before-file",target); cancellation.ThrowIfCancellationRequested();
                using(var output=new FileStream(target,FileMode.CreateNew,FileAccess.Write,FileShare.None,4096,FileOptions.WriteThrough))
                {output.Write(item.Value); output.Flush(true);}
                var pin=WindowsFileGuard.Open(root,target); result.pins.Add(pin);
                Require(Hash(pin.Read(ZipInspection.MaxExpanded))==authenticatedFiles[item.Key],"EXTRACTION_POST_HASH");
                hook?.Invoke("file-pinned",target);
            }
            cancellation.ThrowIfCancellationRequested(); return result;
        }
        catch {result.Dispose();throw;} // Preserve failed staging; never delete unknown paths to make an update pass.
    }
    private static void ExclusiveDirectory(string path)
    {if(!CreateDirectoryW(path,IntPtr.Zero))throw new Win32Exception(Marshal.GetLastWin32Error(),"EXTRACTION_DIRECTORY_EXISTS_OR_DENIED");}
    public void Dispose(){for(int i=pins.Count-1;i>=0;i--)pins[i].Dispose();pins.Clear();}
    [DllImport("kernel32.dll",CharSet=CharSet.Unicode,SetLastError=true)] [return:MarshalAs(UnmanagedType.Bool)] private static extern bool CreateDirectoryW(string path,IntPtr security);
}
