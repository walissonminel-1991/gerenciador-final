    private static Task Sync(Action action){action();return Task.CompletedTask;}
    private static async Task Rejects(Func<Task> action,string? code=null)
    {
        try{await action();}
        catch(Exception e) when(e is Rejected or IOException or InvalidDataException or System.ComponentModel.Win32Exception or OperationCanceledException
            or JsonException or System.Xml.XmlException or OverflowException)
        {if(code is not null)Require(e.Message==code,"WRONG_REJECTION:"+e.Message);return;}
        throw new Rejected("UNSAFE_ACCEPTANCE");
    }
    private static Task Rejects(Action action,string? code=null)=>Rejects(()=>Sync(action),code);

    private static byte[] ReplaceAscii(byte[] bytes,string before,string after)
    {
        Require(before.Length==after.Length,"TEST_MUTATION_LENGTH");byte[] from=Encoding.UTF8.GetBytes(before),to=Encoding.UTF8.GetBytes(after);int found=0;
        for(int i=0;i<=bytes.Length-from.Length;i++)if(bytes.AsSpan(i,from.Length).SequenceEqual(from)){to.CopyTo(bytes,i);found++;}
        Require(found>0,"TEST_RESOURCE_TEXT_MISSING");return bytes;
    }
    private static byte[] TextZip(params (string Name,string Text)[] entries)
    {
        using var buffer=new MemoryStream();using(var zip=new ZipArchive(buffer,ZipArchiveMode.Create,true))foreach(var item in entries)
        {var entry=zip.CreateEntry(item.Name,CompressionLevel.NoCompression);using var file=entry.Open();file.Write(Encoding.UTF8.GetBytes(item.Text));}
        return buffer.ToArray();
