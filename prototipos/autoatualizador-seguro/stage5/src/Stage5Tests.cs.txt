    private static async Task<bool> Test(string id,string description,Func<Task> action)
    {try{await action();Results.Add(new{id,description,status="passed"});Console.WriteLine("PASS "+id);return true;}catch(System.Net.Sockets.SocketException e) when(!OperatingSystem.IsWindows() && e.SocketErrorCode==System.Net.Sockets.SocketError.AccessDenied){Results.Add(new{id,description,status="blocked",reason="Linux AF_UNIX creation denied by environment",error_code=e.ErrorCode});Console.WriteLine("BLOCKED "+id);return false;}catch(Exception e){Results.Add(new{id,description,status="failed",error=e.ToString()});Console.WriteLine("FAIL "+id+" "+e);return false;}}
    private static async Task Canceled(Func<Task> action)
    {try{await action();}catch(OperationCanceledException){return;}throw new Rejected("DEADLINE_ACCEPTED_LATE_RESULT");}
    private static void Refuses(Action a,string code)
    {try{a();}catch(Rejected e){Require(e.Code==code,"WRONG_REFUSAL:"+e.Code);return;}throw new Rejected("UNSAFE_ACCEPTANCE:"+code);}
    private static async Task Drain(Task task)
    {try{await task.WaitAsync(TimeSpan.FromSeconds(5));}catch(Exception e) when(e is IOException or OperationCanceledException){} }
    private sealed class PipePair : IDisposable
    {
        internal readonly NamedPipeServerStream Server;
        internal readonly NamedPipeClientStream Client;
        private PipePair(string name)
        {
            Server=new(name,PipeDirection.InOut,1,PipeTransmissionMode.Byte,PipeOptions.Asynchronous|PipeOptions.CurrentUserOnly|PipeOptions.FirstPipeInstance);
            Client=new(".",name,PipeDirection.InOut,PipeOptions.Asynchronous|PipeOptions.CurrentUserOnly);
        }
        internal static async Task<PipePair> Connect()
        {
            var p=new PipePair("gf-u5-io-"+Guid.NewGuid().ToString("N"));
            try{using var c=new CancellationTokenSource(TimeSpan.FromSeconds(5));Task server=p.Server.WaitForConnectionAsync(c.Token);await p.Client.ConnectAsync(c.Token);await server;return p;}
            catch{p.Dispose();throw;}
        }
        public void Dispose(){Client.Dispose();Server.Dispose();}
    }

    private static async Task Common()
    {
        await Test("U5-IO-SHARED","Two real pipe reads share one monotonic deadline instead of restarting per phase",async()=>{
            using var pair=await PipePair.Connect();using var budget=StageDeadline.ShortFixture(TimeSpan.FromMilliseconds(1500));
            var sender=Task.Run(async()=>{await Task.Delay(300);await pair.Client.WriteAsync(new byte[]{1});await pair.Client.FlushAsync();});
            int first=await budget.ObserveRead(pair.Server.ReadAsync(new byte[1],budget.Token).AsTask());await sender;Require(first==1,"FIRST_PIPE_READ");
            Task<int> second=pair.Server.ReadAsync(new byte[1],budget.Token).AsTask();await Canceled(async()=>{_=await budget.ObserveRead(second);});await Drain(second);
            Require(budget.ElapsedSeconds>=1.4 && budget.ElapsedSeconds<8,"DEADLINE_MEASUREMENT");Observations.Add(new{kind="shared_real_io_deadline",configured_seconds=1.5,elapsed_seconds=Math.Round(budget.ElapsedSeconds,3),physical_storage_hang=false});
        });
        await Test("U5-IO-CANCEL","User cancellation reaches a pending real asynchronous pipe read",async()=>{
            using var pair=await PipePair.Connect();using var user=new CancellationTokenSource();using var budget=StageDeadline.Start(user.Token);
            Task<int> read=pair.Server.ReadAsync(new byte[1],budget.Token).AsTask();Require(!read.IsCompleted,"READ_NOT_PENDING");user.Cancel();await Canceled(async()=>{_=await budget.ObserveRead(read);});await Drain(read);
            Require(read.IsCanceled,"KERNEL_READ_NOT_CANCELED");Observations.Add(new{kind="cooperative_real_io_cancellation",read_task_canceled=read.IsCanceled});
        });
        await Test("U5-IO-LATE","A canceled wait does not stop underlying tokenless kernel I/O; late completion cannot pass Check",async()=>{
            using var pair=await PipePair.Connect();using var budget=StageDeadline.ShortFixture(TimeSpan.FromMilliseconds(200));
            Task<int> raw=pair.Server.ReadAsync(new byte[1],CancellationToken.None).AsTask();await Canceled(async()=>{_=await budget.ObserveRead(raw);});Require(!raw.IsCompleted,"RAW_IO_NOT_PENDING_AFTER_TIMEOUT");
            await pair.Client.WriteAsync(new byte[]{7});await pair.Client.FlushAsync();Require(await raw.WaitAsync(TimeSpan.FromSeconds(5))==1,"LATE_IO_NOT_OBSERVED");
            await Canceled(()=>{budget.Check();return Task.CompletedTask;});
            Observations.Add(new{kind="wait_only_cancellation_counterexample",raw_io_pending_after_deadline=true,late_completion_observed=true,forced_storage_abort_proved=false});
        });
        await Test("U5-IO-FILE","Asynchronous disk hashing reads a real bounded file under the same deadline",async()=>{
            string path=Path.Combine(suite,"inert-read.fixture");byte[] bytes=new byte[8*1024*1024];RandomNumberGenerator.Fill(bytes);await File.WriteAllBytesAsync(path,bytes);
            using var budget=StageDeadline.Start();await using var input=new FileStream(path,FileMode.Open,FileAccess.Read,FileShare.Read,65536,FileOptions.Asynchronous|FileOptions.SequentialScan);
            Require(await budget.Hash(input)==Hash(bytes),"ASYNC_FILE_HASH");
        });
    }
    [SupportedOSPlatform("windows")]

        await Test("U5-IO-120","The default 120 second deadline expires against an actual pending Windows pipe read",async()=>{

    private static BundleRef Candidate(StateCase state,Dictionary<string,string> recipe)
    {
        byte[] marker=StateCodec.Encode(new{schema="inert-pinned-lifecycle/5",recipe_sha256=Hash(StateCodec.Encode(recipe)),entrypoint=StaticProbeInspection.Exe});string digest=Hash(marker);
        string folder=Path.Combine(state.Root,"versions","2-"+digest);Directory.CreateDirectory(folder);string path=Path.Combine(folder,"bundle.fixture.txt");File.WriteAllBytes(path,marker);
        string identity="linux-fixture-model";if(OperatingSystem.IsWindows()){using var pin=WindowsFileGuard.Open(state.Root,path);identity=pin.Identity;}
        return new(2,"0.0.0.2",digest,identity);
    }
    [SupportedOSPlatform("windows")]
    private static void SharingRefuses(Action a)
    {try{a();}catch(IOException e){Require((e.HResult&65535) is 32 or 33,"NOT_A_SHARING_REFUSAL:"+(e.HResult&65535));return;}throw new Rejected("PIN_DID_NOT_DENY_MUTATION");}
    private static void Copy(string source,string target)
    {foreach(string path in Directory.GetFiles(source,"*",SearchOption.AllDirectories)){string next=Path.Combine(target,Path.GetRelativePath(source,path));Directory.CreateDirectory(Path.GetDirectoryName(next)!);File.Copy(path,next,false);}}
