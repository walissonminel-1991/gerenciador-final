    const string Host = "127.0.0.1";
    const string LocalPath = "/inert-only";
    const string PartName = "inert.partial";
    const string FinalName = "inert.fixture.bin";
    const int MaximumBytes = 131072;
    static readonly byte[] Payload = Encoding.ASCII.GetBytes(
        "GF-D1-INERT-PAYLOAD-NOT-A-ZIP-OR-EXECUTABLE\n");
    static readonly byte[] Digest = SHA256.HashData(Payload);
    const string ExpectedRelease =
        "https://github.com/walissonminel-1991/gerenciador-final/releases/download/"
        + "v5.13.11.66/GerenciadorFinal-portable-win-x64.zip";

    sealed record Reply(int Status = 200, string? Length = null,
        string? MediaType = null, byte[]? Body = null, int Fragment = 4096,
        int CutAfter = -1, int DelayBeforeHeadersMs = 0,
        int DelayBeforeBodyMs = 0, int DelayPerChunkMs = 0,
        bool OmitLength = false,
        bool ExtraHeader = false, bool Encoded = false,
        bool TransferChunked = false, bool Location = false);

    sealed class LoopbackServer : IAsyncDisposable
    {
        readonly TcpListener listener;
        readonly CancellationTokenSource closing = new();
        readonly Task serving;
        public int Port { get; }
        public int Accepts { get; private set; }

        public LoopbackServer(Reply reply)
        {
            listener = new TcpListener(IPAddress.Loopback,0);
            listener.Start(1);
            Port = ((IPEndPoint)listener.LocalEndpoint).Port;
            serving = Serve(reply);
        }

        async Task Serve(Reply r)
        {
            try
            {
                using var client = await listener.AcceptTcpClientAsync(closing.Token);
                Accepts++;
                if (!IPAddress.IsLoopback(((IPEndPoint)client.Client.RemoteEndPoint!).Address))
                    throw new InvalidOperationException("NON_LOOPBACK_PEER");
                using var stream = client.GetStream();
                var request = new byte[2048];
                var count = 0;
                while (count < request.Length)
                {
                    var n = await stream.ReadAsync(request.AsMemory(count),closing.Token);
                    if (n == 0) return;
                    count += n;
                    if (Encoding.ASCII.GetString(request,0,count).Contains("\r\n\r\n",
                        StringComparison.Ordinal)) break;
                }
                if (!Encoding.ASCII.GetString(request,0,count).StartsWith(
                    "GET " + LocalPath + " HTTP/1.1\r\n",StringComparison.Ordinal))
                    return;
                if (r.DelayBeforeHeadersMs > 0)
                    await Task.Delay(r.DelayBeforeHeadersMs,closing.Token);
                var body = r.Body ?? Payload;
                var length = r.Length ?? body.Length.ToString(CultureInfo.InvariantCulture);
                var reason = r.Status switch
                {
                    200 => "OK", 206 => "Partial Content", 301 => "Moved",
                    302 => "Found", 307 => "Temporary Redirect",
                    308 => "Permanent Redirect", 404 => "Not Found",
                    429 => "Too Many Requests", 500 => "Server Error", _ => "Other"
                };
                var response = new StringBuilder()
                    .Append("HTTP/1.1 ").Append(r.Status).Append(" ")
                    .Append(reason).Append("\r\n")
                    .Append("Content-Type: ")
                    .Append(r.MediaType ?? "application/octet-stream").Append("\r\n");
                if (!r.OmitLength) response.Append("Content-Length: ")
                    .Append(length).Append("\r\n");
                if (r.ExtraHeader) response.Append("X-Unexpected: yes\r\n");
                if (r.Encoded) response.Append("Content-Encoding: gzip\r\n");
                if (r.TransferChunked) response.Append("Transfer-Encoding: chunked\r\n");
                if (r.Location) response.Append("Location: https://evil.invalid/a\r\n");
                response.Append("Connection: close\r\n\r\n");
                await stream.WriteAsync(Encoding.ASCII.GetBytes(response.ToString()),
                    closing.Token);
                if (r.DelayBeforeBodyMs > 0)
                    await Task.Delay(r.DelayBeforeBodyMs,closing.Token);
                var size = r.CutAfter < 0 ? body.Length : Math.Min(r.CutAfter,body.Length);
                var chunk = Math.Clamp(r.Fragment,1,4096);
                for (var i=0; i<size; i+=chunk)
                {
                    await stream.WriteAsync(body.AsMemory(i,Math.Min(chunk,size-i)),
                        closing.Token);
                    await stream.FlushAsync(closing.Token);
                    if (r.DelayPerChunkMs > 0)
                        await Task.Delay(r.DelayPerChunkMs,closing.Token);
                }
            }
            catch (Exception ex) when (ex is IOException or SocketException or
                OperationCanceledException or ObjectDisposedException)
            {
                // Client abort or cancellation on an intentionally hostile fixture.
            }
        }

        public async ValueTask DisposeAsync()
        {
            closing.Cancel();
            listener.Stop();
            try { await serving.WaitAsync(TimeSpan.FromSeconds(3)); }
            catch (Exception ex) when (ex is OperationCanceledException or
                ObjectDisposedException or SocketException or TimeoutException) { }
            closing.Dispose();
        }
    }

    static void Verify(string file) => Verify(file,Payload.Length,Digest);

    static void Verify(string file,int authorizedLength,byte[] authorizedDigest)
    {
        var info = new FileInfo(file);
        if (info.Length != authorizedLength || info.Length > MaximumBytes ||
            (info.Attributes & FileAttributes.ReparsePoint) != 0)
            throw new InvalidDataException("READBACK_LENGTH_OR_LINK");
        using var stream = File.Open(file,FileMode.Open,FileAccess.Read,FileShare.Read);
        if (!CryptographicOperations.FixedTimeEquals(SHA256.HashData(stream),authorizedDigest))
            throw new InvalidDataException("READBACK_SHA");
    }

    [StructLayout(LayoutKind.Sequential)]
    struct NativeFileInfo
    {
        public uint Attributes;
        public System.Runtime.InteropServices.ComTypes.FILETIME Creation;
        public System.Runtime.InteropServices.ComTypes.FILETIME Access;
        public System.Runtime.InteropServices.ComTypes.FILETIME LastWrite;
        public uint VolumeSerial;
        public uint SizeHigh;
        public uint SizeLow;
        public uint LinkCount;
        public uint FileIndexHigh;
        public uint FileIndexLow;
    }

    [DllImport("kernel32.dll",SetLastError=true)]
    [return: MarshalAs(UnmanagedType.Bool)]
    static extern bool GetFileInformationByHandle(SafeFileHandle handle,
        out NativeFileInfo info);

    static (uint Volume,uint High,uint Low) FileIdentity(SafeFileHandle handle)
    {
        if (!GetFileInformationByHandle(handle,out var id))
            throw new IOException("CANARY_FILE_ID_UNAVAILABLE");
        return (id.VolumeSerial,id.FileIndexHigh,id.FileIndexLow);
    }

    // Asynchronous readback participates in the same acquisition deadline.
    // This does not make synchronous Windows rename atomic versus an attacker.
    static async Task VerifyAsync(string file,int expectedLength,
        byte[] expectedDigest,CancellationToken token)
    {
        token.ThrowIfCancellationRequested();
        var info = new FileInfo(file);
        if (info.Length != expectedLength || info.Length > MaximumBytes ||
            (info.Attributes & FileAttributes.ReparsePoint) != 0)
            throw new InvalidDataException("READBACK_LENGTH_OR_LINK");
        using var hasher = IncrementalHash.CreateHash(HashAlgorithmName.SHA256);
        await using var input = new FileStream(file,FileMode.Open,
            FileAccess.Read,FileShare.Read,4096,FileOptions.Asynchronous);
        var buffer = ArrayPool<byte>.Shared.Rent(4096);
        var length = 0;
        try
        {
            int n;
            while ((n = await input.ReadAsync(buffer.AsMemory(0,4096),token)) != 0)
            {
                length = checked(length + n);
                if (length > expectedLength || length > MaximumBytes)
                    throw new InvalidDataException("READBACK_EXCEEDS_LENGTH");
                hasher.AppendData(buffer,0,n);
            }
            token.ThrowIfCancellationRequested();
            if (length != expectedLength ||
                !CryptographicOperations.FixedTimeEquals(
                    hasher.GetHashAndReset(),expectedDigest))
                throw new InvalidDataException("READBACK_DIGEST");
        }
        finally { ArrayPool<byte>.Shared.Return(buffer); }
    }

    static Task Stage(int port,string identity,string folder,CancellationToken ct)
        => Stage(port,identity,folder,ct,Payload.Length,Digest);

    static async Task Stage(int port,string identity,string folder,CancellationToken ct,
        int authorizedLength,byte[] authorizedDigest)
    {
        if (identity != ExpectedRelease || port is < 1 or > 65535)
            throw new InvalidDataException("RELEASE_IDENTITY");
        using var deadline = CancellationTokenSource.CreateLinkedTokenSource(ct);
        deadline.CancelAfter(TimeSpan.FromSeconds(2));
        var token = deadline.Token;
        token.ThrowIfCancellationRequested();
        var handler = new SocketsHttpHandler
        {
            AllowAutoRedirect = false,
            UseProxy = false,
            UseCookies = false,
            AutomaticDecompression = DecompressionMethods.None,
            Credentials = null,
            PreAuthenticate = false,
            ConnectCallback = async (context,token) =>
            {
                if (context.DnsEndPoint.Host != Host || context.DnsEndPoint.Port != port)
                    throw new InvalidDataException("NON_LOOPBACK_DESTINATION");
                var socket = new Socket(AddressFamily.InterNetwork,SocketType.Stream,
                    ProtocolType.Tcp);
                try
                {
                    await socket.ConnectAsync(new IPEndPoint(IPAddress.Loopback,port),token);
                    if (!((IPEndPoint)socket.RemoteEndPoint!).Address.Equals(
                        IPAddress.Loopback))
                        throw new InvalidDataException("NON_LOOPBACK_REMOTE");
                    return new NetworkStream(socket,ownsSocket:true);
                }
                catch { socket.Dispose(); throw; }
            }
        };
        using var http = new HttpClient(handler,disposeHandler:true)
        { Timeout = TimeSpan.FromSeconds(2) };
        var requestUri = new Uri("http://" + Host + ":" +
            port.ToString(CultureInfo.InvariantCulture) + LocalPath);
        using var req = new HttpRequestMessage(HttpMethod.Get,requestUri)
        {
            Version = HttpVersion.Version11,
            VersionPolicy = HttpVersionPolicy.RequestVersionExact
        };
        req.Headers.AcceptEncoding.Clear();
        var tmp = System.IO.Path.Combine(folder,PartName);
        var final = System.IO.Path.Combine(folder,FinalName);
        {
            using var response = await http.SendAsync(req,
                HttpCompletionOption.ResponseHeadersRead,token);
            if (response.StatusCode != HttpStatusCode.OK ||
                response.Version != HttpVersion.Version11 ||
                response.Headers.Location != null ||
                response.Headers.TransferEncodingChunked == true ||
                response.Content.Headers.ContentEncoding.Count != 0 ||
                response.Content.Headers.ContentLength != authorizedLength ||
                response.Content.Headers.ContentType?.MediaType !=
                    "application/octet-stream" ||
                response.Headers.Any(x => !x.Key.Equals("Connection",
                    StringComparison.OrdinalIgnoreCase)))
                throw new InvalidDataException("HTTP_RESPONSE_NOT_AUTHORIZED");

            var total = 0;
            await using (var output = new FileStream(tmp,FileMode.CreateNew,
                FileAccess.Write,FileShare.None,4096,FileOptions.WriteThrough))
            await using (var remote = await response.Content.ReadAsStreamAsync(token))
            {
                using var hasher = IncrementalHash.CreateHash(HashAlgorithmName.SHA256);
                var buffer = ArrayPool<byte>.Shared.Rent(4096);
                try
                {
                    while (true)
                    {
                        var n = await remote.ReadAsync(buffer.AsMemory(0,4096),token);
                        if (n == 0) break;
                        total = checked(total+n);
                        if (total > authorizedLength || total > MaximumBytes)
                            throw new InvalidDataException("BODY_EXCEEDS_SIGNED_SIZE");
                        hasher.AppendData(buffer,0,n);
                        await output.WriteAsync(buffer.AsMemory(0,n),token);
                    }
                    if (total != authorizedLength ||
                        !CryptographicOperations.FixedTimeEquals(
                            hasher.GetHashAndReset(),authorizedDigest))
                        throw new InvalidDataException("BODY_DIGEST_MISMATCH");
                    token.ThrowIfCancellationRequested();
                    output.Flush(flushToDisk:true);
                }
                finally { ArrayPool<byte>.Shared.Return(buffer); }
            }
            await VerifyAsync(tmp,authorizedLength,authorizedDigest,token);
            token.ThrowIfCancellationRequested();
            File.Move(tmp,final,overwrite:false);
            await VerifyAsync(final,authorizedLength,authorizedDigest,token);
            // A failed attempt does not delete by name: either name could now
            // identify a preexisting or concurrently replaced object.
            // Preserve ambiguous artifacts for HOLD/diagnostic disposition.
        }
    }

    static async Task<int> Main(string[] args)
    {
#if D24_INTEGRATION
        if (args.Length > 0 && args[0] == "--d242")
            return await RunD242(args);
        if (args.Length > 0 && args[0] == "--d242-selftest")
            return RunD242SelfTest();
        // Direct C# handoff from D2.4 is permanently disabled.
        if (args.Length > 0) return 12;
#endif
        if (!OperatingSystem.IsWindows() ||
            Environment.GetEnvironmentVariable("GITHUB_ACTIONS") != "true") return 2;
        var temp = Environment.GetEnvironmentVariable("RUNNER_TEMP");
        if (string.IsNullOrWhiteSpace(temp) || !Directory.Exists(temp)) return 3;
        var lab = System.IO.Path.Combine(temp,"gf-d23-"+Guid.NewGuid().ToString("N"));
        Directory.CreateDirectory(lab);
        var sentinel = System.IO.Path.Combine(lab,"personal-fixture.txt");
        File.WriteAllText(sentinel,"PERSONAL_DATA_PRESERVED");
        var good=0;
        var bad=0;
        var accepted=0;
        var failedAccepted=0;
        var canariesPreserved=0;
        try
        {
            async Task Trial(string label, Reply reply,bool positive,
                bool cancelled=false,string? identity=null,string? preexisting=null)
            {
                await using var server = new LoopbackServer(reply);
                var stage = System.IO.Path.Combine(lab,"test-"+label);
                Directory.CreateDirectory(stage);
                var canaryContent=Encoding.ASCII.GetBytes("PRESERVE_"+label);
                string? canaryPath=null;
                FileStream? pinnedCanary=null;
                (uint Volume,uint High,uint Low)? originalId=null;
                if (preexisting!=null)
                {
                    canaryPath=System.IO.Path.Combine(stage,preexisting);
                    File.WriteAllBytes(canaryPath,canaryContent);
                    pinnedCanary=new FileStream(canaryPath,FileMode.Open,
                        FileAccess.Read,FileShare.ReadWrite | FileShare.Delete);
                    originalId=FileIdentity(pinnedCanary.SafeFileHandle);
                }
                try
                {
                    using var cts = new CancellationTokenSource();
                    if (cancelled) cts.CancelAfter(TimeSpan.FromMilliseconds(120));
                    try
                    {
                        await Stage(server.Port,identity??ExpectedRelease,stage,cts.Token);
                        if (!positive) throw new Exception("FALSE_ACCEPT_"+label);
                        Verify(System.IO.Path.Combine(stage,FinalName));
                        good++;
                    }
                    catch (Exception ex) when (!positive &&
                        ex is InvalidDataException or HttpRequestException or
                            OperationCanceledException or IOException)
                    {
                        if (preexisting!=FinalName &&
                            File.Exists(System.IO.Path.Combine(stage,FinalName)))
                            throw new Exception("FALSE_FINAL_FILE_"+label);
                        if (canaryPath!=null)
                        {
                            using var after=new FileStream(canaryPath,FileMode.Open,
                                FileAccess.Read,FileShare.ReadWrite | FileShare.Delete);
                            if (FileIdentity(after.SafeFileHandle)!=originalId ||
                                !File.ReadAllBytes(canaryPath).SequenceEqual(canaryContent))
                                throw new Exception("CANARY_IDENTITY_OR_BYTES_CHANGED_"+label);
                            canariesPreserved++;
                        }
                        if (label.StartsWith("deadline-",StringComparison.Ordinal) &&
                            !File.Exists(System.IO.Path.Combine(stage,PartName)))
                            throw new Exception("DEADLINE_FAILED_TO_RETAIN_PARTIAL_"+label);
                        bad++;
                    }
                }
                finally
                {
                    // Always count accepted connections, including rejected streams.
                    accepted += server.Accepts;
                    if (!positive) failedAccepted += server.Accepts;
                    pinnedCanary?.Dispose();
                    // Only runner-created test directories are cleaned after
                    // validating canary identity. Stage() never deletes by name.
                    Directory.Delete(stage,recursive:true);
                }
            }

            foreach(var part in new[]{1,7,4096})
                await Trial("valid-"+part,new Reply(Fragment:part),true);
            foreach(var status in new[]{301,302,307,308,206,404,429,500})
                await Trial("status-"+status,new Reply(Status:status),false);
            await Trial("html",new Reply(MediaType:"text/html"),false);
            await Trial("missing-length",new Reply(OmitLength:true),false);
            await Trial("short-length",new Reply(Length:(Payload.Length-1).ToString()),false);
            await Trial("long-length",new Reply(Length:(Payload.Length+1).ToString()),false);
            await Trial("zero-length",new Reply(Length:"0"),false);
            await Trial("negative-length",new Reply(Length:"-1"),false);
            await Trial("text-length",new Reply(Length:"unknown"),false);
            await Trial("huge-length",new Reply(Length:(MaximumBytes+1).ToString()),false);
            await Trial("encoded",new Reply(Encoded:true),false);
            await Trial("chunked",new Reply(TransferChunked:true),false);
            await Trial("location-on-200",new Reply(Location:true),false);
            await Trial("unexpected-header",new Reply(ExtraHeader:true),false);
            await Trial("truncated",new Reply(CutAfter:Payload.Length-2),false);
            var modified=Payload.ToArray(); modified[0]^=1;
            await Trial("changed-same-length",new Reply(Body:modified),false);
            await Trial("cancel-before-headers",new Reply(DelayBeforeHeadersMs:450),
                false,cancelled:true);
            await Trial("cancel-before-body",new Reply(DelayBeforeBodyMs:450),
                false,cancelled:true);
            await Trial("foreign-identity",new Reply(),false,
                identity:"https://evil.invalid/rogue.zip");
            await Trial("preexisting-partial",new Reply(),false,
                preexisting:PartName);
            await Trial("preexisting-final",new Reply(),false,
                preexisting:FinalName);
            await Trial("deadline-body-paused",
                new Reply(DelayBeforeBodyMs:3500),false);
            await Trial("deadline-slow-trickle",
                new Reply(Fragment:1,DelayPerChunkMs:300),false);

            if (good != 3 || bad != 29 ||
                canariesPreserved != 2 || failedAccepted != 28 ||
                File.ReadAllText(sentinel)!="PERSONAL_DATA_PRESERVED" ||
                accepted != 31)
                throw new Exception("TEST_MATRIX_OR_SCOPE_INCOMPLETE");
            var report=new {
                schema="gf-d23-dotnet-httpclient-loopback/1",status="passed",
                positive_inert_streams=good,negative_cases_rejected=bad,
                loopback_connections_observed=accepted,
                failed_loopback_connections_observed=failedAccepted,
                preserved_preexisting_file_ids=canariesPreserved,
                complete_acquisition_deadline_tested=true,
                failure_preserves_unknown_file_identity=true,
                http_client="dotnet10-System.Net.Http.HttpClient",
                only_numeric_loopback_connection=true,
                redirects_disabled=true,cancellation_tests_included=true,
                inert_sha256_readback_verified=true,
                outside_staging_sentinel_preserved=true,
                d0_d1_tuf_chain_verified_in_dotnet_lab=false,
                production_trust_root=false,github_asset_redirect_supported=false,
                tls_or_real_github_tested=false,
                same_user_toctou_resistance_proven=false,
                full_history_rollback_resistance_proven=false,
                real_executable_downloads=0,installations=0,product_modified=false,
                operational_ready=false,automatic_recovery_authorized=false
            };
            Console.WriteLine("GF_D23_RESULT "+JsonSerializer.Serialize(report));
            return 0;
        }
        finally { Directory.Delete(lab,recursive:true); }
    }
