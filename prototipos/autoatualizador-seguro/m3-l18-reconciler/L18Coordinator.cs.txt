// L18 prototype only: local-consistency reconciliation. No GF production code.
// TUF test keys authorize fixture bytes, not completion or independent freshness.
using System.ComponentModel;
using System.Diagnostics;
using System.Text.Json;
using static GfUpdaterFixture.StrictMetadata;

namespace GfUpdaterFixture;

internal enum L18Decision { Busy, UnknownBlocked, FailedKnown, OldVerified, LocalConsistentObserved }
internal sealed record L18PhaseRecord(string Schema,string Transaction,int Sequence,
    string Phase,string Detail,string TargetSha,string StagedIdentity,
    string BaselineDigest);

[System.Runtime.Versioning.SupportedOSPlatform("windows")]
internal static class L18Coordinator
{
    private const string Schema="gf-l18-coordinator-inert/1";
    private const string Start="old-verified", Authorizing="authorizing",
        Preparing="preparing", Staged="staged-verified", Intent="intent-durable",
        Switching="switching", Launched="owner-launched", Exited="owner-exited",
        Switched="switched-unconfirmed", Deadline="timeout", Unknown="unknown",
        Reconciling="reconciling", Old="old-observed",
        New="new-observed", Failure="failed-known";
    internal static string EventPath(string src,int sequence)
        =>Path.Combine(src,$"l18-phase-{sequence:D6}.fixture.json");

    private static void Root(string src,string lab)
    {
        FixtureSandbox.RequireRoot(src);
        L14Model.RequireRoot(lab);
    }
    private static bool Allowed(string previous,string next)=> (previous,next) switch
    {
        ("",Start)=>true,
        (Start,Authorizing)=>true,
        (Authorizing,Preparing)=>true,
        (Preparing,Staged)=>true,
        (Staged,Intent)=>true,
        (Intent,Switching)=>true,
        (Switching,Launched or Deadline or Unknown)=>true,
        (Launched,Exited or Deadline or Unknown)=>true,
        (Deadline,Unknown)=>true,
        (Exited,Switched or Reconciling or Unknown)=>true,
        (Switched,Reconciling or Unknown)=>true,
        (Reconciling,New or Old or Failure or Unknown or Reconciling)=>true,
        (Unknown,Reconciling)=>true,
        (New,Unknown)=>true,
        (Old,Unknown)=>true,
        (Failure,Reconciling or Unknown)=>true,
        _=>false
    };
    private static List<L18PhaseRecord> Load(string src,string lab)
    {
        Root(src,lab);
        var files=Directory.EnumerateFiles(src,"l18-phase-*.fixture.json")
            .OrderBy(x=>x,StringComparer.Ordinal).ToList();
        Require(files.Count<=256,"L18_JOURNAL_LIMIT");
        var records=new List<L18PhaseRecord>();
        for(int i=0;i<files.Count;i++)
        {
            Require(Path.GetFullPath(files[i]).Equals(
                Path.GetFullPath(EventPath(src,i+1)),StringComparison.OrdinalIgnoreCase),
                "L18_SEQUENCE_GAP_OR_NAME");
            using var pin=WindowsFileGuard.Open(src,files[i]);
            byte[] bytes=pin.Read(4096);
            using var doc=Json(bytes,4096);
            Require(doc.RootElement.ValueKind==JsonValueKind.Object &&
                doc.RootElement.EnumerateObject().Count()==8,"L18_JOURNAL_SHAPE");
            var item=StateCodec.Decode<L18PhaseRecord>(bytes);
            Require(item.Schema==Schema && item.Transaction==L14Model.Tx(lab) &&
                item.Sequence==i+1 && Hex(item.TargetSha,64) &&
                Hex(item.BaselineDigest,64) &&
                !string.IsNullOrEmpty(item.StagedIdentity) &&
                item.Detail is {Length:<=128} &&
                Allowed(i==0?"":records[^1].Phase,item.Phase),
                "L18_JOURNAL_FORK");
            if(i>0)Require(item.TargetSha==records[0].TargetSha &&
                item.StagedIdentity==records[0].StagedIdentity &&
                item.BaselineDigest==records[0].BaselineDigest,
                "L18_AUTHORIZATION_OR_STAGE_FORK");
            records.Add(item);
        }
        return records;
    }
    private static L18PhaseRecord Append(string src,string lab,string phase,
        string detail,string target)
    {
        var all=Load(src,lab);
        Require(all.Count<256 && Allowed(all.Count==0?"":all[^1].Phase,phase),
            "L18_TRANSITION_DENIED");
        string digest=all.Count==0?target:all[0].TargetSha;
        Require(Hex(digest,64) && (all.Count==0 || target==digest),
            "L18_TARGET_CHANGED");
        using var stagePin=WindowsFileGuard.Open(src,L16Boundary.Stage(src));
        string stagedId=stagePin.Identity;
        string oldDigest=L14Model.Hash(L14Model.OldText);
        if(all.Count>0)
            Require(stagedId==all[0].StagedIdentity &&
                oldDigest==all[0].BaselineDigest,"L18_STAGE_BASELINE_IDENTITY_FORK");
        var item=new L18PhaseRecord(Schema,L14Model.Tx(lab),all.Count+1,
            phase,detail,digest,stagedId,oldDigest);
        byte[] bytes=StateCodec.Encode(item);
        Require(bytes.Length is >0 and <=4096,"L18_EVENT_SIZE");
        using(var stream=new FileStream(EventPath(src,item.Sequence),
            FileMode.CreateNew,FileAccess.Write,FileShare.None,4096,FileOptions.WriteThrough))
        {
            stream.Write(bytes);stream.Flush(true);
        }
        // Confirm the record was durably observed by the local model.
        var verify=Load(src,lab);
        Require(verify.Count==item.Sequence && verify[^1]==item,
            "L18_EVENT_RECHECK");
        return item;
    }
    private sealed class PinnedFacts : IDisposable
    {
        private readonly WindowsFileGuard selectorPin,stagePin,oldPin;
        private readonly byte[] authorizedEntry,authorizedOld;
        internal string sha {get;}
        internal L16Selector selector {get;}
        internal string selectorIdentity {get;}
        internal string stagedIdentity {get;}
        internal string oldIdentity {get;}
        internal PinnedFacts(string digest,L16Selector selection,
            WindowsFileGuard selectorHandle,WindowsFileGuard stagedHandle,
            WindowsFileGuard oldHandle,byte[] entry,byte[] old)
        {
            sha=digest;selector=selection;
            selectorPin=selectorHandle;stagePin=stagedHandle;oldPin=oldHandle;
            authorizedEntry=(byte[])entry.Clone();
            authorizedOld=(byte[])old.Clone();
            selectorIdentity=selectorPin.Identity;
            stagedIdentity=stagePin.Identity;
            oldIdentity=oldPin.Identity;
        }
        internal void Recheck()
        {
            // L22: retain OLD as a live NTFS capability throughout the
            // complete reconciliation. Revalidate exact inert OLD bytes,
            // not only a historical digest in a local journal.
            oldPin.Recheck();
            byte[] oldNow=oldPin.Read(1024);
            Require(oldPin.Identity==oldIdentity &&
                oldNow.AsSpan().SequenceEqual(authorizedOld) &&
                Hash(oldNow)==L14Model.Hash(L14Model.OldText),
                "L22_OLD_BYTES_OR_ID_CHANGED");
            // L21: the exact signed staged bytes, not just the File ID, must
            // remain pinned until OLD/NEW is classified. NTFS sharing denies
            // writes/deletes while the handle is held (cooperative lab model).
            stagePin.Recheck();
            byte[] current=stagePin.Read(4096);
            Require(stagePin.Identity==stagedIdentity &&
                current.AsSpan().SequenceEqual(authorizedEntry) &&
                Hash(current)==sha,"L21_PINNED_STAGE_BYTES_CHANGED");
            selectorPin.Recheck();
            Require(selectorPin.Identity==selectorIdentity &&
                StateCodec.Decode<L16Selector>(selectorPin.Read(4096))==selector,
                "L20_SELECTOR_CHANGED_WHILE_DECIDING");
        }
        public void Dispose()
        {
            selectorPin.Dispose();
            stagePin.Dispose();
            oldPin.Dispose();
        }
    }

    // A same-user writer can still race the pathname BEFORE pin acquisition;
    // this guard binds the subsequent decision to one observed NTFS object.
    private static PinnedFacts Probe(string src,
        string lab,string bootstrap)
    {
        // Acquire OLD BEFORE L15 performs its lab-side ordinary read;
        // otherwise an in-place write could win between L15 and pinning.
        var oldPin=WindowsFileGuard.OpenLabOld(lab);
        try
        {
        byte[] old=System.Text.Encoding.UTF8.GetBytes(L14Model.OldText);
        Require(oldPin.Read(1024).AsSpan().SequenceEqual(old),
            "L22_OLD_NOT_EXPECTED_BYTES");
        var proof=L15Binding.Guard(lab,src,bootstrap);
        using var signedZip=WindowsFileGuard.Open(src,L15Binding.SignedZip(src));
        byte[] zip=signedZip.Read(ZipInspection.MaxBytes);
        Require(Hash(zip)==proof.SignedZipSha,"L18_ZIP_CHANGED");
        var entry=ZipInspection.Inspect(zip,3,CancellationToken.None)["bundle.fixture.txt"];
        Require(Hash(entry)==proof.SignedEntrySha && entry.Length<=4096,
            "L18_AUTHENTICATED_ENTRY");
        var stage=WindowsFileGuard.Open(src,L16Boundary.Stage(src));
        try
        {
            Require(stage.Read(4096).AsSpan().SequenceEqual(entry),
                "L18_STAGED_MISMATCH");
            var selectorPin=WindowsFileGuard.Open(src,L16Boundary.Selector(src));
            try
            {
                byte[] selected=selectorPin.Read(4096);
                using var doc=Json(selected,4096);
                Require(doc.RootElement.ValueKind==JsonValueKind.Object &&
                    doc.RootElement.EnumerateObject().Count()==3,"L18_SELECTOR_SHAPE");
                var selector=StateCodec.Decode<L16Selector>(selected);
                Require(selector.Schema=="gf-l16-selector-inert/1" &&
                    (selector is {Selected:"old",Generation:1 or 3} ||
                        selector is {Selected:"new",Generation:2}),"L18_SELECTOR_INVALID");
                stage.Recheck();selectorPin.Recheck();
                var pinned=new PinnedFacts(proof.SignedEntrySha,selector,
                    selectorPin,stage,oldPin,entry,old);
                pinned.Recheck();
                return pinned;
            }
            catch { selectorPin.Dispose();throw; }
        }
        catch { stage.Dispose();throw; }
        }
        catch { oldPin.Dispose();throw; }
    }
    // Local coordinator gate is distinct from the L16 owner's lease: it
    // serializes Begin/Execute/Reconcile among cooperative coordinators, even
    // while the owner child holds the L16 lease. Not same-user tamperproof.
    internal static string CoordinatorGatePath(string src)
        =>Path.Combine(src,"l20-coordinator-lease.fixture.txt");
    internal static FileStream? TryCoordinatorGate(string src)
    {
        FixtureSandbox.RequireRoot(src);
        string path=CoordinatorGatePath(src);
        if(!File.Exists(path))
        {
            try
            {
                using var created=new FileStream(path,FileMode.CreateNew,
                    FileAccess.Write,FileShare.None,4096,FileOptions.WriteThrough);
                byte[] marker=System.Text.Encoding.UTF8.GetBytes(
                    "GF-L20-COOPERATIVE-COORDINATOR-LEASE\n");
                created.Write(marker);created.Flush(true);
            }
            catch(IOException) when(File.Exists(path)) { }
        }
        // Verify no reparse/hardlink/foreign ACL before acquiring the
        // exclusive file lock. The close/reopen gap remains SAME-USER TOCTOU.
        try
        {
            using(var pin=WindowsFileGuard.Open(src,path))
                Require(System.Text.Encoding.UTF8.GetString(pin.Read(128))==
                    "GF-L20-COOPERATIVE-COORDINATOR-LEASE\n",
                    "L20_GATE_MARKER_INVALID");
            return new FileStream(path,FileMode.Open,FileAccess.ReadWrite,
                FileShare.None,4096,FileOptions.WriteThrough);
        }
        catch(Win32Exception ex) when(ex.NativeErrorCode is 32 or 33)
        { return null; } // cooperative Windows sharing violation
        catch(IOException ex) when((ex.HResult&0xffff) is 32 or 33)
        { return null; }
    }

    internal static void Begin(string src,string lab,string bootstrap)
    {
        Root(src,lab);
        using var gate=TryCoordinatorGate(src);
        Require(gate is not null,"L20_COORDINATOR_BUSY");
        using var ownerLease=L16Boundary.Exclusive(src);
        BeginCore(src,lab,bootstrap);
    }

    private static void BeginCore(string src,string lab,string bootstrap)
    {
        Root(src,lab);
        Require(!File.Exists(L16Boundary.Uncertain(src)) &&
            !File.Exists(RecoveryBlocked(src)) &&
            Load(src,lab).Count==0,"L20_BEGIN_BLOCKED");
        using var facts=Probe(src,lab,bootstrap);
        Require(facts.selector is {Selected:"old",Generation:1},
            "L18_BEGIN_NOT_OLD");
        Append(src,lab,Start,"old-observed-before-start",facts.sha);
        Append(src,lab,Authorizing,"stage11-stage3-proof-checked",facts.sha);
        // L16 Prepare already created staged bytes. PREPARING in this wrapper
        // verifies that PRE-EXISTING fixture; it does not install real files.
        Append(src,lab,Preparing,"l16-prestaged-fixture-only",facts.sha);
        Append(src,lab,Staged,"exact-signed-entry-staged-id-pinned",facts.sha);
        // Persist intention BEFORE ANY owner that might modify the selector.
        Append(src,lab,Intent,"owner-may-be-launched",facts.sha);
        facts.Recheck();
    }
    private static string ValidateExecutionArguments(string executable,TimeSpan deadline)
    {
        Require(deadline>TimeSpan.Zero && deadline<=TimeSpan.FromMinutes(1),
            "L18_DEADLINE_SCOPE");
        Require(!string.IsNullOrWhiteSpace(executable),"L21_EXE_REQUIRED");
        string own=Path.GetFullPath(executable);
        string current=Path.GetFullPath(Environment.ProcessPath ??
            throw new Rejected("L21_PROCESS_IMAGE_UNAVAILABLE"));
        Require(own.Equals(current,StringComparison.OrdinalIgnoreCase) &&
            Path.GetFileName(own).Equals("GF.Updater.M3.L18Tests.exe",
                StringComparison.OrdinalIgnoreCase),"L18_EXE_ALLOWLIST");
        // Test-only identity is the currently running fixture image at an
        // exact path, NOT a signature or Windows file-ID attestation. No
        // external image is authorized, and no owner launch may precede this.
        Require(File.Exists(own) &&
            (File.GetAttributes(own)&(FileAttributes.Directory|
                FileAttributes.ReparsePoint))==0,"L21_EXE_IMAGE_TYPE");
        return own;
    }

    internal static async Task<L18Decision> Execute(string src,string lab,
        string bootstrap,string executable,string cutpoint,TimeSpan deadline,
        bool cutDuringRecovery=false)
    {
        Root(src,lab);
        // L21: reject invalid owner image/deadline BEFORE touching a gate,
        // transaction journal, selector, or starting the owner process.
        string own=ValidateExecutionArguments(executable,deadline);
        using var gate=TryCoordinatorGate(src);
        if(gate is null)return L18Decision.Busy;
        // The owner hasn't started yet: pin its exclusive lease during
        // preparation, then release it BEFORE launching the child.
        using(var preparationLease=L16Boundary.Exclusive(src))
            BeginCore(src,lab,bootstrap);
        Append(src,lab,Switching,"switch-can-start-only-after-intent",
            Load(src,lab)[0].TargetSha);
        var info=new ProcessStartInfo(own){
            UseShellExecute=false,CreateNoWindow=true,
            RedirectStandardOutput=true,RedirectStandardError=true,
            WorkingDirectory=Path.GetTempPath()
        };
        foreach(string part in new[]{"--owner",src,lab,bootstrap,cutpoint})
            info.ArgumentList.Add(part);
        using var owner=Process.Start(info) ??
            throw new InvalidDataException("L18_OWNER_START_FAILED");
        // PID is diagnostic context, not a security credential.
        Append(src,lab,Launched,"pid-"+owner.Id.ToString(System.Globalization.CultureInfo.InvariantCulture),
            Load(src,lab)[0].TargetSha);
        try
        {
            // Deterministic fixture handshake: deadline starts only once the
            // controlled L16 child reached the late-completion cut. This is
            // a test barrier, NOT a production deadline or readiness signal.
            if(cutpoint=="after-replace-resume")
            {
                string? ready=await owner.StandardOutput.ReadLineAsync()
                    .WaitAsync(TimeSpan.FromSeconds(30));
                Require(ready=="READY:L17-LATE","L18_LATE_CUT_UNREACHED");
            }
            try { await owner.WaitForExitAsync().WaitAsync(deadline); }
            catch(TimeoutException)
            {
                // The coordinator, NOT the test supervisor, owns the timeout.
                // It persists a warning BEFORE returning, while the owner
                // may still be writing. Never kill/assume aborted I/O.
                Append(src,lab,Deadline,"wait-expired-owner-may-continue",
                    Load(src,lab)[0].TargetSha);
                L16Boundary.MarkUncertain(src);
                Append(src,lab,Unknown,"timeout-effects-not-reconciled",
                    Load(src,lab)[0].TargetSha);
                return L18Decision.UnknownBlocked;
            }
            Append(src,lab,Exited,owner.ExitCode==0?"exit-code-zero":
                "exit-code-nonzero",Load(src,lab)[0].TargetSha);
            if(owner.ExitCode==0)
                Append(src,lab,Switched,"switched-pending-reconciliation",
                    Load(src,lab)[0].TargetSha);
            return ReconcileCore(src,lab,bootstrap,cutDuringRecovery,null);
        }
        catch
        {
            // Incomplete switch-intent remains blocking even if this
            // coordinator dies before it can write another event.
            throw;
        }
    }

    // Fallback for incomplete/unreadable append-only journals. A local
    // blocking marker is cooperative, not cryptographically independent.
    // Never clear it because a process vanished or a timeout expired.
    internal static string RecoveryBlocked(string src)
        =>Path.Combine(src,"l19-recovery-blocked.fixture.txt");

    private static void WriteFallbackBlock(string src)
    {
        var path=RecoveryBlocked(src);
        try
        {
            byte[] data=System.Text.Encoding.UTF8.GetBytes(
                "GF-L19-LOCAL-UNRECOVERABLE-JOURNAL-V1\n");
            using var writer=new FileStream(path,FileMode.CreateNew,
                FileAccess.Write,FileShare.None,4096,FileOptions.WriteThrough);
            writer.Write(data);writer.Flush(true);
        }
        catch(IOException) when(File.Exists(path))
        {
            // A pre-existing marker remains blocking. Never overwrite it.
        }
        // If recording the blocker also failed, propagate the exception:
        // cannot claim a durable fail-closed outcome without durable state.
    }

    private static void RememberUnknown(string src,string lab,string why)
    {
        try
        {
            var history=Load(src,lab);
            if(history.Count>0)
            {
                if(history[^1].Phase==Unknown)return;
                if(Allowed(history[^1].Phase,Unknown))
                {
                    Append(src,lab,Unknown,why,history[0].TargetSha);
                    return;
                }
            }
        }
        catch(Exception e) when(e is Rejected or InvalidDataException or
            IOException or UnauthorizedAccessException or JsonException or
            Win32Exception or ArgumentException)
        { /* Incomplete/corrupt journal -> separate conservative marker. */ }
        WriteFallbackBlock(src);
    }

    private static bool AuxiliaryTerminalMatches(string src,string lab,L16Selector selector)
    {
        if(!File.Exists(L16Boundary.Terminal(src)))return false;
        using var terminalPin=WindowsFileGuard.Open(src,L16Boundary.Terminal(src));
        byte[] raw=terminalPin.Read(4096);
        using var doc=Json(raw,4096);
        Require(doc.RootElement.ValueKind==JsonValueKind.Object &&
            doc.RootElement.EnumerateObject().Count()==3,
            "L19_AUX_TERMINAL_SHAPE");
        var receipt=StateCodec.Decode<L16Terminal>(raw);
        return receipt.Schema=="gf-l16-terminal-inert/1" &&
            receipt.Transaction==L14Model.Tx(lab) &&
            receipt.SelectorSha==Hash(StateCodec.Encode(selector));
    }

    internal static L18Decision Reconcile(string src,string lab,string bootstrap,
        bool cutAfterCheckpoint=false,Action? selectorObserved=null)
    {
        Root(src,lab);
        using var gate=TryCoordinatorGate(src);
        if(gate is null)return L18Decision.Busy;
        return ReconcileCore(src,lab,bootstrap,cutAfterCheckpoint,selectorObserved);
    }

    private static L18Decision ReconcileCore(string src,string lab,
        string bootstrap,bool cutAfterCheckpoint,Action? selectorObserved)
    {
        Root(src,lab);
        FileStream lease;
        try {lease=L16Boundary.Exclusive(src);}
        catch(IOException) {return L18Decision.Busy;}
        using(lease)
        {
            try
            {
                if(File.Exists(RecoveryBlocked(src)) ||
                    File.Exists(L16Boundary.Uncertain(src)))
                    return L18Decision.UnknownBlocked;

                var events=Load(src,lab);
                // Do NOT evaluate events[^1] before handling empty journals.
                // The previously authenticated L15 binding validates OLD.
                using var facts=Probe(src,lab,bootstrap);
                selectorObserved?.Invoke(); // deterministic fixture interleaving
                facts.Recheck();
                if(events.Count==0)
                {
                    if(facts.selector is {Selected:"old",Generation:1} &&
                        !File.Exists(L16Boundary.Intent(src)) &&
                        !File.Exists(L16Boundary.Terminal(src)))
                    { facts.Recheck(); return L18Decision.OldVerified; }
                    RememberUnknown(src,lab,"no-journal-inconsistent");
                    return L18Decision.UnknownBlocked;
                }

                bool exited=events.Any(x=>x.Phase==Exited);
                bool timedOut=events.Any(x=>x.Phase==Deadline);
                string last=events[^1].Phase;
                if(events[0].TargetSha!=facts.sha ||
                    events[0].BaselineDigest!=L14Model.Hash(L14Model.OldText))
                {
                    RememberUnknown(src,lab,"signed-baseline-disagrees");
                    return L18Decision.UnknownBlocked;
                }
                using(var stagePin=WindowsFileGuard.Open(src,L16Boundary.Stage(src)))
                    if(stagePin.Identity!=events[0].StagedIdentity)
                    {
                        RememberUnknown(src,lab,"staged-object-identity-changed");
                        return L18Decision.UnknownBlocked;
                    }

                // L19 has NO reliable owner-exit/complete-I/O witness for a
                // timed-out owner. Even if the owner later writes a receipt,
                // do not infer that all effects ended.
                if(timedOut || !exited)
                {
                    RememberUnknown(src,lab,timedOut?"deadline-not-reconciled":
                        "owner-exit-not-observed");
                    return L18Decision.UnknownBlocked;
                }
                string outcome=events.Last(x=>x.Phase==Exited).Detail;
                bool oldIsObserved=facts.selector.Selected=="old" &&
                    facts.selector.Generation is 1 or 3 &&
                    !File.Exists(L16Boundary.Terminal(src));
                bool newIsObserved=outcome=="exit-code-zero" &&
                    facts.selector is {Selected:"new",Generation:2} &&
                    AuxiliaryTerminalMatches(src,lab,facts.selector);

                // An already classified state is idempotently observable.
                // Conflicts after the outcome are persisted, not just returned.
                if(last==New && newIsObserved)
                { facts.Recheck();return L18Decision.LocalConsistentObserved; }
                if(last==Old && oldIsObserved)
                { facts.Recheck();return L18Decision.OldVerified; }
                if(last==Failure && outcome=="exit-code-nonzero" &&
                    oldIsObserved)
                {
                    Append(src,lab,Reconciling,"reverify-old-after-known-failure",
                        facts.sha);
                    Append(src,lab,Old,"old-verified-after-failure",facts.sha);
                    facts.Recheck();
                    return L18Decision.OldVerified;
                }

                // UNKNOWN is a historical event, not an irreversible state.
                // Only a previously observed owner exit plus fresh signed
                // content/selector and non-conflicting auxiliary evidence
                // can justify another controlled reconciliation.
                if(newIsObserved || (outcome=="exit-code-nonzero" && oldIsObserved))
                {
                    if(last!=Reconciling)
                        Append(src,lab,Reconciling,"reobserve-after-owner-exit",
                            facts.sha);
                    if(cutAfterCheckpoint)
                    {
                        Console.WriteLine("CUT:L18-RECONCILE");Console.Out.Flush();
                        Thread.Sleep(Timeout.Infinite);
                    }
                    // Re-observe after persisting intent; no blind replay.
                    // Keep BOTH selector handles pinned until the decision;
                    // a changed namespace cannot be accepted between reads.
                    using var refreshed=Probe(src,lab,bootstrap);
                    Require(refreshed.selectorIdentity==facts.selectorIdentity &&
                        refreshed.selector==facts.selector,
                        "L20_SELECTOR_IDENTITY_FORK");
                    newIsObserved=outcome=="exit-code-zero" &&
                        refreshed.selector is {Selected:"new",Generation:2} &&
                        AuxiliaryTerminalMatches(src,lab,refreshed.selector);
                    oldIsObserved=refreshed.selector.Selected=="old" &&
                        refreshed.selector.Generation is 1 or 3 &&
                        !File.Exists(L16Boundary.Terminal(src));
                    if(newIsObserved)
                    {
                        Append(src,lab,New,"local-fixture-consistent",facts.sha);
                        facts.Recheck();refreshed.Recheck();
                        return L18Decision.LocalConsistentObserved;
                    }
                    if(outcome=="exit-code-nonzero" && oldIsObserved)
                    {
                        // Retain the preexisting semantic of FAILED_KNOWN first.
                        if(last!=Failure && !events.Any(x=>x.Phase==Failure))
                        {
                            Append(src,lab,Failure,"known-owner-error-old-observed",
                                facts.sha);
                            facts.Recheck();refreshed.Recheck();
                            return L18Decision.FailedKnown;
                        }
                        Append(src,lab,Old,"old-reverified-after-failure",facts.sha);
                        facts.Recheck();refreshed.Recheck();
                        return L18Decision.OldVerified;
                    }
                }

                RememberUnknown(src,lab,"current-local-evidence-conflicts");
                return L18Decision.UnknownBlocked;
            }
            catch(Exception ex) when(ex is Rejected or InvalidDataException or
                IOException or UnauthorizedAccessException or JsonException or
                Win32Exception or ArgumentException)
            {
                RememberUnknown(src,lab,"reconcile-error");
                return L18Decision.UnknownBlocked;
            }
        }
    }
