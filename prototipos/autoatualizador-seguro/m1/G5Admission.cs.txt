// M1: isolated, fail-closed admission check. Not a Gerenciador Final updater.
using System.Diagnostics;
using System.IO.Pipes;
using System.Runtime.Versioning;
using System.Security.Cryptography;
using static GfUpdaterFixture.PipeFrames;

namespace GfUpdaterFixture;

internal sealed record G5Trust(
    string ImagePath, string ExecutableSha256, int ChildPid,
    string UserSid, int SessionId, bool Elevated, string Transaction, string Nonce);

[SupportedOSPlatform("windows")]
internal sealed class G5Admission
{
    private readonly G5Trust trust;
    private readonly Stopwatch timer = Stopwatch.StartNew();
    private readonly TimeSpan limit;
    private bool accepted;

    internal G5Admission(G5Trust trust, TimeSpan limit)
    {
        Check(limit > TimeSpan.Zero && limit <= TimeSpan.FromMinutes(2), "M1_BAD_DEADLINE");
        Check(!trust.Elevated, "M1_ELEVATED_CONTEXT");
        Check(trust.ChildPid > 0 && trust.SessionId >= 0 &&
            !string.IsNullOrWhiteSpace(trust.UserSid), "M1_BAD_IDENTITY");
        Check(Path.IsPathFullyQualified(trust.ImagePath) &&
            Path.GetExtension(trust.ImagePath).Equals(".exe", StringComparison.OrdinalIgnoreCase) &&
            !Path.GetFileNameWithoutExtension(trust.ImagePath).Equals("dotnet", StringComparison.OrdinalIgnoreCase),
            "M1_BAD_IMAGE");
        Check(Hex(trust.ExecutableSha256, 64) && Hex(trust.Transaction, 32) && Hex(trust.Nonce, 64),
            "M1_BAD_BINDING");
        this.trust = trust;
        this.limit = limit;
    }

    // Only call after a real pipe connection, with the process handle retained by the creator.
    // Neither process arguments nor received frames supply the observed OS identity.
    internal void Accept(NamedPipeServerStream server, Process expectedChild, ProbeFrame response)
    {
        Check(!accepted, "M1_READY_REPLAY");
        Check(timer.Elapsed < limit, "M1_DEADLINE");
        Check(server.IsConnected && !expectedChild.HasExited && expectedChild.Id == trust.ChildPid,
            "M1_EXPECTED_PROCESS");
        PeerIdentity.Verify(server, true, expectedChild, trust.ImagePath);
        var facts = PeerIdentity.Facts(expectedChild);
        // Fixture identity is a hash of the actual image opened by the verified child.
        // This is NOT the pinned full-bundle identity required by an operational adapter.
        using var file = new FileStream(facts.Image, FileMode.Open, FileAccess.Read,
            FileShare.Read | FileShare.Delete, 65536, FileOptions.SequentialScan);
        var digest = Convert.ToHexStringLower(SHA256.HashData(file));
        G5ContextPolicy.Demand(trust, new G5Observed(expectedChild.Id, facts.Sid,
            facts.Session, facts.Elevated, facts.Image, digest));
        Check(timer.Elapsed < limit, "M1_DEADLINE");
        Bound(response, "ready", trust.Transaction, trust.Nonce);
        accepted = true;
    }
}


// Pure decision kernel, always fed from OS observations in G5Admission.Accept.
// Tests with invented observations are unit tests ONLY, not foreign-user handshakes.
internal sealed record G5Observed(int Pid, string Sid, int Session, bool Elevated,
    string ImagePath, string ExecutableSha256);
internal static class G5ContextPolicy
{
    internal static void Demand(G5Trust trusted, G5Observed observed)
    {
        PipeFrames.Check(trusted.ChildPid == observed.Pid, "M1_EXPECTED_PROCESS");
        PipeFrames.Check(trusted.UserSid == observed.Sid &&
            trusted.SessionId == observed.Session && trusted.Elevated == observed.Elevated,
            "M1_USER_SESSION");
        PipeFrames.Check(Path.GetFullPath(trusted.ImagePath).Equals(
            Path.GetFullPath(observed.ImagePath), StringComparison.OrdinalIgnoreCase), "M1_IMAGE");
        PipeFrames.Check(trusted.ExecutableSha256 == observed.ExecutableSha256, "M1_INSTALLATION_ID");
    }
}
