    private static async Task<bool> Test(string id,string description,Func<Task> action)
    {try{await action();Results.Add(new{id,description,status="passed"});Console.WriteLine("PASS "+id);return true;}catch(Exception e){Results.Add(new{id,description,status="failed",error=e.ToString()});Console.WriteLine("FAIL "+id+" "+e);return false;}}
    private static Task Sync(Action a){a();return Task.CompletedTask;}
    private static void Refuses(Action a,string code)
    {try{a();}catch(Rejected e){Require(e.Code==code,"WRONG_REJECTION:"+e.Code);return;}throw new Rejected("UNSAFE_ACCEPTANCE:"+code);}
    private static async Task RefusesAsync(Func<Task> a)
    {try{await a();}catch(Exception e) when(e is Rejected or InvalidDataException or IOException or OperationCanceledException){return;}throw new Rejected("UNSAFE_ACCEPTANCE");}
    public static async Task<int> Main(string[] args)
    {
        if(args.Length!=5)return 2;fixtures=args[0];probe=args[1];recipePath=args[2];sdkHost=args[3];suite=Path.Combine(Path.GetTempPath(),"gf-u2-u4-"+Guid.NewGuid().ToString("N"));Directory.CreateDirectory(suite);
        try{await Common();if(OperatingSystem.IsWindows())await Windows();}
        finally{Directory.Delete(suite,true);}
        int failed=Results.Count(x=>JsonSerializer.SerializeToElement(x).GetProperty("status").GetString()=="failed");
        var report=new{schema=1,stage=4,contract="gf-self-update/1",fixture_only=true,package_installable=false,product_built=false,product_executed=false,
            runtime=RuntimeInformation.FrameworkDescription,os=RuntimeInformation.OSDescription,total=Results.Count,passed=Results.Count-failed,failed,tests=Results,observations=Observations};
        await File.WriteAllBytesAsync(args[4],StateCodec.Encode(report));return failed==0?0:1;
    }
    private static byte[] Zip(params (string Name,byte[] Data)[] entries)
    {using var m=new MemoryStream();using(var z=new ZipArchive(m,ZipArchiveMode.Create,true))foreach(var x in entries){using var f=z.CreateEntry(x.Name,CompressionLevel.NoCompression).Open();f.Write(x.Data);}return m.ToArray();}

    [SupportedOSPlatform("windows")]
    private static async Task Windows()
    {
        using var c=new StateCase(suite);var recipe=JsonSerializer.Deserialize<Dictionary<string,string>>(File.ReadAllText(recipePath))!;
        string archive=Path.Combine(c.Root,"inert.zip");using(var stream=File.Create(archive))using(var z=new ZipArchive(stream,ZipArchiveMode.Create))foreach(var item in recipe){var e=z.CreateEntry(item.Key,CompressionLevel.Fastest);using var f=e.Open();using var src=File.OpenRead(Path.Combine(probe,item.Key));src.CopyTo(f);}
        using var a=File.OpenRead(archive);string sha=StreamPackage.StreamHash(a);a.Dispose();string? extracted=null;VerifiedInertProbe? proof=null;
        bool prepared=await Test("U4-PACKAGE-01","Full real inert runtime bundle is streamed to new NTFS staging and statically inspected",()=>Sync(()=>{
            extracted=StreamPackage.ExtractInert(c.Root,archive,sha,recipe,CancellationToken.None);proof=VerifiedInertProbe.Inspect(c.Root,extracted,recipe);
            Observations.Add(new{kind="full_inert_package",files=recipe.Count,expanded_bytes=Directory.GetFiles(extracted,"*",SearchOption.AllDirectories).Sum(p=>new System.IO.FileInfo(p).Length),zip_bytes=new System.IO.FileInfo(archive).Length,zip_sha256=sha,gf_package=false});
        }));
        await Test("U4-AUTH-01","Real unsigned probe is accepted only in explicit beta-unsigned fixture mode",()=>Sync(()=>{
            string path=Path.Combine(probe,StaticProbeInspection.Exe);bool cert=OfflineAuthenticode.HasCertificate(File.ReadAllBytes(path));int status=OfflineAuthenticode.Verify(path);OfflineAuthenticode.RequireBetaUnsigned(cert,status);Observations.Add(new{kind="unsigned_authenticode",certificate_table=cert,status_hex=unchecked((uint)status).ToString("x8"),revocation="offline-cache-only; online freshness not proved"});
        }));
        await Test("U4-AUTH-02","SDK-signed host and a tampered copy receive real offline WinVerifyTrust verdicts",()=>Sync(()=>{
            byte[] original=File.ReadAllBytes(sdkHost);Require(OfflineAuthenticode.HasCertificate(original),"SDK_HOST_NOT_EMBEDDED_SIGNED");string path=Path.Combine(c.Root,"signed-sdk.fixture.exe");File.WriteAllBytes(path,original);int clean=OfflineAuthenticode.Verify(path);int pe=BitConverter.ToInt32(original,60),table=pe+24+BitConverter.ToUInt16(original,pe+20);int raw=BitConverter.ToInt32(original,table+20);Require(raw>0 && raw<original.Length,"AUTH_MUTATION_RANGE");original[raw]^=1;File.WriteAllBytes(path,original);int tampered=OfflineAuthenticode.Verify(path);Require(tampered!=0,"AUTH_TAMPER_ACCEPTED");Refuses(()=>OfflineAuthenticode.RequireBetaUnsigned(true,tampered),"AUTH_UNSIGNED_DOWNGRADE_BLOCKED");
            if(clean==0)Refuses(()=>OfflineAuthenticode.RequireProductionPublisher(clean),"GF_PUBLISHER_POLICY_NOT_PROVISIONED");else Refuses(()=>OfflineAuthenticode.RequireProductionPublisher(clean),"AUTH_TRUST_FAILED");
            Observations.Add(new{kind="signed_sdk_authenticode",clean_status_hex=unchecked((uint)clean).ToString("x8"),tampered_status_hex=unchecked((uint)tampered).ToString("x8"),windows_trust_only=true,gf_publisher_approved=false,trust_store_modified=false});
        }));
        await Test("U4-IPC-CONTEXT","Read-only token/session capability inventory; elevated operation is refused before spawn",()=>Sync(()=>{
            using var self=Process.GetCurrentProcess();var facts=PeerIdentity.Facts(self);int before=BoundLifecycleChannel.Launches;
            if(facts.Elevated)Refuses(()=>{try{PeerIdentity.RequireNonElevated();}catch(InvalidDataException e){throw new Rejected(e.Message);}},"ELEVATED_CONTEXT_BLOCKED");else PeerIdentity.RequireNonElevated();Require(before==BoundLifecycleChannel.Launches,"ELEVATED_SPAWN");
            int count=0;bool enumerated=WTSEnumerateSessionsW(IntPtr.Zero,0,1,out var sessions,out int n);if(enumerated){count=n;WTSFreeMemory(sessions);}
            Observations.Add(new{kind="ipc_context_capabilities",elevated=facts.Elevated,wts_enumerated=enumerated,visible_session_count=count,non_elevated_positive_executed=!facts.Elevated,foreign_session_handshake_executed=false,accounts_or_tokens_created=false});
        }));
        if(!prepared || proof is null){Observations.Add(new{kind="lifecycle_not_executed",reason="static package gate failed"});return;}

        foreach(string mode in new[]{"normal","wrong-nonce","early-exit"})

            string m=mode;await Test("U4-LIFE-"+mode,"Isolated real probe lifecycle bound to the persisted transaction: "+mode,async()=>{

                await using(var old=await BoundLifecycleChannel.Open(verified,"normal",Guid.NewGuid().ToString("N"),Convert.ToHexStringLower(RandomNumberGenerator.GetBytes(32)))){await old.Ready();await old.Close();Require(old.Child.HasExited,"OLD_STILL_LIVE");}
                var selected=session.Begin(candidate,state.Clock(),DateTimeOffset.UtcNow+TimeSpan.FromHours(1),true);var tx=selected.Transaction!;string nonce=Hash(Encoding.UTF8.GetBytes(tx.Nonce+candidate.Sha256));
                bool ready=false;
                await using(var child=await BoundLifecycleChannel.Open(verified,m,tx.Id,nonce))
                {if(m=="normal"){await child.Ready();Require(!child.Child.HasExited,"CANDIDATE_ALREADY_EXITED");session.Confirm(new(tx.Id,tx.Nonce,candidate.Sha256));ready=true;await child.Close();}else await RefusesAsync(()=>child.Ready(TimeSpan.FromSeconds(5)));}
                StateRecord result=ready?session.Store.Load().Record:session.Recover();
                Require(result.Trust.ApprovedSequence==2 && result.Confirmed==(ready?candidate:state.Old),"LIFECYCLE_RESULT");
                if(!ready)Refuses(()=>session.Begin(candidate,state.Clock(),DateTimeOffset.UtcNow+TimeSpan.FromHours(1),true),"CANDIDATE_REPLAY");
                Observations.Add(new{kind="inert_transaction_lifecycle",mode=m,confirmed=ready,exact_previous_recovered=!ready,gf_operational=false,elevation_guard_bypassed_for_named_harness_only=true});
            });
        }
        await Test("U4-LIFE-DEFER","Deferral leaves a real owned application alive, no new intent and unchanged selector",async()=>{
            using var state=new StateCase(suite);using var session=new FixtureSession(state.Root);await using var old=await BoundLifecycleChannel.Open(verified,"normal",Guid.NewGuid().ToString("N"),Convert.ToHexStringLower(RandomNumberGenerator.GetBytes(32)));await old.Ready();

        });
