internal sealed class RealLabPins : IDisposable
{
    private readonly List<WindowsFileGuard> pins=[];
    internal RealLabPins(string src,RealLabContext context)
    {
        try
        {
            foreach(var item in context.Files)
            {
                Require(item.Path.Length is >0 and <180 && !item.Path.Contains('\\') &&
                    !item.Path.Contains(':') && !item.Path.Split('/').Any(x=>x is "" or "." or "..") &&
                    item.Length is >0 and <=67108864 && Hex(item.Sha,64),"REAL_PIN_PATH_OR_SIZE");
                var pin=WindowsFileGuard.Open(src,System.IO.Path.Combine(src,item.Path));
                pins.Add(pin);
                byte[] bytes=pin.Read(67108864);
                Require(pin.Identity==item.Identity && bytes.LongLength==item.Length &&
                    Hash(bytes)==item.Sha,"REAL_PACKAGE_BYTES_OR_OBJECT_CHANGED");
            }
        }
        catch{Dispose();throw;}
    }
    internal void Recheck(){foreach(var pin in pins)pin.Recheck();}
    public void Dispose(){for(int i=pins.Count-1;i>=0;i--)pins[i].Dispose();pins.Clear();}
}

    // Shared by the real owner and directed Windows process-cut trials.
    // Does not grant launch, recovery or READY authority.
    internal static async Task<string> SuperviseRealObserver(string src,RealLabContext c,
        string mode,Process child,TimeSpan observerTimeout)
    {
        // RealOwner's observer inherits its non-breakaway job. Directed
        // supervision trials exercise this routine with their own worker.
        var stdout=child.StandardOutput.ReadToEndAsync();var stderr=child.StandardError.ReadToEndAsync();
        bool observerTimedOut=false;
        try{await child.WaitForExitAsync().WaitAsync(observerTimeout);}
        catch(TimeoutException)
        {
            // A live owner must not abandon the observer's HKCU fixture on
            // this path. Stop only its owned observer, confirm exit, then
            // run the same conservative cleanup as after any failed exit.
            observerTimedOut=true;
            try { if(!child.HasExited)child.Kill(); }
            catch(InvalidOperationException) when(child.HasExited) { }
            try { await child.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(10)); }
            catch(TimeoutException)
            { throw new Rejected("REAL_OBSERVER_STOP_UNPROVEN_PRESERVE"); }
        }
        string error=await stderr;_=await stdout;
        if(mode is "--recovery-observe-trial" or "--recovery-observe-trial-negative")
        {
            // A4: original live owner supervises a disposable HKCU fixture.
            // The child stages its exact marker BEFORE registry mutation.
            // Only after observer exit, inspect/cleanup the named fixture.
            // An inconsistent identity returns nonzero and blocks New.
            var cleanInfo=new ProcessStartInfo(c.Python)
            {
                UseShellExecute=false,CreateNoWindow=true,
                RedirectStandardOutput=true,RedirectStandardError=true,
                WorkingDirectory=c.Repository
            };
            foreach(string arg in new[]{c.Script,"--trial-fixture-cleanup",src})
                cleanInfo.ArgumentList.Add(arg);
            using var clean=Process.Start(cleanInfo)
                ?? throw new Rejected("REAL_TRIAL_CLEANUP_NOT_CREATED");
            var cleanStdout=clean.StandardOutput.ReadToEndAsync();
            var cleanStderr=clean.StandardError.ReadToEndAsync();
            try { await clean.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(20)); }
            catch(TimeoutException)
            {
                // Do not leave a cleanup helper mutating HKCU after this
                // owner has already declared its result inconclusive.
                try { if(!clean.HasExited)clean.Kill(); }
                catch(InvalidOperationException) when(clean.HasExited) { }
                try { await clean.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(10)); }
                catch(TimeoutException)
                { throw new Rejected("REAL_TRIAL_CLEANUP_STOP_UNPROVEN_PRESERVE"); }
                throw new Rejected("REAL_TRIAL_CLEANUP_TIMEOUT_PRESERVE");
            }
            string cleanupResult=await cleanStdout;
            _=await cleanStderr;
            Require(clean.ExitCode==0 &&
                (cleanupResult.Trim() is "GF_PHASE1_TRIAL_CLEANUP NO_MARKER" or
                    "GF_PHASE1_TRIAL_CLEANUP NOT_CREATED" or
                    "GF_PHASE1_TRIAL_CLEANUP CLEANED_EXACT_FIXTURE"),
                "REAL_TRIAL_CLEANUP_INCONCLUSIVE_PRESERVE");
        }
        if(observerTimedOut)throw new Rejected("REAL_OBSERVER_TIMEOUT_EFFECTS_UNKNOWN");
        return error;
    }

    internal static async Task RealOwner(string src,string lab,string bootstrap,string cut,int originalOldPid=0)
    {

        // For the focused live handoff ONLY: open the original OLD process
        // BEFORE signalling the selector barrier, retain its kernel handle
        // across exit, and prove the real image path. A numeric PID by itself
        // never authorizes launch, recovery, or READY.
        using Process? originalOld = cut is "handoff" or "handoff-install" or "handoff-install-cut" or "handoff-install-after-second-cut"
            ? Process.GetProcessById(originalOldPid) : null;
        if(cut is "handoff" or "handoff-install" or "handoff-install-cut" or "handoff-install-after-second-cut")
        {
            string originalImage=Path.Combine(
                Environment.GetEnvironmentVariable("RUNNER_TEMP") ??
                    throw new Rejected("REAL_HANDOFF_RUNNER_TEMP_MISSING"),
                "gf-e2e-old","GerenciadorFinal.exe");
            Require(originalOld is not null && !originalOld.HasExited &&
                Path.GetFullPath(originalOld.MainModule?.FileName ?? "").Equals(
                    Path.GetFullPath(originalImage),StringComparison.OrdinalIgnoreCase),
                "REAL_HANDOFF_ORIGINAL_OLD_HANDLE_OR_IMAGE");
            _=originalOld.Handle;
        }
        using var contextPin=WindowsFileGuard.Open(src,RealContextPath(src));
        using var gate=TryCoordinatorGate(src);Require(gate is not null,"REAL_OWNER_BUSY");
        using var job=LabProcessJob.Open(c.OwnerJob);job.RequireCurrentMember();
        using var pins=new RealLabPins(src,c);VerifyRealManifest(src,c);
        using(var lease=L16Boundary.Exclusive(src))BeginCore(src,lab,bootstrap,c.ManifestSha);
        WriteRealSelector(src,c.Installation,c.Transaction,c.ManifestSha,"old",1);
        if(cut=="after-intent")await RealCut(src,cut);

            // an authenticated no-launch witness by the reconciler.
            string snapshot=Path.Combine(src,"real-pre-new-snapshot.fixture");
            Directory.CreateDirectory(snapshot);
            DurableRealFile(Path.Combine(snapshot,"selector-before-new.bin"),
                File.ReadAllBytes(RealSelector(src)));
            foreach(string journal in Directory.GetFiles(src,"l18-phase-*.fixture.json"))
                DurableRealFile(Path.Combine(snapshot,Path.GetFileName(journal)),
                    File.ReadAllBytes(journal));
        }
        WriteRealSelector(src,c.Installation,c.Transaction,c.ManifestSha,"new",2);
        if(cut=="after-select")await RealCut(src,cut);
        // Fence BEFORE the first possible NEW execution. A crash in the gap
        // remains unknown even if no child was actually created.
        DurableRealFile(NewLaunchFence(src),System.Text.Encoding.UTF8.GetBytes("NEW-MAY-HAVE-UNOBSERVABLE-EFFECTS\n"));
        if(cut=="after-launch-fence")await RealCut(src,cut);
        Append(src,lab,Launched,"real-new-effects-fenced-before-create",c.ManifestSha);
        string selected=SelectedRealPath(src,c,out var slot,out _);
        Require(slot=="new","REAL_OWNER_NOT_NEW");
        if(cut is "handoff" or "handoff-install" or "handoff-install-cut" or "handoff-install-after-second-cut")
        {
            Require(originalOld is not null && !originalOld.HasExited &&
                Environment.GetEnvironmentVariable("GITHUB_ACTIONS")=="true" &&
                Environment.GetEnvironmentVariable("GITHUB_REPOSITORY")==
                    "walissonminel-1991/gerenciador-final",
                "REAL_HANDOFF_ORIGINAL_OLD_EXITED_EARLY");
            string runner=Environment.GetEnvironmentVariable("RUNNER_TEMP")!;
            // Coordinator owns the original NEW/2 selector, journal and
            // effects fence. This marker only coordinates OLD shutdown;
            // the authority for the first NEW launch remains the LIVE owner.
            DurableRealFile(Path.Combine(runner,"gf-e2e-selected-new-ready.json"),
                JsonSerializer.SerializeToUtf8Bytes(new{
                    schema="gf-e2e-selected-new-ready/1",
                    selector="new",generation=2,launch_fenced=true,
                    l18_live_owner=true,test_only=true,operational_ready=false
                }));
            try
            {
                await originalOld.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(75));
            }
            catch(TimeoutException)
            {
                throw new Rejected("REAL_HANDOFF_ORIGINAL_OLD_EXIT_TIMEOUT_HOLD");
            }
            Require(originalOld.HasExited && originalOld.ExitCode==0 &&
                File.Exists(Path.Combine(runner,"gf-e2e-old-ui-closed.json")) &&
                File.Exists(Path.Combine(runner,"gf-e2e-old-message-loop-exited.json")),
                "REAL_HANDOFF_ORIGINAL_OLD_NOT_GRACEFUL");
            using(var receipt=JsonDocument.Parse(File.ReadAllBytes(
                Path.Combine(runner,"gf-e2e-old-message-loop-exited.json"))))
                Require(receipt.RootElement.GetProperty("application_run_returned").GetBoolean() &&
                    receipt.RootElement.GetProperty("selected_new_fenced_before_close").GetBoolean() &&
                    !receipt.RootElement.GetProperty("operational_ready").GetBoolean(),
                    "REAL_HANDOFF_OLD_LOOP_RECEIPT_INVALID");
            RequireRealSelector(src,c.Installation,c.Transaction,c.ManifestSha,"new",2);
            pins.Recheck();VerifyRealManifest(src,c);
            DurableRealFile(Path.Combine(runner,"gf-e2e-old-exit-before-new-observation.json"),
                JsonSerializer.SerializeToUtf8Bytes(new{
                    schema="gf-e2e-old-exited-before-l18-new-observation/1",
                    process_handle_wait_completed=true,
                    old_exit_code=0,selector="new",generation=2,
                    test_only=true,operational_ready=false
                }));
            if(cut is "handoff-install" or "handoff-install-cut" or "handoff-install-after-second-cut")
            {
                await SwapRealInstalledPackageAfterOldExit(src,c,
                    cutBetweenMoves:cut=="handoff-install-cut",
                    cutAfterSecondMove:cut=="handoff-install-after-second-cut");
                RequireRealSelector(src,c.Installation,c.Transaction,c.ManifestSha,"new",2);
                pins.Recheck();VerifyRealManifest(src,c);
            }
            // SAME selected NEW/2 observer. In handoff-install mode ONLY,
            // it runs the authenticated bytes now physically installed at
            // the original portable GF location, NOT the fixture slot.
            // It still checks selection and the package in phase1_lab.py.
        }
        if(cut is "positive-trial" or "positive-trial-negative")
        {
            // Experimental B: ONLY the still-live original owner may record
            // a limited positive candidate observation. Never operational
            // READY, never a witness for recovery after crash or reboot.
            await RunRealObserver(src,c,cut=="positive-trial"
                ?"--recovery-observe-trial":"--recovery-observe-trial-negative");
            await job.RequireOnlyCurrentProcess();
            pins.Recheck();VerifyRealManifest(src,c);
            RequireRealSelector(src,c.Installation,c.Transaction,c.ManifestSha,"new",2);
            using(var obs=JsonDocument.Parse(File.ReadAllBytes(
                Path.Combine(src,"real-observed-new.fixture.json"))))
            {
                var h=obs.RootElement.GetProperty("readonly_inventory_search_health");
                Require(h.GetProperty("seeded_inventory_positive_observed").GetBoolean() &&
                    h.GetProperty("seeded_inventory_negative_control_observed").GetBoolean(),
                    "REAL_TRIAL_EVIDENCE_NOT_FUNCTIONAL");
            }
            Append(src,lab,Exited,"real-new-tested-in-original-live-owner",c.ManifestSha);
            Append(src,lab,Reconciling,"real-new-trial-observation-verified-not-recovery",c.ManifestSha);
            Append(src,lab,New,"real-new-experimental-trial-accepted-not-ready",c.ManifestSha);
            Console.WriteLine("GF_PHASE1_TRIAL_NEW_POSITIVE; READY=false; recovery=false; operational=false");
            return;
        }
        if(cut is "normal" or "handoff" or "handoff-install")
        {
            // This is NOT a safe post-NEW rollback: a successful health probe
            // cannot establish that external/brokered effects have quiesced.
            // Leave the launch fence and NEW selector intact for the SAME
            // reconciler to return UnknownBlocked in a separate process.
            await RunRealObserver(src,c,"--recovery-observe",
                observeInstalledNew:cut=="handoff-install");
            await job.RequireOnlyCurrentProcess();
            pins.Recheck();VerifyRealManifest(src,c);
            RequireRealSelector(src,c.Installation,c.Transaction,c.ManifestSha,"new",2);
            Append(src,lab,Exited,"real-new-functional-normal-lab-only",c.ManifestSha);
            RememberUnknown(src,lab,"real-new-effects-not-independently-bounded");
            return;
        }
        if(cut is "after-new-exit" or "after-new-exit-state-rollback")
        {
            await RunRealObserver(src,c,"--recovery-observe");
            Append(src,lab,Exited,"real-new-limited-observation-complete",c.ManifestSha);
            await RealCut(src,cut);
        }
        Require(cut=="during-new","REAL_UNKNOWN_CUT");
        using var gf=Process.Start(new ProcessStartInfo(Path.Combine(selected,"GerenciadorFinal.exe"))
            {UseShellExecute=false,WorkingDirectory=selected,ArgumentList={"--tray"}})
            ?? throw new Rejected("REAL_NEW_NOT_CREATED");
        var helper=new ProcessStartInfo(Environment.ProcessPath!){UseShellExecute=false};
        foreach(string a in new[]{"--real-late-writer",src,lab})helper.ArgumentList.Add(a);
        using var descendant=Process.Start(helper) ?? throw new Rejected("REAL_WRITER_NOT_CREATED");
        await Phase1RecoveryTests.Until(()=>File.Exists(Path.Combine(src,"late-writer-ready.fixture.txt")));
        Require(!gf.HasExited && !descendant.HasExited && job.Active>=3,"REAL_SURVIVOR_NOT_ACTIVE");
        await RealCut(src,cut);
    }
    internal static async Task<string> ReconcileReal(string src,string lab,string bootstrap,string cut,long ownerWitness)
    {
        Root(src,lab);
        using var gate=TryCoordinatorGate(src);
        if(gate is null)return "Busy";
        try
        {
            RequireRealMarkerAbsent(RecoveryBlocked(src),
                "REAL_UNREADABLE_HISTORY_BLOCKED","REAL_RECOVERY_BLOCK_MARKER_NOT_REGULAR");
            var c=LoadRealContext(src,lab);
            using var contextPin=WindowsFileGuard.Open(src,RealContextPath(src));
            using var owner=LabProcessJob.FromInherited(c.OwnerJob,ownerWitness);
            using var recovery=LabProcessJob.Open(c.RecoveryJob);recovery.RequireCurrentMember();
            // OS job accounting, NOT PID, HWND, exit code or a local receipt.
            if(owner.Active!=0 || recovery.Active!=1)
                File.WriteAllText(Path.Combine(src,"real-job-members.fixture.txt"),
                    "owner: "+owner.Diagnostic()+" | recovery: "+recovery.Diagnostic());
            Require(owner.Active==0 && recovery.Active==1,"REAL_CONTAINED_PROCESSES_STILL_ACTIVE");
            using var lease=L16Boundary.Exclusive(src);
            var history=Load(src,lab);
            Require(history.Count>=5 && history.All(x=>x.TargetSha==c.ManifestSha),"REAL_JOURNAL_TARGET_FORK");
            using(var fixture=Probe(src,lab,bootstrap))
                Require(fixture.stagedIdentity==history[0].StagedIdentity,"REAL_STAGE_OBJECT_FORK");
            using var pins=new RealLabPins(src,c);VerifyRealManifest(src,c);
            // Authenticate the existing signed metadata again; never re-sign or
            // infer TUF trust from context/receipt hashes. Python cannot select.
            await RunRealObserver(src,c,"--recovery-verify");
            await recovery.RequireOnlyCurrentProcess();
            Require(owner.Active==0 && recovery.Active==1,"REAL_PROCESS_STARTED_DURING_PROBE");
            _=SelectedRealPath(src,c,out string selected,out int generation);
            RequireRealMarkerAbsent(NewLaunchFence(src),
                "REAL_NEW_EFFECTS_NOT_INDEPENDENTLY_BOUNDED",
                "REAL_NEW_FENCE_NOT_REGULAR_FILE");
            // The fence may have been removed after a real NEW launch.
            // An intact L18 journal provides independent *negative* evidence:
            // it cannot license OLD merely because the fence is now absent.
            // This is NOT an anti-tamper proof when an adversary can rewrite
            // the journal and fence together, or after physical power loss.
            Require(!history.Any(x=>x.Detail=="real-new-effects-fenced-before-create"),
                "REAL_LAUNCHED_NEW_JOURNAL_FENCE_MISSING");
            // Two indistinguishable histories are possible here: a genuine
            // crash after selecting NEW but before writing the launch fence,
            // or NEW execution followed by same-user removal of the fence and
            // trailing journal records. Without a separate, authenticated,
            // retained no-launch witness it is NOT safe to restore OLD.
            Require(selected!="new","REAL_NEW_SELECTED_NO_INDEPENDENT_NO_LAUNCH_PROOF");
            // No independent durable launch authority exists in Phase 1.
            // A same-user adversary can restore OLD/1 and a valid pre-NEW
            // journal snapshot after NEW has actually executed. The retained
            // laboratory Jobs and authenticated package bytes cannot prove
            // that this history is genuine. Do not observe or relaunch OLD.
            Require(false,"REAL_NO_INDEPENDENT_NO_LAUNCH_AUTHORITY");
            if(history.Any(x=>x.Detail=="real-old-functional-lab-only"))
            {
                Require(selected=="old" && history[^1].Phase==Unknown,"REAL_TERMINAL_SELECTOR_CONTRADICTION");
                pins.Recheck();return "OldReobservedNoReady";
            }
            RequireRealMarkerAbsent(OldLaunchFence(src),
                "REAL_INTERRUPTED_OLD_EFFECTS_UNKNOWN",
                "REAL_OLD_FENCE_NOT_REGULAR_FILE");
            if(history[^1].Phase==Intent)
                Append(src,lab,Switching,"real-intent-reopened-no-new-launch",c.ManifestSha);
            if(Load(src,lab)[^1].Phase!=Reconciling)
            {
                RememberUnknown(src,lab,"real-process-crash-under-reconciliation");
                Append(src,lab,Reconciling,"real-old-restore-intent",c.ManifestSha);
            }
            if(cut=="before-old-replace")await RealCut(src,cut);
            if(selected=="new")
                WriteRealSelector(src,c.Installation,c.Transaction,c.ManifestSha,"old",3);
            _=SelectedRealPath(src,c,out selected,out generation);
            Require(selected=="old","REAL_RESTORE_DID_NOT_SELECT_OLD");
            Append(src,lab,Reconciling,"real-old-selector-revalidated",c.ManifestSha);
            if(cut=="after-old-replace")await RealCut(src,cut);
            pins.Recheck();
            DurableRealFile(OldLaunchFence(src),System.Text.Encoding.UTF8.GetBytes("OLD-MAY-HAVE-EFFECTS\n"));
            await RunRealObserver(src,c,"--recovery-observe");
            await recovery.RequireOnlyCurrentProcess();
            Require(recovery.Active==1 && owner.Active==0,"REAL_OLD_DESCENDANTS_STILL_ACTIVE");
            // A fresh full-byte/namespace read, in addition to retaining NTFS
            // file capabilities throughout the execution and external probe.
            using var finalPins=new RealLabPins(src,c);VerifyRealManifest(src,c);
            _=SelectedRealPath(src,c,out selected,out _);
            Require(selected=="old","REAL_SELECTOR_CHANGED_DURING_OLD");
            Append(src,lab,Old,"real-old-functional-lab-only",c.ManifestSha);
            RememberUnknown(src,lab,"real-old-restored-ready-still-unproven");
            return "OldRecoveredNoReady";
        }
        catch(Exception e) when(e is Rejected or IOException or Win32Exception or
            JsonException or UnauthorizedAccessException or InvalidDataException or
            InvalidOperationException or KeyNotFoundException)
        {
            RememberUnknown(src,lab,"real-recovery-evidence-insufficient");
            string reason=e switch
            {
                Rejected rejected=>rejected.Code,
                Win32Exception windows=>"WIN32_"+windows.NativeErrorCode,
                _=>e.GetType().Name
            };
            // Diagnostic only; never read by the recovery decision itself.
            File.WriteAllText(Path.Combine(src,"real-reconcile-reason.fixture.txt"),reason);
            Console.WriteLine("GF_PHASE1_RECOVERY_BLOCK "+reason);
            return "UnknownBlocked";
        }
    }
