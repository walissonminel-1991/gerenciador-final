    private static string Own=>Environment.ProcessPath ?? throw new Rejected("REAL_TEST_IMAGE_MISSING");
    internal static async Task Until(Func<bool> condition)
    {
        var watch=Stopwatch.StartNew();
        while(!condition() && watch.Elapsed<TimeSpan.FromSeconds(90))await Task.Delay(25);
        Require(condition(),"REAL_TEST_BARRIER_UNREACHED");
    }
    private static void CopyTree(string input,string output)
    {
        Directory.CreateDirectory(output);
        foreach(string file in Directory.EnumerateFiles(input))
        {
            Require((File.GetAttributes(file)&FileAttributes.ReparsePoint)==0,"REAL_SOURCE_REPARSE");
            // Copy bytes into a fresh object inheriting the private directory's
            // ACL; never propagate metadata/permissions from publish output.
            using var read=new FileStream(file,FileMode.Open,FileAccess.Read,FileShare.Read);
            using var write=new FileStream(Path.Combine(output,Path.GetFileName(file)),
                FileMode.CreateNew,FileAccess.Write,FileShare.None);
            read.CopyTo(write);write.Flush(true);
        }
        foreach(string dir in Directory.EnumerateDirectories(input))
        {
            Require((File.GetAttributes(dir)&FileAttributes.ReparsePoint)==0,"REAL_SOURCE_REPARSE");
            CopyTree(dir,Path.Combine(output,Path.GetFileName(dir)));
        }
    }
    private static async Task<RealLabContext> Prepare(L16Tests.Scenario s,string[] a)
    {
        string src=s.Source.Root;
        CopyTree(a[7],Path.Combine(src,"real","old"));
        CopyTree(a[8],Path.Combine(src,"real","new"));
        File.WriteAllText(Path.Combine(src,"PERSONAL-KEEP-UNCHANGED.txt"),"DO-NOT-TOUCH-MY-DATA\n",new UTF8Encoding(false));
        using var gate=L18Coordinator.TryCoordinatorGate(src);
        Require(gate is not null,"REAL_SETUP_BUSY");
        var auth=await L18Coordinator.PrepareRealAuthorization(src,s.Lab,L16Tests.bootstrap,
            a[5],a[6],a[4],Path.Combine(src,"real","old"),Path.Combine(src,"real","new"),
            Path.Combine(Environment.GetEnvironmentVariable("RUNNER_TEMP")!,"unused-"+Guid.NewGuid().ToString("N")),
            Path.Combine(src,"unused-report.json"),begin:false);
        CopyTree(auth.AuthDir,Path.Combine(src,"real-auth"));
        Directory.Delete(auth.AuthDir,true);File.Delete(auth.AuthReport);
        var files=new List<RealPinnedFile>();
        foreach(string path in Directory.EnumerateFiles(Path.Combine(src,"real"),"*",SearchOption.AllDirectories)
            .Concat(Directory.EnumerateFiles(Path.Combine(src,"real-auth")))
            .Append(Path.Combine(src,"PERSONAL-KEEP-UNCHANGED.txt")).Order(StringComparer.Ordinal))
        {
            using var pin=WindowsFileGuard.Open(src,path);
            byte[] bytes=pin.Read(67108864);
            files.Add(new(Path.GetRelativePath(src,path).Replace('\\','/'),Hash(bytes),bytes.LongLength,pin.Identity));
        }
        string prefix="Local\\GF-PHASE1-"+auth.Installation+"-"+auth.Transaction;
        var context=new RealLabContext("gf-real-recovery-lab/1",auth.Installation,auth.Transaction,
            auth.Target,prefix+"-owner",prefix+"-recovery",Path.GetFullPath(a[4]),
            Path.GetFullPath(a[5]),Path.GetFullPath(a[6]),files.ToArray());
        L18Coordinator.DurableRealFile(L18Coordinator.RealContextPath(src),StateCodec.Encode(context));
        return context;
    }
    private static (string Slot,int Generation) Selection(string src)
    {
        using var doc=Json(File.ReadAllBytes(Path.Combine(src,"l18-real-active.fixture.json")),2048);
        return (Text(doc.RootElement,"selected"),doc.RootElement.GetProperty("generation").GetInt32());
    }
    private static void History(string src,RealLabContext c,bool mayBeTorn=false)
    {
        var files=Directory.GetFiles(src,"l18-phase-*.fixture.json").Order(StringComparer.Ordinal).ToArray();
        Require(files.Length>=5,"REAL_HISTORY_NOT_PERSISTED");
        for(int i=0;i<files.Length;i++)
        {
            if(mayBeTorn && i==files.Length-1)continue;
            var record=StateCodec.Decode<L18PhaseRecord>(File.ReadAllBytes(files[i]));
            Require(record.Sequence==i+1 && record.Transaction==c.Transaction &&
                record.TargetSha==c.ManifestSha && record.Phase!="new-observed",
                "REAL_HISTORY_IDENTITY_FORK_OR_NEW_PROMOTION");
        }
    }
    private static async Task CutOwner(LabProcessJob owner,string src,string lab,string cut)
    {
        using var child=owner.Start(Own,"--real-owner",src,lab,L16Tests.bootstrap,cut);
        await Until(()=>File.Exists(L18Coordinator.RealCutPath(src)) || child.Exited);
        Require(!child.Exited && File.ReadAllText(L18Coordinator.RealCutPath(src))==cut,
            "REAL_OWNER_DID_NOT_REACH_EXPECTED_CUT:"+ReadError(src));
        await child.KillOnlyThisProcess();
        Require(child.ExitCode==91,"REAL_INJECTED_KILL_NOT_DISTINGUISHED");
        File.Delete(L18Coordinator.RealCutPath(src));
    }
    private static string ReadError(string src)
    {
        return string.Join(" | ",new[]{"real-child-error.fixture.txt",
            "real-reconcile-reason.fixture.txt","real-observer-error.fixture.txt","real-job-members.fixture.txt"}
            .Select(x=>Path.Combine(src,x)).Where(File.Exists).Select(File.ReadAllText));
    }
    private static string Reason(string src)=>File.ReadAllText(Path.Combine(src,"real-reconcile-reason.fixture.txt"));
    private static async Task<string> Reopen(LabProcessJob recovery,LabProcessJob? owner,string src,string lab,string cut="none")
    {
        string result=Path.Combine(src,"real-reconcile-result.fixture.txt");File.Delete(result);
        File.Delete(Path.Combine(src,"real-reconcile-reason.fixture.txt"));
        // A missing owner handle MUST be a conservative block even if a new
        // kernel Job with the identical name has been created afterward.
        using var child=owner is null
            ?recovery.Start(Own,"--real-reconcile",src,lab,L16Tests.bootstrap,cut,"0")
            :recovery.StartWithWitness(owner,Own,"--real-reconcile",src,lab,L16Tests.bootstrap,cut);
        if(cut!="none")
        {
            await Until(()=>File.Exists(L18Coordinator.RealCutPath(src)) || child.Exited);
            Require(!child.Exited && File.ReadAllText(L18Coordinator.RealCutPath(src))==cut,
                "REAL_RECONCILE_CUT_UNREACHED:"+ReadError(src));
            await child.KillOnlyThisProcess();
            Require(child.ExitCode==91,"REAL_RECONCILE_CUT_WRONG_EXIT");
            File.Delete(L18Coordinator.RealCutPath(src));
            return "InjectedCrash";
        }
        await child.Wait();
        Require(child.ExitCode==0 && File.Exists(result),"REAL_RECONCILER_FAILED:"+ReadError(src));
        return File.ReadAllText(result);
    }
    private static async Task<object> Scenario(string name,string[] a)
    {
        using var s=await L16Tests.Prepared();string src=s.Source.Root;
        var c=await Prepare(s,a);
        using var contextPin=WindowsFileGuard.Open(src,L18Coordinator.RealContextPath(src));
        using var pins=new RealLabPins(src,c); // retained through owner death and reconciliation
        using var owner=LabProcessJob.Create(c.OwnerJob);
        using var recovery=LabProcessJob.Create(c.RecoveryJob);
        bool blocked=name is "normal" or "new-live-descendant" or "new-exited" or "launch-fence" or
            "selector-fork" or "journal-torn" or "witness-lost" or
            "witness-recreated" or "witness-wrong-object" or "external-broker" or
            "external-broker-handoff" or "launch-fence-lost-after-new" or
            "selected" or "selector-locked" or "fence-directory-before-new" or "joint-fence-journal-erasure" or
            "intent" or "recover-before-replace" or "recover-after-replace" or
            "same-user-full-state-rollback-after-new";
        bool counterexample=name=="same-user-full-state-rollback-after-new";
        // Historical recover-* names now exercise conservative refusal.
        // No recovery cut is injected when no trusted no-launch authority exists.
        bool crashRecovery=false;
        string ownerCut=name switch
        {
            "intent" or "recover-before-replace" or "recover-after-replace" or
                "selector-locked"=>"after-intent",
            "witness-recreated"=>"after-intent",
            "witness-wrong-object"=>"after-intent",
            "normal"=>"normal",
            "new-live-descendant"=>"during-new",
            "new-exited" or "external-broker" or "external-broker-handoff" or
                "launch-fence-lost-after-new" or "joint-fence-journal-erasure"=>"after-new-exit",
            "same-user-full-state-rollback-after-new"=>"after-new-exit-state-rollback",
            "launch-fence"=>"after-launch-fence",
            _=>"after-select"
        };
        LabProcessJob? renamedWitness=null;
        try
        {
            if(name=="normal")
            {
                // Normal owner exits without an injected crash. The same
                // durable recovery context must be usable by a NEW process.
                using var live=owner.Start(Own,"--real-owner",src,s.Lab,L16Tests.bootstrap,"normal");
                await live.Wait();
                Require(live.ExitCode==0,"REAL_NORMAL_OWNER_FAILED:"+ReadError(src));
            }
            else await CutOwner(owner,src,s.Lab,ownerCut);
            Require(Selection(src)==(name is "intent" or "recover-before-replace" or
                "recover-after-replace" or "selector-locked" or "witness-recreated" or
                "witness-wrong-object" ?("old",1):("new",2)),
                "REAL_CUT_SELECTION_WRONG");
            History(src,c);
            if(name=="normal")
            {
                foreach(string slot in new[]{"old","new"})
                {
                    string observationPath=Path.Combine(src,"real-observed-"+slot+".fixture.json");
                    using var observation=JsonDocument.Parse(File.ReadAllBytes(observationPath));
                    var o=observation.RootElement;
                    Require(o.GetProperty("created").GetBoolean() &&
                        o.GetProperty("terminated").GetBoolean() &&
                        o.GetProperty("graceful_shutdown_verified_for_owned_process").GetBoolean() &&
                        !o.GetProperty("forced_cleanup_required").GetBoolean() &&
                        o.GetProperty("readonly_inventory_search_health")
                            .GetProperty("minimum_readonly_health_contract_observed").GetBoolean(),
                        "REAL_NORMAL_"+slot+"_FUNCTIONAL_OBSERVATION_MISSING");
                }
                Require(File.Exists(L18Coordinator.NewLaunchFence(src)) &&
                    File.Exists(Path.Combine(src,"l18-real-old-may-have-effects.fixture.txt")),
                    "REAL_NORMAL_BYPASSED_COMMON_FENCES");
            }
            bool lateWriterDenied=false,leaseRefused=false;
            bool externalBrokerEffectObserved=false,externalHandoffEffectObserved=false;
            bool missingFenceVsJournalDetected=false;
            bool markerWrongTypeObserved=false,jointEvidenceErasureObserved=false;
            if(name=="new-live-descendant")
            {
                // The original C# owner is gone; GF and a delayed writer remain.
                Require(owner.Active>=2,"REAL_DESCENDANTS_NOT_SURVIVING_OWNER");
                File.WriteAllText(Path.Combine(src,"l18-real-ready.fixture.json"),"{\"ready\":true,\"exited\":true}");
                Require(await Reopen(recovery,name is "witness-lost" or "witness-recreated" ? null :
                    name=="witness-wrong-object" ? recovery : owner,src,s.Lab)=="UnknownBlocked" && Selection(src)==("new",2),
                    "REAL_LIVE_DESCENDANTS_OR_FORGED_RECEIPT_ACCEPTED");
                Require(Reason(src)=="REAL_CONTAINED_PROCESSES_STILL_ACTIVE","REAL_WRONG_LIVE_BLOCK_REASON");
                File.WriteAllText(Path.Combine(src,"late-writer-release.fixture.txt"),"release");
                await Until(()=>File.Exists(Path.Combine(src,"late-writer-result.fixture.txt")));
                Require(File.ReadAllText(Path.Combine(src,"late-writer-result.fixture.txt"))=="both-writes-denied",
                    "REAL_LATE_WRITE_NOT_CONTAINED");
                lateWriterDenied=true;
                await owner.Cleanup(); // intentional scope-limited cleanup, never graceful shutdown
            }
            else Require(owner.Active==0,"REAL_OWNER_JOB_NOT_EMPTY");
            if(name is "new-exited" or "external-broker" or "external-broker-handoff" or "launch-fence-lost-after-new")
            {
                // Diagnostic timings are fractional. Do not feed an observation
                // report into the TUF/state parser that requires unsigned integers.
                var raw=File.ReadAllBytes(Path.Combine(src,"real-observed-new.fixture.json"));
                Require(raw.Length is >0 and <=32768,"REAL_NEW_DIAGNOSTIC_SIZE");
                using var o=JsonDocument.Parse(raw);
                Require(o.RootElement.GetProperty("graceful_shutdown_verified_for_owned_process").GetBoolean(),
                    "REAL_NEW_GRACEFUL_EXIT_NOT_DEMONSTRATED");
            }
            if(name=="external-broker")
            {
                // Negative proof: NEW was already observed exiting, owner Job
                // has no members. A separate broker process (created by this
                // test supervisor, NOT by GF) is outside BOTH Jobs and can
                // still perform a late write to an unprotected inert fixture.
                // This does not claim that GF actually delegates to WMI.
                Require(owner.Active==0 && recovery.Active==0,
                    "REAL_BROKER_JOBS_MUST_START_EMPTY");
                string target=Path.Combine(src,"external-broker-effect.fixture.txt");
                L18Coordinator.DurableRealFile(target,Encoding.UTF8.GetBytes("before\\n"));
                var psi=new ProcessStartInfo(Own)
                {
                    UseShellExecute=false,CreateNoWindow=true,
                    WorkingDirectory=Path.GetDirectoryName(Own)!
                };
                foreach(string arg in new[]{"--real-external-broker",src,s.Lab})
                    psi.ArgumentList.Add(arg);
                using var broker=Process.Start(psi) ?? throw new Rejected("REAL_BROKER_NOT_CREATED");
                await Until(()=>File.Exists(Path.Combine(src,"external-broker-ready.fixture.txt")) ||
                    broker.HasExited);
                Require(!broker.HasExited && !owner.ContainsProcess(broker) &&
                    !recovery.ContainsProcess(broker) &&
                    owner.Active==0 && recovery.Active==0,
                    "REAL_BROKER_NOT_INDEPENDENT_OF_JOBS");
                // Forged same-user quiescence assertion: the reconciler must
                // ignore it even while an uncontained writer is still alive.
                File.WriteAllText(Path.Combine(src,"real-false-quiescence.fixture.json"),
                    """{ "ready":true, "all_effects_complete":true }""");
                Require(await Reopen(recovery,owner,src,s.Lab)=="UnknownBlocked" &&
                    Reason(src)=="REAL_NEW_EFFECTS_NOT_INDEPENDENTLY_BOUNDED" &&
                    Selection(src)==("new",2),
                    "REAL_BROKER_FORGED_QUIESCENCE_ACCEPTED");
                File.WriteAllText(Path.Combine(src,"external-broker-release.fixture.txt"),"go");
                await broker.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(15));
                Require(broker.ExitCode==0 &&
                    File.ReadAllText(target)=="after\\n" &&
                    owner.Active==0 && recovery.Active==0,
                    "REAL_UNCONTAINED_LATE_EFFECT_NOT_DEMONSTRATED");
                externalBrokerEffectObserved=true;
            }
            if(name=="external-broker-handoff")
            {
                // Negative proof: the uncontained intermediate broker HAS EXITED
                // but an unrelated uncontained process still holds a future
                // fixture write. The GF application never starts this broker.
                Require(owner.Active==0 && recovery.Active==0,
                    "REAL_HANDOFF_JOBS_NOT_EMPTY");
                string target=Path.Combine(src,"external-broker-effect.fixture.txt");
                L18Coordinator.DurableRealFile(target,Encoding.UTF8.GetBytes("before\\n"));
                var psi=new ProcessStartInfo(Own)
                {
                    UseShellExecute=false,CreateNoWindow=true,
                    WorkingDirectory=Path.GetDirectoryName(Own)!
                };
                foreach(string arg in new[]{"--real-external-handoff-broker",src,s.Lab})
                    psi.ArgumentList.Add(arg);
                using(var intermediary=Process.Start(psi) ??
                    throw new Rejected("REAL_HANDOFF_BROKER_NOT_CREATED"))
                {
                    Require(!owner.ContainsProcess(intermediary) &&
                        !recovery.ContainsProcess(intermediary),
                        "REAL_HANDOFF_INTERMEDIARY_INSIDE_JOBS");
                    await intermediary.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(30));
                    Require(intermediary.ExitCode==0,"REAL_HANDOFF_INTERMEDIARY_FAILED:"+ReadError(src));
                }
                string ready=Path.Combine(src,"handoff-writer-ready.fixture.txt");
                string writerRecord="";
                // The writer publishes this marker with FileShare.None and
                // flushes before closing. Existence can precede readable
                // completion: retry only ERROR_SHARING_VIOLATION (32).
                await Until(() =>
                {
                    if(!File.Exists(ready))return false;
                    try
                    {
                        writerRecord=File.ReadAllText(ready);
                        return writerRecord.Length is >0 and <=12;
                    }
                    catch(IOException e) when((e.HResult & 0xFFFF)==32){return false;}
                });
                int writerPid=0;
                Require(int.TryParse(writerRecord,
                        System.Globalization.NumberStyles.None,
                        System.Globalization.CultureInfo.InvariantCulture,out writerPid) &&
                    writerPid>0,"REAL_HANDOFF_WRITER_NOT_DISCOVERED");
                // PID only acquires a stable handle for TEST cleanup, never
                // authorizes state or proves continuity of a real updater.
                using var writer=Process.GetProcessById(writerPid);
                string image=writer.MainModule?.FileName ??
                    throw new Rejected("REAL_HANDOFF_WRITER_IMAGE_MISSING");
                Require(Path.GetFullPath(image).Equals(Path.GetFullPath(Own),
                    StringComparison.OrdinalIgnoreCase),"REAL_HANDOFF_WRITER_IMAGE_WRONG");
                string release=Path.Combine(src,"handoff-writer-release.fixture.txt");
                try
                {
                    Require(!writer.HasExited &&
                        !owner.ContainsProcess(writer) &&
                        !recovery.ContainsProcess(writer) &&
                        owner.Active==0 && recovery.Active==0 &&
                        File.ReadAllText(target)=="before\\n",
                        "REAL_HANDOFF_WRITER_NOT_INDEPENDENT");
                    File.WriteAllText(Path.Combine(src,"real-false-quiescence.fixture.json"),
                        """{ "ready":true, "all_effects_complete":true, "broker_exited":true }""");
                    Require(await Reopen(recovery,owner,src,s.Lab)=="UnknownBlocked" &&
                        Reason(src)=="REAL_NEW_EFFECTS_NOT_INDEPENDENTLY_BOUNDED" &&
                        Selection(src)==("new",2),"REAL_HANDOFF_FALSE_QUIESCENCE_ACCEPTED");
                    File.WriteAllText(release,"go");
                    await writer.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(20));
                    Require(writer.ExitCode==0 &&
                        File.ReadAllText(target)=="after\\n" &&
                        owner.Active==0 && recovery.Active==0,
                        "REAL_HANDOFF_LATE_EFFECT_NOT_DEMONSTRATED");
                    externalHandoffEffectObserved=true;
                }
                finally
                {
                    // Only the verified laboratory test-image handle can be
                    // cleaned up, never an arbitrary process or PID from OS.
                    if(!writer.HasExited)
                    {
                        File.WriteAllText(release,"go");
                        await Task.Delay(100);
                        if(!writer.HasExited)writer.Kill();
                        await writer.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(20));
                    }
                }
            }
            if(counterexample)
            {
                // SAME-USER ATTACK: NEW has actually executed and exited.
                // Re-present a byte-for-byte valid prefix of the launch
                // history, and the pre-NEW OLD/1 selector together.
                // This is *not* a safe recovery case, regardless of outcome.
                string snapshot=Path.Combine(src,"real-pre-new-snapshot.fixture");
                var saved=Directory.GetFiles(snapshot,"l18-phase-*.fixture.json")
                    .Order(StringComparer.Ordinal).ToArray();
                Require(saved.Length>=5,"REAL_ROLLBACK_PRE_NEW_SNAPSHOT_INCOMPLETE");
                using(var observed=JsonDocument.Parse(File.ReadAllBytes(
                    Path.Combine(src,"real-observed-new.fixture.json"))))
                {
                    Require(observed.RootElement.GetProperty("created").GetBoolean() &&
                        observed.RootElement.GetProperty("terminated").GetBoolean() &&
                        observed.RootElement.GetProperty("graceful_shutdown_verified_for_owned_process").GetBoolean(),
                        "REAL_ROLLBACK_REQUIRES_ACTUAL_GF_NEW_EXECUTION");
                }
                Require(Selection(src)==("new",2) &&
                    File.Exists(L18Coordinator.NewLaunchFence(src)) &&
                    Directory.GetFiles(src,"l18-phase-*.fixture.json").Length>saved.Length,
                    "REAL_ROLLBACK_NEW_LAUNCH_NOT_PERSISTED");
                // Only the disposable test supervisor performs this tamper.
                // Packages, signed manifests, installation IDs and Job handles
                // remain unchanged. Remove the mutable local tail and restore
                // both pieces of the earlier valid no-launch snapshot.
                foreach(string live in Directory.GetFiles(src,"l18-phase-*.fixture.json"))
                    File.Delete(live);
                foreach(string previous in saved)
                    File.Copy(previous,Path.Combine(src,Path.GetFileName(previous)));
                File.Copy(Path.Combine(snapshot,"selector-before-new.bin"),
                    Path.Combine(src,"l18-real-active.fixture.json"),overwrite:true);
                File.Delete(L18Coordinator.NewLaunchFence(src));
                Require(Selection(src)==("old",1) &&
                    !File.Exists(L18Coordinator.NewLaunchFence(src)) &&
                    Directory.GetFiles(src,"l18-phase-*.fixture.json").Length==saved.Length,
                    "REAL_ROLLBACK_HISTORY_NOT_RESTORED");
                History(src,c);
            }
            if(name=="launch-fence-lost-after-new")
            {
                // Negative test only: the real GF NEW has already run and
                // exited. Deleting a single unpinned disposable fixture must
                // NOT allow the reconciler to treat this as PRE-LAUNCH.
                string fence=L18Coordinator.NewLaunchFence(src);
                Require(File.Exists(fence),"REAL_LAUNCH_FENCE_NOT_CREATED");
                Require(Directory.GetFiles(src,"l18-phase-*.fixture.json").Any(path=>
                    StateCodec.Decode<L18PhaseRecord>(File.ReadAllBytes(path)).Detail
                        =="real-new-effects-fenced-before-create"),
                    "REAL_LAUNCH_EVENT_NOT_RECORDED");
                File.Delete(fence);
                Require(!File.Exists(fence),"REAL_TEST_FENCE_DELETE_FAILED");
                missingFenceVsJournalDetected=true;
            }
            if(name=="fence-directory-before-new")
            {
                // File.Exists(directory) is false, despite an occupied path.
                // Do not convert an invalid marker namespace into a recovery.
                string marker=L18Coordinator.NewLaunchFence(src);
                Require(!File.Exists(marker),"REAL_DIRECTORY_CASE_ALREADY_FENCED");
                Directory.CreateDirectory(marker);
                Require(Directory.Exists(marker) && !File.Exists(marker),
                    "REAL_DIRECTORY_COUNTEREXAMPLE_NOT_CONSTRUCTED");
                markerWrongTypeObserved=true;
            }
            if(name=="joint-fence-journal-erasure")
            {
                // The real NEW has executed. Remove both locally mutable
                // negative witnesses: launch fence + the last two journal
                // records (Launched, Exited). The remaining valid history
                // looks like the legitimate AFTER-SELECT pre-launch cut.
                // This is an indistinguishability test, not cold recovery.
                var journal=Directory.GetFiles(src,"l18-phase-*.fixture.json")
                    .Order(StringComparer.Ordinal).ToArray();
                Require(journal.Length>=8,"REAL_JOINT_ERASURE_JOURNAL_SHORT");
                var launched=StateCodec.Decode<L18PhaseRecord>(
                    File.ReadAllBytes(journal[^2]));
                var exited=StateCodec.Decode<L18PhaseRecord>(
                    File.ReadAllBytes(journal[^1]));
                Require(launched.Detail=="real-new-effects-fenced-before-create" &&
                    exited.Phase=="owner-exited",
                    "REAL_JOINT_ERASURE_NO_NEW_LAUNCH");
                string fence=L18Coordinator.NewLaunchFence(src);
                Require(File.Exists(fence),"REAL_JOINT_ERASURE_FENCE_MISSING");
                File.Delete(journal[^1]);File.Delete(journal[^2]);File.Delete(fence);
                Require(!File.Exists(fence),"REAL_JOINT_ERASURE_FENCE_REMAINED");
                History(src,c); // Still structurally valid, but not authentic.
                jointEvidenceErasureObserved=true;
            }
            if(name=="selector-fork")
            {
                string p=Path.Combine(src,"l18-real-active.fixture.json");
                File.WriteAllText(p,File.ReadAllText(p).Replace(c.ManifestSha,new string('0',64)));
            }
            if(name=="journal-torn")
                File.WriteAllText(Directory.GetFiles(src,"l18-phase-*.fixture.json").Order(StringComparer.Ordinal).Last(),"{\"schema\":");
            if(name is "witness-lost" or "witness-recreated")
            {
                owner.Dispose();
                if(name=="witness-recreated")
                {
                    // Demonstrate the name is reusable, while the original
                    // kernel identity is gone. Merely reopening by name must
                    // never count as evidence for a cold recovery.
                    renamedWitness=LabProcessJob.Create(c.OwnerJob);
                    Require(renamedWitness.Active==0,"REAL_REPLACEMENT_JOB_NONEMPTY");
                }
            }
            if(name=="selected")
            {
                using(var gate=L18Coordinator.TryCoordinatorGate(src))
                {

                {

                    File.Exists(L18Coordinator.RecoveryBlocked(src)),
                    "REAL_LOCKED_SELECTOR_MUST_PERSIST_BLOCKER");
            }
            if(name is "recover-before-replace" or "recover-after-replace")
            {
                // A pre-NEW history is not an independent launch authority.
                // The former crash-recovery cuts must now be unreachable:
                // do not execute OLD and do not claim crash recovery coverage.
                Require(Selection(src)==("old",1) &&
                    !File.Exists(Path.Combine(src,"l18-real-old-may-have-effects.fixture.txt")),
                    "REAL_PRE_NEW_RECOVERY_MUST_NOT_HAVE_LAUNCHED_OLD");
            }

                    name=="witness-wrong-object" ? recovery : owner,src,s.Lab);

                    "journal-torn"=>"JSON_INVALID",
                    "witness-lost" or "witness-recreated"=>"REAL_OWNER_WITNESS_NOT_INHERITED",

                    "launch-fence-lost-after-new"=>"REAL_LAUNCHED_NEW_JOURNAL_FENCE_MISSING",
                    "selected" or "joint-fence-journal-erasure"=>

            Require(Selection(src)==(name is "intent" or "recover-before-replace" or

                "witness-wrong-object" or "same-user-full-state-rollback-after-new"
                ?("old",1):blocked?("new",2):("old",3)),
                "REAL_FINAL_SELECTOR_INCONSISTENT");
            if(blocked)
                Require(Hash(File.ReadAllBytes(Path.Combine(src,"l18-real-active.fixture.json")))==before,
                    "REAL_BLOCKED_RECOVERY_MOVED_SELECTOR");
            else
            {
                var raw=File.ReadAllBytes(Path.Combine(src,"real-observed-old.fixture.json"));
                Require(raw.Length is >0 and <=32768,"REAL_OLD_DIAGNOSTIC_SIZE");

            }
            string journalBefore=string.Join("|",Directory.GetFiles(src,"l18-phase-*.fixture.json").Order(StringComparer.Ordinal).Select(x=>Hash(File.ReadAllBytes(x))));
            string replay=await Reopen(recovery,name is "witness-lost" or "witness-recreated" ? null :

            Require(replay==(blocked?"UnknownBlocked":"OldReobservedNoReady"),"REAL_REPLAY_PROMOTED_OR_RELAUNCHED");
            if(!blocked)
                Require(journalBefore==string.Join("|",Directory.GetFiles(src,"l18-phase-*.fixture.json").Order(StringComparer.Ordinal).Select(x=>Hash(File.ReadAllBytes(x)))),
                    "REAL_IDEMPOTENT_REPLAY_CHANGED_HISTORY");
            History(src,c,name=="journal-torn");pins.Recheck();
            using var finalPins=new RealLabPins(src,c);
            s.Preserved();
            Require(recovery.Active==0,"REAL_RECOVERY_SURVIVORS_REMAIN");
            return new{test=name,status="passed",
                classification=blocked?"PASS_CONTAINMENT":"PASS_RECOVERY_LAB_ONLY",
                unsafe_recovery_after_real_new=counterexample && outcome=="OldRecoveredNoReady",
                full_state_rollback_blocked=counterexample && outcome=="UnknownBlocked" &&
                    reason=="REAL_NO_INDEPENDENT_NO_LAUNCH_AUTHORITY",
                interrupted_owner=name!="normal",normal_owner_completed_without_crash=name=="normal",
                reconciler_new_process=true,old_and_canary_preserved=true,
                manifest_and_object_identities_rechecked=true,late_writer_denied=lateWriterDenied,
                concurrent_coordinator_refused=leaseRefused,recovery_process_also_interrupted=crashRecovery,
                external_broker_effect_observed=externalBrokerEffectObserved,
                external_handoff_effect_observed=externalHandoffEffectObserved,
                missing_launch_fence_with_new_journal_observed=missingFenceVsJournalDetected,
                wrong_type_marker_fail_closed=markerWrongTypeObserved,
                joint_fence_journal_erasure_blocked=jointEvidenceErasureObserved,
                outcome,replay,block_reason=reason,operational_ready=false};
        }
        finally
        {
            renamedWitness?.Dispose();
            if(name is not ("witness-lost" or "witness-recreated"))await owner.Cleanup();
            await recovery.Cleanup();
        }
    }
    private static async Task LateWriter(string src,string lab)
    {
        _=L18Coordinator.LoadRealContext(src,lab);
        void Publish(string name,string text)
        {
            string pending=Path.Combine(src,name+".pending");
            L18Coordinator.DurableRealFile(pending,Encoding.UTF8.GetBytes(text));
            File.Move(pending,Path.Combine(src,name));
        }
        Publish("late-writer-ready.fixture.txt","ready");
        await Until(()=>File.Exists(Path.Combine(src,"late-writer-release.fixture.txt")));
        bool Denied(string file)
        {
            try{using var w=new FileStream(file,FileMode.Open,FileAccess.Write,FileShare.ReadWrite|FileShare.Delete);w.WriteByte(0);return false;}
            catch(IOException){return true;}
        }
        bool old=Denied(Path.Combine(src,"real","old","GerenciadorFinal.dll"));
        bool canary=Denied(Path.Combine(src,"PERSONAL-KEEP-UNCHANGED.txt"));
        Publish("late-writer-result.fixture.txt",old&&canary?"both-writes-denied":"WRITE-ESCAPED");
        await Task.Delay(Timeout.Infinite);
    }
    private static async Task ExternalBroker(string src,string lab)
    {
        // Deliberately uncontained LABORATORY proxy only. A real external
        // Windows broker may be outside a Job; this process is not GF/WMI.
        _=L18Coordinator.LoadRealContext(src,lab);
        string target=Path.Combine(src,"external-broker-effect.fixture.txt");

        File.WriteAllText(target,"after\\n",new UTF8Encoding(false));
    }
    private static async Task ExternalHandoffBroker(string src,string lab)
    {
        // Intermediary exits while the delegated LAB writer remains alive.
        _=L18Coordinator.LoadRealContext(src,lab);
        Require(File.ReadAllText(Path.Combine(src,"external-broker-effect.fixture.txt"))=="before\\n",
            "REAL_HANDOFF_INTERMEDIARY_SCOPE");
        var psi=new ProcessStartInfo(Own)
        {
            UseShellExecute=false,CreateNoWindow=true,
            WorkingDirectory=Path.GetDirectoryName(Own)!
        };
        foreach(string arg in new[]{"--real-external-handoff-writer",src,lab})
            psi.ArgumentList.Add(arg);
        using var writer=Process.Start(psi) ??
            throw new Rejected("REAL_HANDOFF_WRITER_NOT_STARTED");
        await Until(()=>File.Exists(Path.Combine(src,"handoff-writer-ready.fixture.txt")) || writer.HasExited);
        Require(!writer.HasExited,"REAL_HANDOFF_WRITER_EXITED_PREMATURELY");
    }
    private static async Task ExternalHandoffWriter(string src,string lab)
    {
        _=L18Coordinator.LoadRealContext(src,lab);
        string target=Path.Combine(src,"external-broker-effect.fixture.txt");
        Require(File.ReadAllText(target)=="before\\n","REAL_HANDOFF_WRITER_SCOPE");
        L18Coordinator.DurableRealFile(Path.Combine(src,"handoff-writer-ready.fixture.txt"),
            Encoding.UTF8.GetBytes(Environment.ProcessId.ToString(
                System.Globalization.CultureInfo.InvariantCulture)));
        await Until(()=>File.Exists(Path.Combine(src,"handoff-writer-release.fixture.txt")));
        File.WriteAllText(target,"after\\n",new UTF8Encoding(false));
    }
