    private static string Own=>Environment.ProcessPath ?? throw new Rejected("REAL_TEST_IMAGE_MISSING");
    internal static async Task Until(Func<bool> condition)
    {
        var watch=Stopwatch.StartNew();
        while(!condition() && watch.Elapsed<TimeSpan.FromSeconds(90))await Task.Delay(25);
        Require(condition(),"REAL_TEST_BARRIER_UNREACHED");
    }
    private static void CopyTree(string input,string output)
    {
        Directory.CreateDirectory(output);
        foreach(string file in Directory.EnumerateFiles(input))
        {
            Require((File.GetAttributes(file)&FileAttributes.ReparsePoint)==0,"REAL_SOURCE_REPARSE");
            // Copy bytes into a fresh object inheriting the private directory's
            // ACL; never propagate metadata/permissions from publish output.
            using var read=new FileStream(file,FileMode.Open,FileAccess.Read,FileShare.Read);
            using var write=new FileStream(Path.Combine(output,Path.GetFileName(file)),
                FileMode.CreateNew,FileAccess.Write,FileShare.None);
            read.CopyTo(write);write.Flush(true);
        }
        foreach(string dir in Directory.EnumerateDirectories(input))
        {
            Require((File.GetAttributes(dir)&FileAttributes.ReparsePoint)==0,"REAL_SOURCE_REPARSE");
            CopyTree(dir,Path.Combine(output,Path.GetFileName(dir)));
        }
    }
    private static async Task<RealLabContext> Prepare(L16Tests.Scenario s,string[] a)
    {
        string src=s.Source.Root;
        CopyTree(a[7],Path.Combine(src,"real","old"));
        CopyTree(a[8],Path.Combine(src,"real","new"));
        File.WriteAllText(Path.Combine(src,"PERSONAL-KEEP-UNCHANGED.txt"),"DO-NOT-TOUCH-MY-DATA\n",new UTF8Encoding(false));
        using var gate=L18Coordinator.TryCoordinatorGate(src);
        Require(gate is not null,"REAL_SETUP_BUSY");
        var auth=await L18Coordinator.PrepareRealAuthorization(src,s.Lab,L16Tests.bootstrap,
            a[5],a[6],a[4],Path.Combine(src,"real","old"),Path.Combine(src,"real","new"),
            Path.Combine(Environment.GetEnvironmentVariable("RUNNER_TEMP")!,"unused-"+Guid.NewGuid().ToString("N")),
            Path.Combine(src,"unused-report.json"),begin:false);
        CopyTree(auth.AuthDir,Path.Combine(src,"real-auth"));
        Directory.Delete(auth.AuthDir,true);File.Delete(auth.AuthReport);
        var files=new List<RealPinnedFile>();
        foreach(string path in Directory.EnumerateFiles(Path.Combine(src,"real"),"*",SearchOption.AllDirectories)
            .Concat(Directory.EnumerateFiles(Path.Combine(src,"real-auth")))
            .Append(Path.Combine(src,"PERSONAL-KEEP-UNCHANGED.txt")).Order(StringComparer.Ordinal))
        {
            using var pin=WindowsFileGuard.Open(src,path);
            byte[] bytes=pin.Read(67108864);
            files.Add(new(Path.GetRelativePath(src,path).Replace('\\','/'),Hash(bytes),bytes.LongLength,pin.Identity));
        }
        string prefix="Local\\GF-PHASE1-"+auth.Installation+"-"+auth.Transaction;
        var context=new RealLabContext("gf-real-recovery-lab/1",auth.Installation,auth.Transaction,
            auth.Target,prefix+"-owner",prefix+"-recovery",Path.GetFullPath(a[4]),
            Path.GetFullPath(a[5]),Path.GetFullPath(a[6]),files.ToArray());
        L18Coordinator.DurableRealFile(L18Coordinator.RealContextPath(src),StateCodec.Encode(context));
        return context;
    }

    {

    private static void History(string src,RealLabContext c,bool mayBeTorn=false)
    {
        var files=Directory.GetFiles(src,"l18-phase-*.fixture.json").Order(StringComparer.Ordinal).ToArray();
        Require(files.Length>=5,"REAL_HISTORY_NOT_PERSISTED");
        for(int i=0;i<files.Length;i++)
        {
            if(mayBeTorn && i==files.Length-1)continue;
            var record=StateCodec.Decode<L18PhaseRecord>(File.ReadAllBytes(files[i]));
            Require(record.Sequence==i+1 && record.Transaction==c.Transaction &&
                record.TargetSha==c.ManifestSha && record.Phase!="new-observed",
                "REAL_HISTORY_IDENTITY_FORK_OR_NEW_PROMOTION");
        }
    }
    private static async Task CutOwner(LabProcessJob owner,string src,string lab,string cut)
    {
        using var child=owner.Start(Own,"--real-owner",src,lab,L16Tests.bootstrap,cut);
        await Until(()=>File.Exists(L18Coordinator.RealCutPath(src)) || child.Exited);
        Require(!child.Exited && File.ReadAllText(L18Coordinator.RealCutPath(src))==cut,
            "REAL_OWNER_DID_NOT_REACH_EXPECTED_CUT:"+ReadError(src));
        await child.KillOnlyThisProcess();
        Require(child.ExitCode==91,"REAL_INJECTED_KILL_NOT_DISTINGUISHED");
        File.Delete(L18Coordinator.RealCutPath(src));
    }
    private static string ReadError(string src)
    {
        return string.Join(" | ",new[]{"real-child-error.fixture.txt",
            "real-reconcile-reason.fixture.txt","real-observer-error.fixture.txt","real-job-members.fixture.txt"}
            .Select(x=>Path.Combine(src,x)).Where(File.Exists).Select(File.ReadAllText));
    }
    private static string Reason(string src)=>File.ReadAllText(Path.Combine(src,"real-reconcile-reason.fixture.txt"));
    private static async Task<string> Reopen(LabProcessJob recovery,LabProcessJob? owner,string src,string lab,string cut="none")
    {
        string result=Path.Combine(src,"real-reconcile-result.fixture.txt");File.Delete(result);
        File.Delete(Path.Combine(src,"real-reconcile-reason.fixture.txt"));
        // A missing owner handle MUST be a conservative block even if a new
        // kernel Job with the identical name has been created afterward.
        using var child=owner is null
            ?recovery.Start(Own,"--real-reconcile",src,lab,L16Tests.bootstrap,cut,"0")
            :recovery.StartWithWitness(owner,Own,"--real-reconcile",src,lab,L16Tests.bootstrap,cut);
        if(cut!="none")
        {
            cutStage="awaiting_cut_marker";
            await Until(()=>File.Exists(L18Coordinator.RealCutPath(src)) || child.Exited);
            Require(!child.Exited && File.ReadAllText(L18Coordinator.RealCutPath(src))==cut,
                "REAL_RECONCILE_CUT_UNREACHED:"+ReadError(src));
            cutStage="cut_seen_killing_owner";
            await child.KillOnlyThisProcess();
            Require(child.ExitCode==91,"REAL_RECONCILE_CUT_WRONG_EXIT");
            File.Delete(L18Coordinator.RealCutPath(src));
            return "InjectedCrash";
        }
        await child.Wait();
        Require(child.ExitCode==0 && File.Exists(result),"REAL_RECONCILER_FAILED:"+ReadError(src));
        return File.ReadAllText(result);
    }
    private static async Task<object> PreparationScenario(string name,string[] a)
    {
        using var s=await L16Tests.Prepared();string src=s.Source.Root;
        var c=await Prepare(s,a);
        using var contextPin=WindowsFileGuard.Open(src,L18Coordinator.RealContextPath(src));
        using var pins=new RealLabPins(src,c);
        using var owner=LabProcessJob.Create(c.OwnerJob);
        using var recovery=LabProcessJob.Create(c.RecoveryJob);
        bool completed=name=="prepare-only",oldFenced=name=="before-old-observation";
        string oldFence=Path.Combine(src,"l18-real-old-may-have-effects.fixture.txt");
        void AssertNoObservation()
        {
            // This assertion covers the audited owner/verifier route, not
            // launches elsewhere. The old out-of-band test remains mandatory.
            Require(!File.Exists(Path.Combine(src,"real-observed-old.fixture.json")) &&
                !File.Exists(Path.Combine(src,"real-observed-new.fixture.json")) &&
                !File.Exists(L18Coordinator.NewLaunchFence(src)) &&
                File.Exists(oldFence)==oldFenced && Selection(src)==("old",1),
                "REAL_PREPARATION_CROSSED_OBSERVATION_BOUNDARY");
        }
        try
        {
            if(completed)
            {
                using var child=owner.Start(Own,"--real-owner",src,s.Lab,L16Tests.bootstrap,name);
                await child.Wait();
                Require(child.ExitCode==0,"REAL_DATA_PREPARE_FAILED:"+ReadError(src));
            }
            else await CutOwner(owner,src,s.Lab,name);
            Require(owner.Active==0,"REAL_PREPARE_OWNER_SURVIVOR");
            History(src,c);AssertNoObservation();
            var journal=Directory.GetFiles(src,"l18-phase-*.fixture.json").Order(StringComparer.Ordinal)
                .Select(x=>StateCodec.Decode<L18PhaseRecord>(File.ReadAllBytes(x))).ToArray();
            Require(journal.Length==(oldFenced?6:5) &&
                !journal.Any(x=>x.Phase=="owner-launched") &&
                journal[^1].Phase==(oldFenced?"switching":"intent-durable"),
                "REAL_PREPARE_FALSE_LAUNCH_EVENT");
            if(oldFenced)
                Require(File.ReadAllText(oldFence)=="OLD-MAY-HAVE-EFFECTS\n",
                    "REAL_OLD_PRE_OBSERVATION_FENCE_MISSING");
            string before=Hash(File.ReadAllBytes(Path.Combine(src,"l18-real-active.fixture.json")));
            string outcome=await Reopen(recovery,owner,src,s.Lab);
            string reason=Reason(src);
            cutStage="replay_two";
            string replay=await Reopen(recovery,owner,src,s.Lab);
            // Intent cannot transition directly to Unknown in the inherited
            // journal. Its fallback blocker intentionally dominates replay.
            string replayReason=Reason(src);
            Require(outcome=="UnknownBlocked" && replay=="UnknownBlocked" &&
                reason=="REAL_NO_INDEPENDENT_NO_LAUNCH_AUTHORITY" &&
                replayReason==(oldFenced?reason:"REAL_UNREADABLE_HISTORY_BLOCKED") &&
                Hash(File.ReadAllBytes(Path.Combine(src,"l18-real-active.fixture.json")))==before,
                "REAL_PREPARATION_BECAME_RECOVERY_AUTHORITY");
            AssertNoObservation();History(src,c);pins.Recheck();
            using var finalPins=new RealLabPins(src,c);
            s.Preserved();
            Require(owner.Active==0 && recovery.Active==0,"REAL_PREPARATION_REPLAY_SURVIVOR");
            return new{test=name,status="passed",classification="PASS_CONTAINMENT",
                preparation_completed_without_observation=completed,
                data_only_verification_completed=completed || name=="prepare-after-verify",
                preparation_interrupted=!completed && !oldFenced,
                local_old_may_launch_boundary_observed=oldFenced,
                no_functional_observation_on_audited_route=true,
                no_launch_claim_scope="AUDITED_COORDINATOR_AND_PYTHON_API_GUARD_ONLY",
                global_no_launch_proven=false,independent_recovery_authority=false,
                unsafe_recovery_after_real_new=false,
                interrupted_owner=!completed,normal_owner_completed_without_crash=false,
                reconciler_new_process=true,old_and_canary_preserved=true,
                manifest_and_object_identities_rechecked=true,
                outcome,replay,block_reason=reason,replay_block_reason=replayReason,operational_ready=false};

        finally{await owner.Cleanup();await recovery.Cleanup();}

    private static async Task<object> RetryAfterUnknownRetainsBoth(string[] a)
    {
        // A second cooperative RealOwner in a new process on the same root,
        // after GF OLD/NEW ran and reconciliation returned UnknownBlocked.
        // This tests intact local state only, not antirollback or H1/H2/H3.
        using var s=await L16Tests.Prepared();string src=s.Source.Root;
        var c=await Prepare(s,a);
        using var contextPin=WindowsFileGuard.Open(src,L18Coordinator.RealContextPath(src));
        using var pins=new RealLabPins(src,c);
        using var owner=LabProcessJob.Create(c.OwnerJob);
        using var recovery=LabProcessJob.Create(c.RecoveryJob);
        try
        {
            using(var first=owner.Start(Own,"--real-owner",src,s.Lab,L16Tests.bootstrap,"normal"))
            {
                await first.Wait();
                Require(first.ExitCode==0,"REAL_HOLD_FIRST_OWNER_FAILED:"+ReadError(src));
            }
            string oldObservation=Path.Combine(src,"real-observed-old.fixture.json");
            string newObservation=Path.Combine(src,"real-observed-new.fixture.json");
            foreach(string file in new[]{oldObservation,newObservation})
            {
                using var report=JsonDocument.Parse(File.ReadAllBytes(file));
                Require(report.RootElement.GetProperty("created").GetBoolean() &&
                    report.RootElement.GetProperty("terminated").GetBoolean() &&
                    report.RootElement.GetProperty("graceful_shutdown_verified_for_owned_process").GetBoolean(),
                    "REAL_HOLD_REQUIRES_REAL_GF_OBSERVATION");
            }
            Require(Selection(src)==("new",2),"REAL_HOLD_FIRST_OWNER_NOT_NEW");
            string outcome=await Reopen(recovery,owner,src,s.Lab);
            string reason=Reason(src);
            Require(outcome=="UnknownBlocked" &&
                reason=="REAL_NEW_EFFECTS_NOT_INDEPENDENTLY_BOUNDED" &&
                owner.Active==0 && recovery.Active==0,
                "REAL_HOLD_FIRST_UNKNOWN_NOT_ESTABLISHED");
            string selector=Path.Combine(src,"l18-real-active.fixture.json");
            string oldFence=Path.Combine(src,"l18-real-old-may-have-effects.fixture.txt");
            string newFence=L18Coordinator.NewLaunchFence(src);
            string selectorBefore=Hash(File.ReadAllBytes(selector));
            string oldFenceBefore=Hash(File.ReadAllBytes(oldFence));
            string newFenceBefore=Hash(File.ReadAllBytes(newFence));
            string oldObservedBefore=Hash(File.ReadAllBytes(oldObservation));
            string newObservedBefore=Hash(File.ReadAllBytes(newObservation));
            string Journals()=>string.Join("|",Directory.GetFiles(src,"l18-phase-*.fixture.json")
                .Order(StringComparer.Ordinal).Select(x=>Hash(File.ReadAllBytes(x))));
            string journalsBefore=Journals();
            void CheckPackages()
            {
                using var check=new RealLabPins(src,c); // all pinned file bytes/NTFS identities, plus canary
                foreach(string slot in new[]{"old","new"})
                {
                    string prefix="real/"+slot+"/";
                    string[] actual=Directory.GetFiles(Path.Combine(src,"real",slot),"*",
                        SearchOption.AllDirectories).Select(x=>
                        Path.GetRelativePath(src,x).Replace('\\','/')).Order(StringComparer.Ordinal).ToArray();
                    string[] expected=c.Files.Where(x=>x.Path.StartsWith(prefix,StringComparison.Ordinal))
                        .Select(x=>x.Path).Order(StringComparer.Ordinal).ToArray();
                    Require(actual.SequenceEqual(expected),"REAL_HOLD_PACKAGE_NAMESPACE_CHANGED");
                }
            }
            CheckPackages();
            using(var second=owner.Start(Own,"--real-owner",src,s.Lab,L16Tests.bootstrap,"normal"))
            {
                await second.Wait();
                string error=Path.Combine(src,"real-child-error.fixture.txt");
                Require(second.ExitCode!=0 && File.Exists(error) &&
                    File.ReadAllText(error).Contains("L20_BEGIN_BLOCKED",StringComparison.Ordinal),
                    "REAL_HOLD_SECOND_OWNER_NOT_REJECTED_AT_BEGIN");
            }
            Require(owner.Active==0 && recovery.Active==0 &&
                Selection(src)==("new",2) && Hash(File.ReadAllBytes(selector))==selectorBefore &&
                Hash(File.ReadAllBytes(oldFence))==oldFenceBefore &&
                Hash(File.ReadAllBytes(newFence))==newFenceBefore &&
                Hash(File.ReadAllBytes(oldObservation))==oldObservedBefore &&
                Hash(File.ReadAllBytes(newObservation))==newObservedBefore &&
                Journals()==journalsBefore,"REAL_HOLD_RETRY_CHANGED_HISTORY_OR_OBSERVATION");
            CheckPackages();pins.Recheck();
            string replay=await Reopen(recovery,owner,src,s.Lab);
            Require(replay=="UnknownBlocked" &&
                Reason(src)=="REAL_NEW_EFFECTS_NOT_INDEPENDENTLY_BOUNDED" &&
                Selection(src)==("new",2) && Journals()==journalsBefore,
                "REAL_HOLD_REPLAY_PROMOTED_OR_MUTATED_HISTORY");
            CheckPackages();pins.Recheck();s.Preserved();
            Require(owner.Active==0 && recovery.Active==0,"REAL_HOLD_JOB_SURVIVOR");
            return new{test="retry-after-unknown-retains-both",status="passed",
                classification="PASS_CONTAINMENT",outcome,replay,block_reason=reason,
                cooperating_retry_rejected_before_launch=true,
                second_owner_rejection="L20_BEGIN_BLOCKED",
                old_and_new_bytes_preserved=true,selector_unchanged=true,
                fences_and_observations_unchanged=true,journal_unchanged=true,
                unknown_replay_blocked=true,same_user_rollback_resistant=false,
                external_launch_blocked=false,independent_recovery_authority=false,
                real_gf_old_and_new_observed_before_retry=true,
                unsafe_recovery_after_real_new=false,
                old_and_canary_preserved=true,reconciler_new_process=true,
                manifest_and_object_identities_rechecked=true,operational_ready=false};
        }
        finally{await owner.Cleanup();await recovery.Cleanup();}
    }

    private static async Task ObserverSupervisionTrials(string[] a)
    {
        // Real Python/RegCreateKeyExW/TerminateProcess, supervised by the
        // SAME C# wait/cleanup routine as RealOwner. No GF activation here.
        foreach(bool timeout in new[]{false,true})
        {
            using var s=await L16Tests.Prepared();string src=s.Source.Root;
            var c=new RealLabContext("directed-supervision-not-transaction",
                "","","","","",a[4],a[5],a[6],[]);
            var info=new ProcessStartInfo(c.Python){UseShellExecute=false,CreateNoWindow=true,
                RedirectStandardOutput=true,RedirectStandardError=true,WorkingDirectory=c.Repository};
            foreach(string arg in new[]{Path.Combine(Path.GetDirectoryName(c.Script)!,
                "registry_fixture_adversarial_tests.py"),"--worker",src})info.ArgumentList.Add(arg);
            using var child=Process.Start(info) ?? throw new Rejected("A4_DIRECTED_WORKER_NOT_CREATED");
            try
            {
                string? active=await child.StandardOutput.ReadLineAsync().WaitAsync(TimeSpan.FromSeconds(20));
                Require(active=="GF_PHASE1_FIXTURE_ACTIVE","A4_DIRECTED_WORKER_NOT_ACTIVE");
                string marker=Path.Combine(src,"trial-hkcu-owner.fixture.json");
                using var doc=JsonDocument.Parse(File.ReadAllBytes(marker));
                string name=Text(doc.RootElement,"name");
                Require(name.StartsWith("GF-PHASE1-INVENTORY-",StringComparison.Ordinal) &&
                    name.Length=="GF-PHASE1-INVENTORY-".Length+24,"A4_DIRECTED_KEY_SCOPE");
                if(!timeout){child.Kill();await child.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(10));}
                bool timeoutRejected=false;
                try { _=await L18Coordinator.SuperviseRealObserver(src,c,
                    "--recovery-observe-trial",child,TimeSpan.FromMilliseconds(100)); }
                catch(Rejected e) when(timeout && e.Message.Contains(
                    "REAL_OBSERVER_TIMEOUT_EFFECTS_UNKNOWN",StringComparison.Ordinal))
                { timeoutRejected=true; }
                Require(child.HasExited && child.ExitCode!=0 &&
                    timeoutRejected==timeout && !File.Exists(marker),"A4_DIRECTED_CLEANUP_UNPROVEN");
                using var hkcu=Microsoft.Win32.RegistryKey.OpenBaseKey(
                    Microsoft.Win32.RegistryHive.CurrentUser,Microsoft.Win32.RegistryView.Registry64);
                using var remaining=hkcu.OpenSubKey(
                    @"Software\Microsoft\Windows\CurrentVersion\Uninstall\"+name);
                Require(remaining is null,"A4_DIRECTED_REGISTRY_RESIDUE");
                s.Preserved();
                Console.WriteLine("GF_PHASE1_CSHARP_SUPERVISION "+JsonSerializer.Serialize(new{
                    test=timeout?"observer-timeout-live-owner":"observer-killed-live-owner",
                    status="passed",same_owner_cleanup_routine=true,exact_key_absent=true,
                    marker_absent=true,timeout_rejected=timeoutRejected,
                    operational_ready=false,automatic_recovery_authorized=false}));
            }
            finally
            {
                if(!child.HasExited)child.Kill();
                await child.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(10));
            }
        }
    }

                "handoff-install-cut",a[9]);

            // Directory.Move(live,backup) returned. The owner must still be

            await Until(()=>File.Exists(L18Coordinator.RealCutPath(src)) || child.Exited);

                File.ReadAllText(canary,new UTF8Encoding(false))==
                    "DO-NOT-TOUCH-MY-DATA\n",
                "REAL_BETWEEN_MOVES_PREKILL_GAP_UNPROVEN");

            cutStage="owner_killed_replay_one";

                Directory.Exists(incoming) && !File.Exists(installedReceipt),

                !Directory.Exists(live) && Directory.Exists(backup) &&
                Directory.Exists(incoming) && !File.Exists(installedReceipt) &&
                File.ReadAllText(canary,new UTF8Encoding(false))==
                    "DO-NOT-TOUCH-MY-DATA\n",
                "REAL_BETWEEN_MOVES_REPLAY_MUTATED_INSTALL");

            pins.Recheck();

            // manifest and exact package identities with its RealLabPins.
            using(var independentPins=new RealLabPins(src,c))
                independentPins.Recheck();

                "REAL_BETWEEN_MOVES_CHILDREN_STILL_LIVE");

                schema="gf-e2e-interrupted-physical-swap-l18/1",
                scenario="between-install-moves",
                root=src,owner_exited_by_retained_process_handle=killed,
                owner_exit_code=91,selector="new",generation=2,
                first_directory_move_completed=true,second_directory_move_executed=false,
                old_present_in_backup=true,new_present_in_staging=true,
                original_executable_path_absent=true,personal_canary_untouched=true,
                old_and_new_fixture_pins_rechecked=true,

                rollback_authorized=false,rollback_performed=false,
                automatic_retry_authorized=false,cleanup_performed=false,
                test_only=true,operational_ready=false
            };
        }
        catch(Exception ex)
        {
            // Strictly runner-scoped diagnostic; never dump URLs, tokens or
            // stack traces. This is NOT recovery evidence or authorization.
            try
            {
                string reason=ex is Rejected rejected ? rejected.Code :
                    ex.GetType().Name;
                var diag=JsonSerializer.SerializeToUtf8Bytes(new{
                    schema="gf-e2e-cut-failure-diagnostic/1",
                    stage=cutStage,reason,owner_killed=killed,
                    selector_test_only=true,operational_ready=false
                });
                L18Coordinator.DurableRealFile(Path.Combine(runner,
                    "gf-e2e-cut-failure-diagnostic.json"),diag);
            }
            catch(IOException){ /* Cannot turn failed test into success. */ }
            catch(UnauthorizedAccessException){ }
            throw;
        }
        finally
        {
            // No recovery, rollback or filesystem cleanup in this test.
            // Only a surviving child would be contained by existing Jobs.
            if(!killed && owner.Active!=0)await owner.Cleanup();
            if(recovery.Active!=0)await recovery.Cleanup();
        }
    }
