    private static void Require(bool ok, string reason)
    {
        if (!ok) throw new InvalidDataException("GF_E2E_OLD_" + reason);
    }

    private static HttpClient Client()
    {
        var handler = new SocketsHttpHandler
        {
            AllowAutoRedirect = false,
            UseCookies = false,
            UseProxy = false,
            AutomaticDecompression = DecompressionMethods.None,
            Credentials = null,
            ConnectTimeout = TimeSpan.FromSeconds(20)
        };
        return new HttpClient(handler, disposeHandler: true)
        {
            Timeout = TimeSpan.FromMinutes(4)
        };
    }

    private static HttpRequestMessage Get(Uri uri, string? token, bool metadata)
    {
        var request = new HttpRequestMessage(HttpMethod.Get, uri);
        request.Headers.UserAgent.ParseAdd("GerenciadorFinal-E2E-Lab/1");
        request.Headers.Accept.Add(new MediaTypeWithQualityHeaderValue(
            metadata ? "application/vnd.github+json" : "application/octet-stream"));
        request.Headers.AcceptEncoding.Clear();
        if (token is not null)
        {
            request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
            request.Headers.TryAddWithoutValidation("X-GitHub-Api-Version", "2022-11-28");
        }
        return request;
    }

    private static async Task<byte[]> ReadJson(HttpClient http, Uri endpoint, string token)
    {
        using var req = Get(endpoint, token, metadata: true);
        using var resp = await http.SendAsync(req, HttpCompletionOption.ResponseHeadersRead);
        Require(resp.StatusCode == HttpStatusCode.OK,
                "METADATA_HTTP_STATUS_" + ((int)resp.StatusCode).ToString(
                    System.Globalization.CultureInfo.InvariantCulture));
        Require(resp.RequestMessage?.RequestUri == endpoint &&
                resp.Headers.Location is null, "METADATA_ORIGIN_OR_REDIRECT");
        var type = resp.Content.Headers.ContentType?.MediaType;
        Require(type is "application/json" or "application/vnd.github+json",
                "METADATA_CONTENT_TYPE");
        Require(resp.Content.Headers.ContentLength is null or <= 524288,
                "METADATA_TOO_LARGE");
        await using var body = await resp.Content.ReadAsStreamAsync();
        using var output = new MemoryStream();
        byte[] buffer = new byte[8192];
        int n;
        while ((n = await body.ReadAsync(buffer)) > 0)
        {
            Require(output.Length + n <= 524288, "METADATA_UNBOUNDED");
            output.Write(buffer, 0, n);
        }
        return output.ToArray();
    }

    private static Dictionary<string,long> CheckMetadata(byte[] raw)
    {
        using var doc = JsonDocument.Parse(raw, new JsonDocumentOptions
        {
            AllowTrailingCommas = false,
            CommentHandling = JsonCommentHandling.Disallow,
            MaxDepth = 12
        });
        var root = doc.RootElement;
        Require(root.ValueKind == JsonValueKind.Object &&
                root.GetProperty("id").GetInt64() == ReleaseId &&
                root.GetProperty("tag_name").GetString() == Tag &&
                root.GetProperty("draft").ValueKind == JsonValueKind.True &&
                root.GetProperty("published_at").ValueKind == JsonValueKind.Null,
                "DRAFT_RELEASE_IDENTITY");
        var assets = root.GetProperty("assets");
        Require(assets.ValueKind == JsonValueKind.Array &&
                assets.GetArrayLength() == Allowed.Length, "ASSET_COUNT");
        var found = new HashSet<long>();
        var chosen = new Dictionary<string,long>(StringComparer.Ordinal);
        foreach (var entry in assets.EnumerateArray())
        {
            long id = entry.GetProperty("id").GetInt64();
            long size = entry.GetProperty("size").GetInt64();
            string? name = entry.GetProperty("name").GetString();
            Require(id > 0 && found.Add(id) && name is not null,
                    "ASSET_ID_OR_NAME");
            var declared = Allowed.SingleOrDefault(a => a.Name == name);
            Require(declared is not null && declared.Length == size &&
                    (declared.Id == 0 || declared.Id == id),
                    "ASSET_ID_OR_SIZE");
            chosen.Add(name!,id);
        }
        Require(found.Count == Allowed.Length &&
                chosen.Count == Allowed.Length, "ASSET_SET");
        return chosen;
    }

    private static bool IsApprovedCdn(Uri uri) =>
        uri.Scheme == Uri.UriSchemeHttps &&
        uri.UserInfo.Length == 0 &&
        uri.Port == 443 &&
        (uri.Host.Equals("release-assets.githubusercontent.com", StringComparison.OrdinalIgnoreCase) ||
         uri.Host.Equals("objects.githubusercontent.com", StringComparison.OrdinalIgnoreCase));

    private static async Task FetchAsset(HttpClient http, Asset asset, long assetId,
                                         string token, string target)
    {
        bool cutProbe=asset.Name=="gf-e2e-new-win-x64.zip" &&
            Environment.GetEnvironmentVariable("GF_E2E_HTTP_CUT_NEGATIVE")=="1" &&
            Environment.GetEnvironmentVariable("GITHUB_ACTIONS")=="true" &&
            Environment.GetEnvironmentVariable("GITHUB_REPOSITORY")==Repo;
        Uri url;
        if(cutProbe)
        {
            Require(int.TryParse(Environment.GetEnvironmentVariable("GF_E2E_HTTP_CUT_PORT"),
                System.Globalization.NumberStyles.None,
                System.Globalization.CultureInfo.InvariantCulture,out int port) &&
                port>=1024 && port<=65535, "HTTP_CUT_LOOPBACK_PORT_INVALID");
            url=new Uri("http://127.0.0.1:"+port.ToString(
                System.Globalization.CultureInfo.InvariantCulture)+
                "/gf-e2e-new-win-x64.zip");
            Require(url.IsLoopback && url.Host=="127.0.0.1" && url.Scheme=="http" &&
                url.UserInfo.Length==0,"HTTP_CUT_ENDPOINT_NOT_LOOPBACK");
        }
        else
            url = new Uri("https://api.github.com/repos/" + Repo + "/releases/assets/" +
                assetId.ToString(System.Globalization.CultureInfo.InvariantCulture));
        // Bearer token is never sent to loopback or CDN.
        using var original = Get(url,cutProbe?null:token, metadata: false);
        using var initial = await http.SendAsync(original, HttpCompletionOption.ResponseHeadersRead);
        Require(initial.RequestMessage?.RequestUri == url, "ASSET_API_ORIGIN");
        HttpResponseMessage response = initial;
        HttpRequestMessage? redirectedRequest = null;
        HttpResponseMessage? redirectedResponse = null;
        try
        {
            if (initial.StatusCode == HttpStatusCode.Found ||
                initial.StatusCode == HttpStatusCode.TemporaryRedirect)
            {
                Uri? location = initial.Headers.Location;
                Require(location is not null && location.IsAbsoluteUri &&
                        IsApprovedCdn(location), "ASSET_REDIRECT_NOT_TRUSTED");
                // The bearer token must NEVER follow the redirect to the CDN.
                redirectedRequest = Get(location!, null, metadata: false);
                redirectedResponse = await http.SendAsync(redirectedRequest,
                    HttpCompletionOption.ResponseHeadersRead);
                response = redirectedResponse;
                Require(response.RequestMessage?.RequestUri == location,
                        "CDN_UNEXPECTED_REDIRECT");
            }
            Require(response.StatusCode == HttpStatusCode.OK &&
                    response.Headers.Location is null, "ASSET_HTTP_NOT_OK");
            Require(response.Content.Headers.ContentLength is null ||
                    response.Content.Headers.ContentLength == asset.Length,
                    "ASSET_CONTENT_LENGTH");
            Require(response.Content.Headers.ContentEncoding.Count == 0,
                    "ASSET_CONTENT_ENCODING");
            var mediaType = response.Content.Headers.ContentType?.MediaType;
            Require(mediaType is not ("application/json" or "text/html"),
                    "ASSET_JSON_OR_HTML_BODY");
            await using var remote = await response.Content.ReadAsStreamAsync();
            await using var output = new FileStream(target, FileMode.CreateNew,
                FileAccess.Write, FileShare.None, 65536, FileOptions.WriteThrough);
            using var hash = IncrementalHash.CreateHash(HashAlgorithmName.SHA256);
            long length = 0;
            var buffer = new byte[65536];
            int n;
            while ((n = await remote.ReadAsync(buffer)) > 0)
            {
                length = checked(length + n);
                Require(length <= asset.Length, "ASSET_OVERRUN");
                hash.AppendData(buffer, 0, n);
                await output.WriteAsync(buffer.AsMemory(0, n));
            }
            Require(length == asset.Length, "ASSET_INCOMPLETE");
            string sha = Convert.ToHexString(hash.GetHashAndReset()).ToLowerInvariant();
            if (asset.Sha is not null)
                Require(CryptographicOperations.FixedTimeEquals(
                    Convert.FromHexString(sha), Convert.FromHexString(asset.Sha)),
                    "ASSET_PINNED_HASH_MISMATCH");
            output.Flush(flushToDisk: true);
        }
        finally
        {
            redirectedResponse?.Dispose();
            redirectedRequest?.Dispose();
        }
    }

    // This is a lab-only owner of the existing Python-TUF -> C# L18 test
    // sequence, never a second selector/transaction or production update path.
    // The runner must prepare the PINNED bridge and signed fixtures in advance.
    private static async Task HandoffAuthenticatedDownloadToL18Async(string runner, Action<string> recordStage)
    {
        Require(Environment.GetEnvironmentVariable("GF_E2E_HANDOFF_L18") == "1" &&
                Environment.GetEnvironmentVariable("GITHUB_ACTIONS") == "true" &&
                Environment.GetEnvironmentVariable("GITHUB_REPOSITORY") == Repo,
                "L18_HANDOFF_UNAUTHORIZED");
        string repoRoot = Path.GetFullPath(
            Environment.GetEnvironmentVariable("GITHUB_WORKSPACE") ?? "");
        Require(Directory.Exists(repoRoot) &&
                File.Exists(Path.Combine(repoRoot, "GerenciadorAplicativos.csproj")),
                "L18_WORKSPACE_NOT_PINNED");
        string pythonRoot = Path.GetFullPath(
            Environment.GetEnvironmentVariable("pythonLocation") ?? "");
        string python = Path.Combine(pythonRoot, "python.exe");
        string script = Path.Combine(repoRoot, "prototipos", "autoatualizador-seguro",
                                    "phase1-composed", "phase1_lab.py");
        string tufScript = Path.Combine(repoRoot, "prototipos",
                    "autoatualizador-seguro", "phase1-composed", "github_draft_package.py");
        string anchor = Path.Combine(repoRoot, "prototipos",
                    "autoatualizador-seguro", "fixtures", "bootstrap.test-only.json");
        string bridge = Path.GetFullPath(
            Environment.GetEnvironmentVariable("GF_E2E_L18_EXE") ?? "");
        string rootPrefix = Path.TrimEndingDirectorySeparator(runner) +
                            Path.DirectorySeparatorChar;
        Require(File.Exists(python) && File.Exists(script) && File.Exists(tufScript) &&
                File.Exists(anchor) && File.Exists(bridge) &&
                bridge.StartsWith(rootPrefix, StringComparison.OrdinalIgnoreCase) &&
                Path.GetFileName(bridge) == "GF.Updater.M3.L18Tests.exe",
                "L18_TEST_BRIDGE_OR_TUF_MISSING");
        string fixtures = Path.Combine(runner, "gf-e2e-signed-fixtures");
        string old = Path.TrimEndingDirectorySeparator(AppContext.BaseDirectory);
        string download = Path.Combine(runner, "gf-e2e-github-download");
        string extracted = Path.Combine(runner, "gf-e2e-authenticated-new");
        string report = Path.Combine(runner, "gf-e2e-real-flow-report.json");
        Require(Directory.Exists(fixtures) && Directory.Exists(old) &&
                Directory.Exists(download) && !Directory.Exists(extracted) &&
                !File.Exists(report) &&
                old.Equals(Path.Combine(runner, "gf-e2e-old"),
                           StringComparison.OrdinalIgnoreCase),
                "L18_SOURCE_DESTINATION_OR_REPORT_NOT_FRESH");

        async Task RunScopedChild(string image, TimeSpan timeout, params string[] args)
        {
            var psi = new ProcessStartInfo(image)
            {
                UseShellExecute = false, CreateNoWindow = true,
                WorkingDirectory = repoRoot
            };
            foreach (string arg in args) psi.ArgumentList.Add(arg);
            // Never forward the draft-reading bearer token to Python or L18.
            psi.Environment.Remove("GH_TOKEN");
            psi.Environment.Remove("GITHUB_TOKEN");
            // L18 observers spawn real OLD and NEW, whose startup hook must
            // NOT re-enter the lab downloader recursively.
            psi.Environment["GF_E2E_STARTUP_DRAFT_PROBE"] = "0";
            psi.Environment["GF_E2E_HANDOFF_L18"] = "0";
            using var child = Process.Start(psi);
            Require(child is not null, "L18_VERIFIER_OR_COORDINATOR_NOT_STARTED");
            try
            {
                await child!.WaitForExitAsync().WaitAsync(timeout);
                Require(child.ExitCode == 0, "L18_VERIFIER_OR_COORDINATOR_REJECTED");
            }
            catch (TimeoutException)
            {
                if (!child!.HasExited)
                {
                    child.Kill(entireProcessTree: true);
                    await child.WaitForExitAsync();
                }
                throw new InvalidDataException("GF_E2E_OLD_L18_CHILD_TIMEOUT_BLOCKED");
            }
        }

        // The OLD application, not the CI shell, invokes the TEST TUF verifier.
        // It cannot execute NEW before Python-TUF has authenticated the ZIP
        // and all 284 decompressed files.
        recordStage("tuf_verify");
        await RunScopedChild(python, TimeSpan.FromMinutes(4),
            tufScript, "verify", "--repo", repoRoot, "--old", old,
            "--downloaded", download, "--extracted", extracted,
            "--lab-old-variant");
        Require(File.Exists(Path.Combine(extracted, "GerenciadorFinal.exe")) &&
                AssemblyName.GetAssemblyName(
                    Path.Combine(extracted, "GerenciadorFinal.dll")).Version ==
                    new Version(5,13,11,66),
                "L18_NEW_NOT_TUF_VERIFIED_OR_WRONG_VERSION");

        recordStage("tuf_verified");
        if (Environment.GetEnvironmentVariable("GF_E2E_SELECTED_NEW_CONTINUATION") == "1")
        {
            // TEST ONLY: the original OLD must exit before the LIVE L18
            // owner executes the first selected NEW/2. Start the same L18
            // process, do NOT wait for the completed recovery report while
            // keeping OLD's WinForms message loop alive.
            var psi = new ProcessStartInfo(bridge)
            {
                UseShellExecute = false, CreateNoWindow = true,
                WorkingDirectory = repoRoot
            };
            string l18Mode=
                Environment.GetEnvironmentVariable("GF_E2E_SWAP_CUT_AFTER_OLD_MOVE")=="1" &&
                Environment.GetEnvironmentVariable("GF_E2E_LIVE_INSTALL_SWAP")=="1" &&
                Environment.GetEnvironmentVariable("GITHUB_ACTIONS")=="true" &&
                Environment.GetEnvironmentVariable("GITHUB_REPOSITORY")==Repo
                ? "--real-install-cut-focused" :
                Environment.GetEnvironmentVariable("GF_E2E_LIVE_INSTALL_SWAP")=="1"
                ? "--real-install-focused" : "--real-handoff-focused";
            foreach (string arg in new []{l18Mode,report,fixtures,
                anchor,repoRoot,python,script,old,extracted,
                Environment.ProcessId.ToString(System.Globalization.CultureInfo.InvariantCulture)})
                psi.ArgumentList.Add(arg);
            psi.Environment.Remove("GH_TOKEN");
            psi.Environment.Remove("GITHUB_TOKEN");
            psi.Environment["GF_E2E_STARTUP_DRAFT_PROBE"]="0";
            psi.Environment["GF_E2E_HANDOFF_L18"]="0";
            psi.Environment["GF_E2E_SELECTED_NEW_CONTINUATION"]="0";
            recordStage("l18_bridge_start");
            using var bridgeChild = Process.Start(psi);
            Require(bridgeChild is not null,"L18_SELECTED_NEW_BRIDGE_NOT_CREATED");
            if (Environment.GetEnvironmentVariable("GF_E2E_POST_OWNER_PERSISTENCE") == "1")
            {
                // A process-identity diagnostic only. CI later compares
                // the start time to avoid mistaking PID reuse for L18 exit.
                // Neither this record nor process exit grants READY.
                using var diag=new FileStream(Path.Combine(runner,
                    "gf-e2e-original-l18-process.json"),FileMode.CreateNew,
                    FileAccess.Write,FileShare.None,4096,FileOptions.WriteThrough);
                byte[] bytes=JsonSerializer.SerializeToUtf8Bytes(new{
                    schema="gf-e2e-original-l18-process/1",
                    process_id=bridgeChild!.Id,
                    process_start_utc_ticks=bridgeChild.StartTime.ToUniversalTime().Ticks,
                    test_only=true,operational_ready=false
                });
                diag.Write(bytes);diag.Flush(true);
            }
            recordStage("l18_selector_wait");
            string selectedReady=Path.Combine(runner,"gf-e2e-selected-new-ready.json");
            var deadline=Stopwatch.StartNew();
            while(!File.Exists(selectedReady) && deadline.Elapsed<TimeSpan.FromMinutes(3))
            {
                Require(!bridgeChild.HasExited,"L18_SELECTED_NEW_BRIDGE_EXITED_BEFORE_ARM");
                await Task.Delay(150);
            }
            Require(File.Exists(selectedReady) && !bridgeChild.HasExited,
                "L18_SELECTED_NEW_BARRIER_UNPROVEN");
            using var ready=JsonDocument.Parse(File.ReadAllBytes(selectedReady));
            Require(ready.RootElement.GetProperty("schema").GetString() ==
                    "gf-e2e-selected-new-ready/1" &&
                ready.RootElement.GetProperty("selector").GetString()=="new" &&
                ready.RootElement.GetProperty("generation").GetInt32()==2 &&
                ready.RootElement.GetProperty("launch_fenced").GetBoolean() &&
                ready.RootElement.GetProperty("l18_live_owner").GetBoolean() &&
                !ready.RootElement.GetProperty("operational_ready").GetBoolean(),
                "L18_SELECTED_NEW_BARRIER_INVALID");
            // Return to the UI callback so ORIGINAL OLD closes gracefully.
            // The still-live L18 process owns all later NEW observation.
            return;
        }

        // Invoke the SAME tested L18 process coordinator with the original
        // 9-argument real-suite command line and unchanged fixture sources.
        await RunScopedChild(bridge, TimeSpan.FromMinutes(6),
            "--real-normal-focused", report, fixtures, anchor, repoRoot,
            python, script, old, extracted);

        // Validate a single real transition using the SAME C# L18 scenario.
        // All 29 adversarial regression scenarios were tested separately.
        using var reportDocument = JsonDocument.Parse(File.ReadAllBytes(report));
        JsonElement r = reportDocument.RootElement;
        JsonElement normal = r.GetProperty("result");
        Require(r.GetProperty("schema").GetString() ==
                    "gf-real-single-normal-lab/1" &&
                r.GetProperty("scenario").GetString() == "normal" &&
                r.GetProperty("passed").GetInt32() == 1 &&
                r.GetProperty("failed").GetInt32() == 0 &&
                !r.GetProperty("operational_ready").GetBoolean() &&
                !r.GetProperty("recovery_authorized").GetBoolean() &&
                !r.GetProperty("independent_recovery_authority").GetBoolean() &&
                !r.GetProperty("phase1_complete").GetBoolean() &&
                normal.GetProperty("test").GetString() == "normal" &&
                normal.GetProperty("status").GetString() == "passed" &&
                normal.GetProperty("classification").GetString() == "PASS_CONTAINMENT" &&
                normal.GetProperty("outcome").GetString() == "UnknownBlocked" &&
                normal.GetProperty("replay").GetString() == "UnknownBlocked" &&
                normal.GetProperty("normal_owner_completed_without_crash").GetBoolean() &&
                normal.GetProperty("reconciler_new_process").GetBoolean() &&
                normal.GetProperty("old_and_canary_preserved").GetBoolean() &&
                normal.GetProperty("manifest_and_object_identities_rechecked").GetBoolean(),
                "L18_NORMAL_NOT_CONTAINED_OR_DATA_NOT_PRESERVED");
        File.WriteAllText(Path.Combine(runner, "gf-e2e-old-l18-handoff-report.json"),
            JsonSerializer.Serialize(new {
                schema = "gf-e2e-old-owns-l18-handoff/2",
                initiated_by_gf_old_mainform = true,
                tuf_child_verified_download = true,
                same_l18_coordinator_invoked_by_old = true,
                l18_cases = 1,
                case_classification = "PASS_CONTAINMENT",
                normal_outcome = "UnknownBlocked",
                test_only = true, operational_ready = false,
                automatic_recovery_authorized = false
            }));
    }

    internal static async Task<int> Run(Action? afterVerifiedLabHandoff = null)
    {
        string stage="guard";
        bool httpCutNegative=Environment.GetEnvironmentVariable("GF_E2E_HTTP_CUT_NEGATIVE")=="1";
        bool guiNegative=httpCutNegative &&
            Environment.GetEnvironmentVariable("GF_E2E_HTTP_UI_NEGATIVE")=="1";
        bool normalGuiNegative=httpCutNegative &&
            Environment.GetEnvironmentVariable("GF_E2E_HTTP_UI_NORMAL_NEGATIVE")=="1";
        try
        {
            // Lab mode is absent from normal binaries; the additional guards
            // ensure a test-mode binary cannot run on personal Windows machines.
            Require(OperatingSystem.IsWindows() &&
                    Environment.GetEnvironmentVariable("GITHUB_ACTIONS") == "true" &&
                    Environment.GetEnvironmentVariable("GITHUB_REPOSITORY") == Repo &&
                    !string.IsNullOrWhiteSpace(Environment.GetEnvironmentVariable("RUNNER_TEMP")),
                    "DISPOSABLE_RUNNER_ONLY");
            var current = AssemblyName.GetAssemblyName(
                Path.Combine(AppContext.BaseDirectory, "GerenciadorFinal.dll")).Version;
            Require(current == new Version(5, 13, 11, 65),
                    "OLD_EXECUTABLE_VERSION_MISMATCH");
            string token = Environment.GetEnvironmentVariable("GH_TOKEN") ?? "";
            Require(token.Length >= 20 && token.Length <= 2048, "EPHEMERAL_GITHUB_TOKEN_MISSING");
            var runner = Path.GetFullPath(Environment.GetEnvironmentVariable("RUNNER_TEMP")!);
            Require(!(guiNegative && normalGuiNegative),
                "HTTP_GUI_NEGATIVE_MODES_MUTUALLY_EXCLUSIVE");
            string dest = Path.GetFullPath(Path.Combine(runner, normalGuiNegative
                ? "gf-e2e-http-normal-download"
                : guiNegative ? "gf-e2e-http-ui-download"
                : httpCutNegative ? "gf-e2e-http-cut-download"
                : "gf-e2e-github-download"));
            if(httpCutNegative)
                Require(Environment.GetEnvironmentVariable("GF_E2E_HANDOFF_L18")!="1" &&
                    Environment.GetEnvironmentVariable("GF_E2E_SELF_GRACEFUL_CLOSE")!="1" &&
                    Environment.GetEnvironmentVariable("GF_E2E_SELECTED_NEW_CONTINUATION")!="1" &&
                    Environment.GetEnvironmentVariable("GF_E2E_SWAP_CUT_AFTER_OLD_MOVE")!="1",
                    "HTTP_CUT_MUST_NEVER_HANDOFF");
            Require(Directory.Exists(runner) && !Directory.Exists(dest) &&
                    !File.Exists(dest) &&
                    Path.GetDirectoryName(dest) == runner, "DOWNLOAD_DESTINATION_NOT_FRESH");
            using var http = Client();
            // Draft tags are not reliably resolvable through releases/tags.
            // Read the authorized existing Draft by its pinned numeric ID.
            var endpoint = new Uri("https://api.github.com/repos/" + Repo +
                                   "/releases/" + ReleaseId.ToString(
                                       System.Globalization.CultureInfo.InvariantCulture));
            stage="draft_metadata";
            var assets = CheckMetadata(await ReadJson(http, endpoint, token));
            stage="asset_download";
            Directory.CreateDirectory(dest);
            foreach (var asset in Allowed)
            {
                await FetchAsset(http, asset, assets[asset.Name], token,
                    Path.Combine(dest, asset.Name));
                Require(!httpCutNegative,"HTTP_CUT_UNEXPECTED_COMPLETE_ZIP");
            }
            Require(Directory.GetFiles(dest).Length == Allowed.Length, "EXTRA_DOWNLOAD_FILE");
            stage="acquisition_report";
            var evidence = new
            {
                schema = "gf-e2e-old-github-draft/1",
                observed_version = current!.ToString(),
                draft_release_id = ReleaseId,
                draft_tag = Tag,
                downloaded_file_count = Allowed.Length,
                real_dotnet_httpclient = true,
                initiated_by_gf_old_executable = true,
                downloaded_archive_sha256 = ZipSha,
                test_only = true,
                operational_ready = false,
                automatic_recovery_authorized = false
            };
            File.WriteAllText(Path.Combine(runner, "gf-e2e-old-acquire-report.json"),
                JsonSerializer.Serialize(evidence));
            if (Environment.GetEnvironmentVariable("GF_E2E_HANDOFF_L18") == "1")
            {
                stage="handoff_entered";
                await HandoffAuthenticatedDownloadToL18Async(runner, value=>stage=value);
                // The completed, fail-closed L18 normal scenario leaves
                // UnknownBlocked: request a graceful OLD window exit ONLY
                // in the disposable laboratory, after the report is durable.
                if (Environment.GetEnvironmentVariable("GF_E2E_SELF_GRACEFUL_CLOSE") == "1" &&
                    (Environment.GetEnvironmentVariable("GF_E2E_SELECTED_NEW_CONTINUATION") == "1" ||
                     File.Exists(Path.Combine(runner,"gf-e2e-old-l18-handoff-report.json"))))
                    afterVerifiedLabHandoff?.Invoke();
            }
            return 0;
        }
        catch (Exception e) when (e is HttpRequestException or IOException or
                  InvalidDataException or JsonException or UnauthorizedAccessException or
                  OperationCanceledException or System.Security.SecurityException)
        {
            // Never record URL (could include signed CDN query) or bearer token.
            // The incomplete file remains for scoped runner diagnostics.
            string reason = e is InvalidDataException
                ? e.Message : e is HttpRequestException httpEx && httpEx.StatusCode is not null
                ? "HTTP_" + ((int)httpEx.StatusCode.Value).ToString(
                    System.Globalization.CultureInfo.InvariantCulture)
                : e.GetType().Name;
            if (Environment.GetEnvironmentVariable("GITHUB_ACTIONS") == "true")
            {
                try
                {
                    string temp = Environment.GetEnvironmentVariable("RUNNER_TEMP") ?? "";
                    if (Directory.Exists(temp))
                        File.WriteAllText(Path.Combine(temp, normalGuiNegative
                            ? "gf-e2e-http-normal-failure.json"
                            : guiNegative ? "gf-e2e-http-ui-failure.json"
                            : httpCutNegative ? "gf-e2e-http-cut-failure.json"
                            : "gf-e2e-old-acquire-failure.json"),
                            JsonSerializer.Serialize(new { schema = "gf-e2e-old-blocked/2", reason,
                                stage, io_hresult = e is IOException ? e.HResult : 0,
                                no_token_logged = true, no_auto_recovery = true }));
                }
                catch (IOException) { }
            }
            if(httpCutNegative)
                Console.WriteLine("GF_E2E_HTTP_CUT_OLD_BLOCKED "
                    +"stage="+stage+" failure_category="+
                    (e is IOException ? "IO" : e is HttpRequestException ? "HTTP" : "OTHER")
                    +" NEW_executed=false READY=false");
            return 8;
        }
    }
