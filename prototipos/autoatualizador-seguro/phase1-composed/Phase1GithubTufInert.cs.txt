// Data-only PRE-HANDOFF gate inside the EXISTING L18 laboratory executable.
// NEVER enters L18 RealOwner, writes a selector/journal, or grants launch/READY.
// It invokes the existing Python-TUF verifier with the received ZIP pinned
// read-only, then rechecks the test-only signed manifest and full file set.
using System.Diagnostics;
using System.Reflection;

[System.Runtime.Versioning.SupportedOSPlatform("windows")]

{
    private const string Repo="walissonminel-1991/gerenciador-final";
    private const string Archive="gf-e2e-new-win-x64.zip";
    private const string Manifest="GF-LAB-REAL-PACKAGE-MANIFEST.json";
    private static readonly string[] Roles=[Manifest,"targets.json","snapshot.json","timestamp.json"];

    private static string Under(string path,string runner,bool exists)
    {
        string full=Path.GetFullPath(path);
        Require(full.StartsWith(runner+Path.DirectorySeparatorChar,
            StringComparison.OrdinalIgnoreCase),"GH_L18_INERT_RUNNER_SCOPE");
        if(exists)Require(File.Exists(full)||Directory.Exists(full),
            "GH_L18_INERT_PATH_ABSENT");
        return full;
    }
    private static FileStream Pin(string path,long limit)
    {
        Require(File.Exists(path) && (File.GetAttributes(path)&FileAttributes.ReparsePoint)==0,
            "GH_L18_INERT_REPARSE_OR_ABSENT");
        var file=new FileStream(path,FileMode.Open,FileAccess.Read,FileShare.Read);
        try {
            Require(file.Length is >0 && file.Length<=limit,
                "GH_L18_INERT_LENGTH_INVALID");
            return file;
        } catch {file.Dispose();throw;}
    }
    private static string HashStream(FileStream file)
    {
        file.Position=0;
        string sha=Convert.ToHexString(SHA256.HashData(file)).ToLowerInvariant();
        file.Position=0;
        return sha;
    }
    // A snapshot of every ledger generation, not just generation 1.
    // Hold NTFS file IDs and ancestor names throughout the same process.
    // This does NOT prevent an attacker replaying a coherent history AFTER
    // these handles are closed. Such a restart has NO authorization here.
    private sealed class PinnedLedger : IDisposable
    {
        private readonly string root;
        private readonly string[] paths;
        private readonly List<WindowsFileGuard> pinned = [];
        private readonly List<byte[]> contents = [];
        private readonly List<string> identities = [];
        internal int Count => paths.Length;
        internal string TailPath => paths[^1];
        internal byte[] TailBytes => contents[^1];

        internal PinnedLedger(string root)
        {
            this.root = root;
            paths = Directory.GetFiles(Path.Combine(root,"ledger"))
                .Order(StringComparer.Ordinal).ToArray();
            Require(paths.Length is >=2 and <=64,
                "GH_L18_INERT_MULTIGENERATION_REQUIRED");
            try
            {
                for(int i=0;i<paths.Length;i++)
                {
                    string expected=Path.Combine(root,"ledger",
                        (i+1).ToString("D16",
                            System.Globalization.CultureInfo.InvariantCulture)
                        +".fixture.json");
                    Require(paths[i].Equals(expected,
                        StringComparison.OrdinalIgnoreCase),
                        "GH_L18_INERT_LEDGER_SEQUENCE_OR_PENDING");
                    var guard=WindowsFileGuard.Open(root,paths[i]);
                    pinned.Add(guard);
                    contents.Add(guard.Read(StateCodec.MaxRecordBytes));
                    identities.Add(guard.Identity);
                }
            }
            catch {Dispose();throw;}
        }
        internal void Recheck()
        {
            string[] current=Directory.GetFiles(Path.Combine(root,"ledger"))
                .Order(StringComparer.Ordinal).ToArray();
            Require(current.SequenceEqual(paths,StringComparer.Ordinal),
                "GH_L18_INERT_LEDGER_NAMESPACE_CHANGED");
            for(int i=0;i<paths.Length;i++)
            {
                pinned[i].Recheck();
                Require(pinned[i].Identity==identities[i] &&
                    pinned[i].Read(StateCodec.MaxRecordBytes).AsSpan()
                        .SequenceEqual(contents[i]),
                    "GH_L18_INERT_LEDGER_OBJECT_OR_BYTES_CHANGED");
            }
        }
        public void Dispose()
        {
            for(int i=pinned.Count-1;i>=0;i--)pinned[i].Dispose();
            pinned.Clear();
        }
    }

    internal static async Task Run(string[] a)
    {
        Require(a.Length==12 && a[0]=="--real-github-tuf-inert" &&
            OperatingSystem.IsWindows() &&
            Environment.GetEnvironmentVariable("GITHUB_ACTIONS")=="true" &&
            Environment.GetEnvironmentVariable("GITHUB_REPOSITORY")==Repo &&
            Environment.GetEnvironmentVariable("GF_GITHUB_TUF_ONLY")=="1" &&
            Path.GetFileName(Environment.ProcessPath)=="GF.Updater.M3.L18Tests.exe",
            "GH_L18_INERT_CI_ONLY");
        string runner=Path.TrimEndingDirectorySeparator(Path.GetFullPath(
            Environment.GetEnvironmentVariable("RUNNER_TEMP") ??
            throw new Rejected("GH_L18_INERT_RUNNER_REQUIRED")));
        string repository=Path.GetFullPath(a[4]);
        Require(repository==Path.GetFullPath(
            Environment.GetEnvironmentVariable("GITHUB_WORKSPACE") ?? "") &&
            File.Exists(Path.Combine(repository,"GerenciadorAplicativos.csproj")),
            "GH_L18_INERT_REPOSITORY");
        string python=Path.GetFullPath(a[5]);
        Require(python.Equals(Path.Combine(
             Environment.GetEnvironmentVariable("pythonLocation") ?? "",
             "python.exe"),StringComparison.OrdinalIgnoreCase) &&
             File.Exists(python),"GH_L18_INERT_PYTHON");
        string old=Under(a[1],runner,true), bundle=Under(a[2],runner,true);
        string extracted=Under(a[3],runner,false);
        // L14 and PersistentState deliberately require OS GetTempPath(),
        // which may differ from RUNNER_TEMP on hosted Windows images.
        // These two fixture-only roots are direct children of the OS temp
        // directory, still under the disposable CI guard above.
        string osTemp=Path.TrimEndingDirectorySeparator(Path.GetFullPath(Path.GetTempPath()));
        string stateRoot=Under(a[10],osTemp,true), labRoot=Under(a[11],osTemp,true);
        Require(Path.GetFileName(stateRoot).StartsWith("gf-u2-github-binding-",StringComparison.Ordinal) &&
            Path.GetFileName(labRoot).StartsWith("gf-u2-l14-",StringComparison.Ordinal) &&
            Path.GetDirectoryName(labRoot)==osTemp,
            "GH_L18_INERT_TEST_ROOTS_ONLY");
        Require(Directory.Exists(old) && Directory.Exists(bundle) &&
            !Directory.Exists(extracted) && !File.Exists(extracted) &&
            a[6].Length==64 && a[7].Length==64 &&
            a[6].All(Uri.IsHexDigit) && a[7].All(Uri.IsHexDigit) &&
            a[8].Length==32 && a[9].Length==32 &&
            a[8].All(c=>c is >= '0' and <= '9' or >= 'a' and <= 'f') &&
            a[9].All(c=>c is >= '0' and <= '9' or >= 'a' and <= 'f'),
            "GH_L18_INERT_ARGUMENTS");
        // Snapshot all generations before invoking the TUF subprocess.
        using var ledger=new PinnedLedger(stateRoot);
        var pins=new List<FileStream>();
        try {
            // Retain read-only handles throughout Python-TUF authentication,
            // refusing concurrent overwrite/delete; this alone is NOT the
            // production L18 NTFS no-follow/object-identity proof.
            foreach(string name in new[]{Archive}.Concat(Roles))
                pins.Add(Pin(Path.Combine(bundle,name),name==Archive?250_000_000:2_097_152));
            string actualZip=HashStream(pins[0]);
            string actualManifest=HashStream(pins[1]);
            Require(actualZip.Equals(a[6],StringComparison.OrdinalIgnoreCase),
                "GH_L18_INERT_ZIP_MISMATCH");
            Require(actualManifest.Equals(a[7],StringComparison.OrdinalIgnoreCase),
                "GH_L18_INERT_MANIFEST_MISMATCH");
            string verifier=Path.Combine(repository,"prototipos",
                "autoatualizador-seguro","phase1-composed","github_draft_package.py");
            Require(File.Exists(verifier),"GH_L18_INERT_TUF_VERIFIER_ABSENT");
            var psi=new ProcessStartInfo(python) {
                UseShellExecute=false,CreateNoWindow=true,
                RedirectStandardOutput=true,RedirectStandardError=true,
                WorkingDirectory=repository
            };
            foreach(string arg in new[]{verifier,"verify","--repo",repository,
                "--old",old,"--downloaded",bundle,"--extracted",extracted})
                psi.ArgumentList.Add(arg);
            // The verifier never needs an OAuth token.
            psi.Environment.Remove("GH_TOKEN");
            psi.Environment.Remove("GITHUB_TOKEN");
            using var proc=Process.Start(psi) ??
                throw new Rejected("GH_L18_INERT_VERIFIER_NOT_STARTED");
            Task<string> output=proc.StandardOutput.ReadToEndAsync();
            Task<string> error=proc.StandardError.ReadToEndAsync();
            try {await proc.WaitForExitAsync().WaitAsync(TimeSpan.FromMinutes(4));}
            catch(TimeoutException) {
                if(!proc.HasExited){proc.Kill(entireProcessTree:true);await proc.WaitForExitAsync();}
                throw new Rejected("GH_L18_INERT_TUF_TIMEOUT");
            }
            string stdout=await output;
            _=await error; // do not log raw stderr (may contain test paths)
            Require(proc.ExitCode==0 && stdout.Contains("GF_DRAFT_AUTHENTICATED ",
                StringComparison.Ordinal),"GH_L18_INERT_TUF_DENIED");
            // Synthetic abrupt process termination at the precise data-only
            // boundary. There is no L18 context, selector, phase journal,
            // installer or NEW launch to resume. The OS closes every handle.
            // NEVER enable this cut except in the disposable CI-only mode.
            if(Environment.GetEnvironmentVariable(
                "GF_GITHUB_TUF_INERT_CUT_AFTER_TUF")=="1")
            {
                Console.WriteLine("GF_GITHUB_TUF_INERT_PRECOMMIT_CUT " +
                    "tuf_checked=true context=false selector=false " +
                    "journal=false NEW=false READY=false");
                Console.Out.Flush();
                Environment.Exit(29);
            }
            // Recheck pinned bytes after the authenticator, and check its
            // target against the exact extracted file set without any launch.
            Require(HashStream(pins[0])==actualZip &&
                HashStream(pins[1])==actualManifest,
                "GH_L18_INERT_PIN_CHANGED");
            using var doc=JsonDocument.Parse(pins[1]);
            var root=doc.RootElement;
            Require(root.GetProperty("schema").GetString()=="gf-real-package-lab/3" &&
                root.GetProperty("selected_before").GetString()=="old" &&
                !root.GetProperty("operational_installable").GetBoolean() &&
                root.GetProperty("installation").GetString()==a[8] &&
                root.GetProperty("transaction").GetString()==a[9] &&
                a[8]==new string('a',32) && a[9]==new string('b',32),
                "GH_L18_INERT_SIGNED_SCOPE");
            // Local context is decoded by the EXISTING L18 persistent
            // ledger parser. These values are test-only fixture state, NOT
            // publisher authority, recovery permission, or a launch ticket.
            // The signed manifest and trusted local IDs must agree; a caller
            // supplied ID, a receipt, or local SHA cannot substitute either.
            try { L14Model.RequireRoot(labRoot); }
            catch(InvalidDataException e)
            {
                // Only a static L14 error code, no path, SID or secrets.
                Console.Error.WriteLine("GF_GITHUB_TUF_L14_SCOPE_BLOCK " +
                    (e.Message.StartsWith("L14_",StringComparison.Ordinal) &&
                     e.Message.Length<80 ? e.Message : "INVALID"));
                throw;
            }
            try
            {
            var loaded=new PersistentState(stateRoot).Load().Record;
            Require(loaded.InstallationId==a[8] &&
                L14Model.Tx(labRoot)==a[9] &&
                loaded.Generation==ledger.Count && loaded.Transaction is null &&
                loaded.Confirmed.Sequence==1 &&
                loaded.Trust.ApprovedSequence==1,
                "GH_L18_INERT_PERSISTENT_IDENTITY_OR_TRANSACTION");
            // Confront the trusted floor with the TUF roles that Python-TUF
            // already authenticated. Reject replay/rollback; do not create
            // any new signature or advance the persistent floor.
            int RoleVersion(int index)
            {
                pins[index].Position=0;
                using var role=JsonDocument.Parse(pins[index]);
                int version=role.RootElement.GetProperty("signed")
                    .GetProperty("version").GetInt32();
                pins[index].Position=0;
                return version;
            }
            Require(loaded.Trust.Root==1 &&
                loaded.Trust.Targets<=RoleVersion(2) &&
                loaded.Trust.Snapshot<=RoleVersion(3) &&
                loaded.Trust.Timestamp<=RoleVersion(4),
                "GH_L18_INERT_TRUST_FLOOR_ROLLBACK");
            }
            catch(Rejected reason)
            {
                // Rejected.Code is a static internal validation code.
                Console.Error.WriteLine("GF_GITHUB_TUF_STATE_SCOPE_BLOCK "+
                    (reason.Code.Length<100 ? reason.Code : "INVALID"));
                throw;
            }
            // Reuse the same data-only file-set proof for BOTH OLD and NEW.
            // The previously signed OLD identity must match the actual OLD
            // bytes; a locally re-signed NEW must never confer OLD authority.
            void CheckSignedTree(string directory,JsonElement files,string label)
            {
                Require(files.ValueKind==JsonValueKind.Object &&
                    files.EnumerateObject().Count()>40,
                    "GH_L18_INERT_"+label+"_SET_INVALID");
                string[] names=Directory.EnumerateFiles(directory,"*",SearchOption.AllDirectories)
                    .Select(p=>Path.GetRelativePath(directory,p).Replace('\\','/'))
                    .Order(StringComparer.Ordinal).ToArray();
                string[] signed=files.EnumerateObject().Select(p=>p.Name)
                    .Order(StringComparer.Ordinal).ToArray();
                Require(names.SequenceEqual(signed,StringComparer.Ordinal),
                    "GH_L18_INERT_"+label+"_SET_FORK");
                foreach(string rel in signed) {
                    string full=Path.GetFullPath(Path.Combine(directory,
                        rel.Replace('/',Path.DirectorySeparatorChar)));
                    Require(full.StartsWith(directory+Path.DirectorySeparatorChar,
                        StringComparison.OrdinalIgnoreCase) &&
                        (File.GetAttributes(full)&FileAttributes.ReparsePoint)==0,
                        "GH_L18_INERT_"+label+"_PATH");
                    var info=files.GetProperty(rel);
                    using var f=Pin(full,67_108_864);
                    Require(f.Length==info.GetProperty("length").GetInt64() &&
                        HashStream(f).Equals(info.GetProperty("sha256").GetString(),
                            StringComparison.OrdinalIgnoreCase),
                        "GH_L18_INERT_"+label+"_SIGNED_FILE_CHANGED");
                }
            }
            CheckSignedTree(old,root.GetProperty("old_files"),"OLD");
            CheckSignedTree(extracted,root.GetProperty("files"),"NEW");
            ledger.Recheck();
            // Same bytes under a different NTFS file object MUST NOT
            // replace a pinned ledger or signed manifest. These attempts
            // operate only on disposable CI fixture paths; no ACL is changed.
            // If the OS unexpectedly permits replacement, fail the proof.
            static void RequireDeniedReplacement(string original,byte[] bytes)
            {
                string candidate=original+".ntfs-object-replace-inert";
                Require(!File.Exists(candidate),"GH_L18_INERT_REPLACE_NOT_FRESH");
                File.WriteAllBytes(candidate,bytes);
                bool refused=false;
                try { File.Move(candidate,original,overwrite:true); }
                catch(IOException) { refused=true; }
                catch(UnauthorizedAccessException) { refused=true; }
                finally { if(File.Exists(candidate))File.Delete(candidate); }
                Require(refused,"GH_L18_INERT_OBJECT_REPLACEMENT_ALLOWED");
            }
            RequireDeniedReplacement(ledger.TailPath,ledger.TailBytes);
            RequireDeniedReplacement(Path.Combine(bundle,Manifest),
                File.ReadAllBytes(Path.Combine(bundle,Manifest))); // exactly the same bytes
            ledger.Recheck();
            Require(HashStream(pins[1])==actualManifest,
                "GH_L18_INERT_MANIFEST_CHANGED_AFTER_DENIED_SWAP");
            var version=AssemblyName.GetAssemblyName(
                Path.Combine(extracted,"GerenciadorFinal.dll")).Version;
            Require(version==new Version(5,13,11,66),
                "GH_L18_INERT_NEW_VERSION_WRONG");
            // A synthetic same-process handoff barrier. Standard input
            // synchronizes a disposable test; it is NEVER an authority.
            // Keep the exact OLD/NEW file objects open across the barrier.
            string? continuous=Environment.GetEnvironmentVariable(
                "GF_GITHUB_TUF_INERT_CONTINUOUS");
            if(continuous is not null)
            {
                Require(continuous is "normal" or "drop" or "cut",
                    "GH_L18_CONTINUOUS_MODE_DENIED");
                var retained=new List<(string Path,FileStream Handle,long Length,string Sha)>();
                try
                {
                    foreach(var pair in new[]{
                        (Directory:old, Signed:root.GetProperty("old_files")),
                        (Directory:extracted, Signed:root.GetProperty("files"))})
                    {
                        foreach(var entry in pair.Signed.EnumerateObject())
                        {
                            string full=Path.GetFullPath(Path.Combine(
                                pair.Directory,
                                entry.Name.Replace('/',Path.DirectorySeparatorChar)));
                            Require(full.StartsWith(pair.Directory+Path.DirectorySeparatorChar,
                                StringComparison.OrdinalIgnoreCase),
                                "GH_L18_CONTINUOUS_PATH_ESCAPE");
                            var handle=Pin(full,67_108_864);
                            retained.Add((full,handle,
                                entry.Value.GetProperty("length").GetInt64(),
                                entry.Value.GetProperty("sha256").GetString() ?? ""));
                            Require(handle.Length==retained[^1].Length &&
                                HashStream(handle).Equals(retained[^1].Sha,
                                    StringComparison.OrdinalIgnoreCase),
                                "GH_L18_CONTINUOUS_INITIAL_BYTES");
                        }
                    }
                    var initialState=new PersistentState(stateRoot).Load();
                    string[] originalRoles=pins.Select(HashStream).ToArray();
                    ledger.Recheck();
                    // This marker is only an ephemeral synchronization point.
                    Console.WriteLine("GF_GITHUB_TUF_INERT_CONTINUOUS_BARRIER " +
                        "pinned=true decision=false journal=false selector=false READY=false");
                    Console.Out.Flush();
                    if(continuous=="cut")
                    {
                        Environment.Exit(39);
                    }
                    string? signal=await Console.In.ReadLineAsync()
                        .WaitAsync(TimeSpan.FromSeconds(90));
                    Require(signal=="resume",
                        "GH_L18_CONTINUOUS_UNTRUSTED_SIGNAL");
                    if(continuous=="drop")
                    {
                        retained[0].Handle.Dispose();
                        Require(false,"GH_L18_CONTINUOUS_PIN_LOST");
                    }
                    // The actual security decision comes solely from fresh
                    // verification of trusted data, not from stdin or a receipt.
                    foreach(var pair in new[]{
                        (Directory:old,Signed:root.GetProperty("old_files")),
                        (Directory:extracted,Signed:root.GetProperty("files"))})
                    {
                        string[] names=Directory.EnumerateFiles(pair.Directory,
                                "*",SearchOption.AllDirectories)
                            .Select(x=>Path.GetRelativePath(pair.Directory,x)
                                .Replace('\\','/'))
                            .Order(StringComparer.Ordinal).ToArray();
                        string[] signed=pair.Signed.EnumerateObject()
                            .Select(x=>x.Name).Order(StringComparer.Ordinal).ToArray();
                        Require(names.SequenceEqual(signed,StringComparer.Ordinal),
                            "GH_L18_CONTINUOUS_NAMESPACE_CHANGED");
                    }
                    foreach(var item in retained)
                    {
                        Require((File.GetAttributes(item.Path)&FileAttributes.ReparsePoint)==0 &&
                            item.Handle.Length==item.Length &&
                            HashStream(item.Handle).Equals(item.Sha,
                                StringComparison.OrdinalIgnoreCase),
                            "GH_L18_CONTINUOUS_FINAL_BYTES");
                    }
                    ledger.Recheck();
                    var finalState=new PersistentState(stateRoot).Load();
                    Require(finalState.Digest==initialState.Digest &&
                        finalState.Record.InstallationId==a[8] &&
                        finalState.Record.Generation==ledger.Count &&
                        L14Model.Tx(labRoot)==a[9],
                        "GH_L18_CONTINUOUS_STATE_OR_TX_CHANGED");
                    Require(pins.Select(HashStream)
                            .SequenceEqual(originalRoles,StringComparer.Ordinal),
                        "GH_L18_CONTINUOUS_TUF_OBJECT_CHANGED");
                    Console.WriteLine("GF_GITHUB_TUF_INERT_CONTINUOUS_PASS " +
                        "same_process=true tuf_preverified=true held_old_new=true " +
                        "ledger_multigeneration=true tx_rechecked=true " +
                        "no_authority_from_stdin=true selector=false journal=false " +
                        "NEW=false READY=false");
                }
                finally
                {
                    foreach(var item in retained)item.Handle.Dispose();
                }
            }
            // This is NOT PrepareRealAuthorization, RealOwner or a selector.
            // The same L18 test executable has accepted only DATA-ONLY input.
            Console.WriteLine("GF_GITHUB_TUF_L18_INERT_PASS " +
                "l18_executable=EXISTING tif=TEST_ONLY signed_manifest=true " +
                "zip_same_bytes=true old_new_signed_files=true persisted_fixture_ids_match=true multigeneration_ledger_pinned=true trust_floor_not_rolled_back=true latest_ledger_object_swap=REJECTED ledger_object_swap=REJECTED manifest_object_swap=REJECTED worker=false " +
                "selector=false journal=false NEW_not_started=true READY=false");
        } finally {foreach(var f in pins)f.Dispose();}
    }
