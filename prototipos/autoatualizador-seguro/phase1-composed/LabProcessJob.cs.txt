internal sealed class LabProcessJob : IDisposable
{
    private readonly SafeFileHandle handle;
    private LabProcessJob(SafeFileHandle value){handle=value;CheckPolicy();}
    internal static LabProcessJob Create(string name)
    {
        var h=CreateJobObjectW(IntPtr.Zero,name);
        int error=Marshal.GetLastWin32Error();
        if(h.IsInvalid){h.Dispose();throw new Win32Exception(error);}
        if(error==183){h.Dispose();throw new Rejected("REAL_JOB_ALREADY_EXISTS");}
        var limits=new ExtendedLimits(); limits.Basic.Flags=0x2000; // kill on last close
        if(!SetInformationJobObject(h,9,ref limits,(uint)Marshal.SizeOf<ExtendedLimits>()))
        {error=Marshal.GetLastWin32Error();h.Dispose();throw new Win32Exception(error);}
        return new(h);
    }
    internal static LabProcessJob Open(string name)
    {
        var h=OpenJobObjectW(4,false,name); // query only; no kill authority in reconciler
        if(h.IsInvalid){int e=Marshal.GetLastWin32Error();h.Dispose();throw new Win32Exception(e,"REAL_JOB_WITNESS_UNAVAILABLE");}
        return new(h);
    }
    // A name is NOT an object identity. The owner Job must be inherited
    // directly from the surviving test supervisor into the new reconciler.
    // A newly created Job with the same name must never substitute for it.
    internal static LabProcessJob FromInherited(string name,long raw)
    {
        Require(raw>0,"REAL_OWNER_WITNESS_NOT_INHERITED");
        using var named=Open(name);
        var handle=new SafeFileHandle(new IntPtr(raw),true);
        try
        {
            var owner=new LabProcessJob(handle);
            Require(CompareObjectHandles(owner.handle,named.handle),
                "REAL_OWNER_WITNESS_OBJECT_MISMATCH");
            return owner;
        }
        catch {handle.Dispose();throw;}
    }
    private void CheckPolicy()
    {
        if(!QueryInformationJobObject(handle,9,out ExtendedLimits limits,
            (uint)Marshal.SizeOf<ExtendedLimits>(),IntPtr.Zero))throw new Win32Exception(Marshal.GetLastWin32Error());
        Require(limits.Basic.Flags==0x2000,"REAL_JOB_BREAKAWAY_OR_POLICY_CHANGED");
    }
    internal uint Active
    {
        get
        {
            CheckPolicy();
            if(!QueryInformationJobObject(handle,1,out Accounting a,
                (uint)Marshal.SizeOf<Accounting>(),IntPtr.Zero))throw new Win32Exception(Marshal.GetLastWin32Error());
            return a.Active;
        }
    }
    internal string Diagnostic()
    {
        // Image names/counts are diagnostic, never identity or authorization.
        IntPtr buffer=Marshal.AllocHGlobal(8+32*IntPtr.Size);
        try
        {
            if(!QueryInformationJobObject(handle,3,buffer,(uint)(8+32*IntPtr.Size),IntPtr.Zero))
                return "active="+Active+"; member-query-error="+Marshal.GetLastWin32Error();
            int count=Marshal.ReadInt32(buffer,4);
            Require(count is >=0 and <=32,"REAL_JOB_DIAGNOSTIC_BUDGET");
            var names=new List<string>();
            for(int i=0;i<count;i++)
            {
                int pid=checked((int)Marshal.ReadIntPtr(buffer,8+i*IntPtr.Size));
                try{using var p=System.Diagnostics.Process.GetProcessById(pid);names.Add(p.ProcessName);}
                catch(ArgumentException){names.Add("exited-during-diagnostic");}
                catch(InvalidOperationException){names.Add("unavailable");}
            }
            return "active="+Active+"; members="+string.Join(",",names);
        }
        finally{Marshal.FreeHGlobal(buffer);}
    }
    internal void RequireCurrentMember()
    {
        if(!IsProcessInJob(GetCurrentProcess(),handle,out bool member))throw new Win32Exception(Marshal.GetLastWin32Error());
        Require(member,"REAL_RECONCILER_OUTSIDE_CONTAINMENT");
    }
    // Process membership is checked against the kernel Job handle. This is
    // evidence of a test broker outside the Job, not a trust/READY witness.
    internal bool ContainsProcess(Process process)
    {
        if(!IsProcessInJob(process.Handle,handle,out bool member))
            throw new Win32Exception(Marshal.GetLastWin32Error());
        return member;
    }
    internal LabChild Start(string exe,params string[] args)
        => StartCore(exe,args,null);
    internal LabChild StartWithWitness(LabProcessJob owner,string exe,params string[] args)
        => StartCore(exe,args,owner);
    private LabChild StartCore(string exe,string[] args,LabProcessJob? owner)
    {
        // Assign at creation with JOB_LIST. If this is a reconciler, inherit
        // ONLY the original owner Job's duplicated handle via HANDLE_LIST;
        // the handle is a retained lab capability, not a durable witness.
        // Interrupted CreateProcess internals / physical power loss unproven.
        IntPtr size=IntPtr.Zero;
        _=InitializeProcThreadAttributeList(IntPtr.Zero,owner is null?1:2,0,ref size);
        Require(size.ToInt64() is >0 and <65536,"REAL_JOB_ATTRIBUTE_SIZE");
        IntPtr attributes=Marshal.AllocHGlobal(size),jobs=Marshal.AllocHGlobal(IntPtr.Size);
        IntPtr handleList=IntPtr.Zero, inherited=IntPtr.Zero;
        bool initialized=false;
        try
        {
            if(owner is not null)
            {
                if(!DuplicateHandle(GetCurrentProcess(),owner.handle.DangerousGetHandle(),
                    GetCurrentProcess(),out inherited,0,true,2))
                    throw new Win32Exception(Marshal.GetLastWin32Error(),"REAL_OWNER_WITNESS_DUPLICATE_FAILED");
                handleList=Marshal.AllocHGlobal(IntPtr.Size);
                Marshal.WriteIntPtr(handleList,inherited);
            }
            if(!InitializeProcThreadAttributeList(attributes,owner is null?1:2,0,ref size))
                throw new Win32Exception(Marshal.GetLastWin32Error());
            initialized=true;Marshal.WriteIntPtr(jobs,handle.DangerousGetHandle());
            if(!UpdateProcThreadAttribute(attributes,0,(IntPtr)0x2000d,jobs,
                (IntPtr)IntPtr.Size,IntPtr.Zero,IntPtr.Zero))
                throw new Win32Exception(Marshal.GetLastWin32Error());
            if(owner is not null &&
                !UpdateProcThreadAttribute(attributes,0,(IntPtr)0x20002,handleList,
                    (IntPtr)IntPtr.Size,IntPtr.Zero,IntPtr.Zero))
                throw new Win32Exception(Marshal.GetLastWin32Error(),"REAL_HANDLE_LIST_FAILURE");
            var startup=new StartupEx { Info=new Startup { Size=Marshal.SizeOf<StartupEx>() }, Attributes=attributes };
            var allArgs=owner is null?args:args.Append(inherited.ToInt64()
                .ToString(System.Globalization.CultureInfo.InvariantCulture)).ToArray();
            var command=new StringBuilder(string.Join(" ",new[]{exe}.Concat(allArgs).Select(Quote)));
            // DETACHED_PROCESS avoids a headless conhost in the Job.
            // HANDLE_LIST prevents accidental inheritance of unrelated handles.
            if(!CreateProcessW(exe,command,IntPtr.Zero,IntPtr.Zero,owner is not null,
                0x00080008,IntPtr.Zero,Path.GetDirectoryName(exe),ref startup,out var process))
                throw new Win32Exception(Marshal.GetLastWin32Error(),"REAL_JOB_CREATE_FAILED");
            CloseHandle(process.Thread);
            return new LabChild(new SafeFileHandle(process.Process,true));
        }
        finally
        {
            if(initialized)DeleteProcThreadAttributeList(attributes);
            if(inherited!=IntPtr.Zero)CloseHandle(inherited);
            if(handleList!=IntPtr.Zero)Marshal.FreeHGlobal(handleList);
            Marshal.FreeHGlobal(attributes);Marshal.FreeHGlobal(jobs);
        }
    }
    // CommandLineToArgvW/CRT quoting, including trailing backslashes.
    private static string Quote(string value)
    {
        Require(!value.Contains('\0'),"REAL_ARGUMENT_NUL");
        var text=new StringBuilder("\"");int slashes=0;
        foreach(char c in value)
        {
            if(c=='\\'){slashes++;continue;}
            text.Append('\\',c=='"'?slashes*2+1:slashes);text.Append(c);slashes=0;
        }
        return text.Append('\\',slashes*2).Append('"').ToString();
    }
    internal async Task Cleanup()
    {
        if(Active!=0 && !TerminateJobObject(handle,93))throw new Win32Exception(Marshal.GetLastWin32Error());
        var watch=System.Diagnostics.Stopwatch.StartNew();
        while(Active!=0 && watch.Elapsed<TimeSpan.FromSeconds(15))await Task.Delay(25);
        Require(Active==0,"REAL_JOB_CLEANUP_UNVERIFIED");
    }
    internal async Task RequireOnlyCurrentProcess()
    {
        RequireCurrentMember();
        var watch=System.Diagnostics.Stopwatch.StartNew();
        // A terminated redirected Python console can have a retiring conhost.
        // Wait for ACTUAL membership to end; never subtract a process by name.
        while(Active!=1 && watch.Elapsed<TimeSpan.FromSeconds(5))await Task.Delay(25);
        Require(Active==1,"REAL_CONTAINED_PROCESSES_STILL_ACTIVE");
    }
    public void Dispose()=>handle.Dispose();

internal sealed class LabChild(SafeFileHandle handle) : IDisposable
{
    internal bool Exited=>WaitForSingleObject(handle,0) switch
    {0=>true,258=>false,_=>throw new Win32Exception(Marshal.GetLastWin32Error())};
    internal uint ExitCode
    {get {Require(Exited,"REAL_CHILD_STILL_ACTIVE");if(!GetExitCodeProcess(handle,out uint code))throw new Win32Exception(Marshal.GetLastWin32Error());return code;}}
    internal async Task Wait()
    {
        var watch=System.Diagnostics.Stopwatch.StartNew();
        while(!Exited && watch.Elapsed<TimeSpan.FromSeconds(100))await Task.Delay(25);
        Require(Exited,"REAL_CHILD_TIMEOUT");
    }
    internal async Task KillOnlyThisProcess()
    {
        if(!Exited && !TerminateProcess(handle,91))throw new Win32Exception(Marshal.GetLastWin32Error());
        await Wait(); // deliberately leaves descendant processes observable in the job
    }
    public void Dispose()=>handle.Dispose();
    [DllImport("kernel32.dll",SetLastError=true)] private static extern uint WaitForSingleObject(SafeFileHandle handle,uint milliseconds);
    [DllImport("kernel32.dll",SetLastError=true)] private static extern bool TerminateProcess(SafeFileHandle handle,uint code);
    [DllImport("kernel32.dll",SetLastError=true)] private static extern bool GetExitCodeProcess(SafeFileHandle handle,out uint code);
}
