// Phase 1 experimental BRIDGE, not a production updater or independent witness.
// Reuses the L18-L22 coordinator gate, signed INERT fixture state machine,
// and historical UNKNOWN policy. Separately, the existing Python-TUF real-GF
// laboratory child verifies/observes real packages and restores OLD.
// The fixture TUF target DOES NOT cryptographically authorize the GF package;
// therefore the composed decision MUST REMAIN UNKNOWN, not local NEW/READY.
using System.Diagnostics;

    // LAB-ONLY real GF selector stored INSIDE the SAME L18 source root as its
    // journal. Never the L14 inert fixture selector or a Python-owned selector.
    // The coordinator exclusively owns transitions while its gate is held.
    private const string RealSelectorName="l18-real-active.fixture.json";
    private static string RealSelector(string src)=>Path.Combine(src,RealSelectorName);

    private static void RequireRealSelector(string src,string installation,string tx,
        string sha,string selected,int generation)
    {
        using var pin=WindowsFileGuard.Open(src,RealSelector(src));
        byte[] content=pin.Read(2048);
        using var doc=JsonDocument.Parse(content);
        var v=doc.RootElement;
        Require(v.ValueKind==JsonValueKind.Object &&
            v.EnumerateObject().Count()==8 &&
            v.GetProperty("schema").GetString()=="gf-l18-real-active-lab/1" &&
            v.GetProperty("installation").GetString()==installation &&
            v.GetProperty("transaction").GetString()==tx &&
            v.GetProperty("manifest_sha256").GetString()==sha &&
            v.GetProperty("selected").GetString()==selected &&
            v.GetProperty("generation").GetInt32()==generation &&
            v.GetProperty("test_only").GetBoolean() &&
            !v.GetProperty("operational_installable").GetBoolean(),
            "PHASE1_REAL_SELECTOR_WRONG_STATE");
        pin.Recheck();
    }

    private static void WriteRealSelector(string src,string installation,string tx,
        string sha,string selected,int generation)
    {
        FixtureSandbox.RequireRoot(src);
        Require(Environment.GetEnvironmentVariable("GITHUB_ACTIONS")=="true" &&
            (selected=="old" || selected=="new") && generation is >=1 and <=3 &&
            Hex(sha,64),"PHASE1_REAL_SELECTOR_OUTSIDE_LAB");
        string dest=RealSelector(src);
        if(generation==1)
            Require(!File.Exists(dest),"PHASE1_REAL_SELECTOR_MUST_START_ABSENT");
        else
            RequireRealSelector(src,installation,tx,sha,
                generation==2?"old":"new",generation-1);
        var item=new {
            schema="gf-l18-real-active-lab/1",installation,transaction=tx,
            manifest_sha256=sha,selected,generation,
            test_only=true,operational_installable=false
        };
        byte[] bytes=JsonSerializer.SerializeToUtf8Bytes(item);
        Require(bytes.Length is >0 and <=2048,"PHASE1_SELECTOR_SERIALIZATION");
        string tmp=Path.Combine(src,"l18-real-pending-"+Guid.NewGuid().ToString("N")+".fixture.json");
        try
        {
            using(var stream=new FileStream(tmp,FileMode.CreateNew,FileAccess.Write,
                FileShare.None,4096,FileOptions.WriteThrough))
            {
                stream.Write(bytes);
                stream.Flush(true);
            }
            File.Move(tmp,dest,generation>1);
            RequireRealSelector(src,installation,tx,sha,selected,generation);
        }
        finally { if(File.Exists(tmp)) File.Delete(tmp); }
    }

    private static async Task ReadBarrier(Process child,string expected,TimeSpan timeout)
    {
        string? message;
        try{ message=await child.StandardOutput.ReadLineAsync().WaitAsync(timeout); }
        catch(TimeoutException){throw new Rejected("PHASE1_HANDOFF_TIMED_OUT_"+expected);}
        if(message!=expected)
        {
            Console.Error.WriteLine("GF_PHASE1_BARRIER_DIAG "+
                "received_tag="+(message?.Split(':')[0] ?? "<eof>")+
                " received_len="+(message?.Length ?? -1)+
                " expected_tag="+expected.Split(':')[0]+
                " equal="+(message==expected));
            throw new Rejected("PHASE1_HANDOFF_UNEXPECTED_"+expected.Split(':')[0]);
        }
    }

    private static async Task SendBarrier(Process child,string command)
    {
        await child.StandardInput.WriteLineAsync(command);
        await child.StandardInput.FlushAsync();
    }

    internal static async Task<L18Decision> ExecuteRealLabInertOnly(
        string src,string lab,string bootstrap,string python,string script,
        string repository,string old,string candidate,string sandbox,string report)
    {
        Root(src,lab);
        Require(Environment.GetEnvironmentVariable("GITHUB_ACTIONS")=="true",
            "PHASE1_GITHUB_WINDOWS_ONLY");
        string temp=Path.GetFullPath(Environment.GetEnvironmentVariable("RUNNER_TEMP") ??
            throw new Rejected("PHASE1_NO_TEMP"));
        string folder=Path.GetFullPath(sandbox);
        Require(folder.StartsWith(temp+Path.DirectorySeparatorChar,
            StringComparison.OrdinalIgnoreCase) && !Directory.Exists(folder),
            "PHASE1_SANDBOX_NEW_AND_TEMP_ONLY");
        Require(Path.GetFullPath(old).StartsWith(temp+Path.DirectorySeparatorChar,
                StringComparison.OrdinalIgnoreCase) &&
            Path.GetFullPath(candidate).StartsWith(temp+Path.DirectorySeparatorChar,
                StringComparison.OrdinalIgnoreCase),"PHASE1_PACKAGE_OUTSIDE_TEMP");
        Require(Path.GetFullPath(script).Equals(Path.Combine(
            Path.GetFullPath(repository),"prototipos","autoatualizador-seguro",
            "phase1-composed","phase1_lab.py"),StringComparison.OrdinalIgnoreCase)
            && File.Exists(script) && File.Exists(python),"PHASE1_FIXED_PYTHON_SCRIPT");
        Require(!File.Exists(report),"PHASE1_REPORT_ALREADY_EXISTS");

        // The same L18 gate and phase journal now persist the REAL GF
        // signed manifest hash before selecting NEW. No production trust:
        // this is a disposable laboratory delegation to Python-TUF test keys.
        using var gate=TryCoordinatorGate(src);
        if(gate is null)return L18Decision.Busy;
        string installation=new PersistentState(src).Load().Record.InstallationId;
        string tx=L14Model.Tx(lab);
        string authDir=Path.Combine(temp,"gf-phase1-auth-"+Guid.NewGuid().ToString("N"));
        string authReport=Path.Combine(temp,"gf-phase1-preflight-"+Guid.NewGuid().ToString("N")+".json");
        string signedTarget;
        using(var preparationLease=L16Boundary.Exclusive(src))
        {
            var preflight=new ProcessStartInfo(Path.GetFullPath(python))
            {
                UseShellExecute=false,CreateNoWindow=true,
                RedirectStandardOutput=true,RedirectStandardError=true,
                WorkingDirectory=Path.GetFullPath(repository)
            };
            foreach(string word in new[]{Path.GetFullPath(script),"--preflight-only",
                "--repository",repository,"--old",old,"--new",candidate,
                "--sandbox",sandbox,"--report",report,"--auth-dir",authDir,
                "--auth-report",authReport,"--installation",installation,
                "--transaction",tx})
                preflight.ArgumentList.Add(word);
            using var prep=Process.Start(preflight) ??
                throw new Rejected("PHASE1_PREFLIGHT_NOT_STARTED");
            Task<string> prepOut=prep.StandardOutput.ReadToEndAsync();
            Task<string> prepErr=prep.StandardError.ReadToEndAsync();
            try {await prep.WaitForExitAsync().WaitAsync(TimeSpan.FromMinutes(2));}
            catch(TimeoutException)
            {
                if(!prep.HasExited){prep.Kill(entireProcessTree:true);await prep.WaitForExitAsync();}
                throw new Rejected("PHASE1_PREFLIGHT_TIMED_OUT_NO_JOURNAL");
            }
            string prepDiagnostic=await prepErr;
            _=await prepOut;
            if(prep.ExitCode!=0)
            {
                Console.Error.WriteLine("GF_PHASE1_PREP_ERROR "+
                    prepDiagnostic[..Math.Min(prepDiagnostic.Length,1300)]);
                throw new Rejected("PHASE1_TUF_PREFLIGHT_FAILURE_NO_JOURNAL");
            }
            Require(File.Exists(authReport) && new FileInfo(authReport).Length is >0 and <=4096,
                "PHASE1_PREFLIGHT_REPORT_MISSING");
            using var proof=JsonDocument.Parse(File.ReadAllBytes(authReport));
            var p=proof.RootElement;
            signedTarget=p.GetProperty("manifest_sha256").GetString() ??
                throw new Rejected("PHASE1_NO_SIGNED_TARGET");
            Require(Hex(signedTarget,64) &&
                p.GetProperty("schema").GetString()=="gf-real-lab-auth-preflight/1" &&
                p.GetProperty("installation").GetString()==installation &&
                p.GetProperty("transaction").GetString()==tx &&
                p.GetProperty("selected_before").GetString()=="old" &&
                p.GetProperty("test_only").GetBoolean() &&
                p.GetProperty("tuf_verified").GetBoolean() &&
                p.GetProperty("no_real_process_launched").GetBoolean() &&
                !p.GetProperty("operational_installable").GetBoolean(),
                "PHASE1_PREFLIGHT_IDENTITY_OR_AUTHORIZATION");
            string path=Path.Combine(authDir,"GF-LAB-REAL-PACKAGE-MANIFEST.json");
            Require(File.Exists(path) && new FileInfo(path).Length is >0 and <=2097152,
                "PHASE1_SIGNED_MANIFEST_MISSING");
            byte[] manifest=File.ReadAllBytes(path);
            string independentlyHashed=Convert.ToHexString(
                System.Security.Cryptography.SHA256.HashData(manifest)).ToLowerInvariant();
            using var manifestData=JsonDocument.Parse(manifest);
            Require(independentlyHashed==signedTarget &&
                manifestData.RootElement.GetProperty("installation").GetString()==installation &&
                manifestData.RootElement.GetProperty("transaction").GetString()==tx &&
                manifestData.RootElement.GetProperty("schema").GetString()=="gf-real-package-lab/2" &&
                manifestData.RootElement.GetProperty("selected_before").GetString()=="old" &&
                !manifestData.RootElement.GetProperty("operational_installable").GetBoolean(),
                "PHASE1_SIGNED_MANIFEST_SCOPE_OR_DIGEST");
            // The lab signed manifest digest, not L16's inert entry digest,
            // becomes TargetSha in the original durable L18 phase journal.
            BeginCore(src,lab,bootstrap,signedTarget);
            Require(Load(src,lab).All(x=>x.TargetSha==signedTarget),
                "PHASE1_JOURNAL_WRONG_TARGET");
        }
        // From this point, the ONLY real-package selector is controlled
        // by C#, persisted within the same src root as the L18 journal.
        WriteRealSelector(src,installation,tx,signedTarget,"old",1);
        Append(src,lab,Switching,"real-gf-lab-selector-old-coordinator-owned",
            Load(src,lab)[0].TargetSha);
        string token=Guid.NewGuid().ToString("N"); // correlation only, NOT authentication
        var psi=new ProcessStartInfo(Path.GetFullPath(python))
        {
            UseShellExecute=false,CreateNoWindow=true,
            RedirectStandardOutput=true,RedirectStandardError=true,
            RedirectStandardInput=true,
            WorkingDirectory=Path.GetFullPath(repository)
        };
        psi.ArgumentList.Add(Path.GetFullPath(script));
        foreach(string word in new[]{"--repository",repository,"--old",old,
            "--new",candidate,"--sandbox",sandbox,"--report",report,
            "--auth-dir",authDir,"--auth-report",authReport,
            "--installation",installation,"--transaction",tx,
            "--selector-root",src,"--handoff-token",token})
            psi.ArgumentList.Add(word);
        using Process process=Process.Start(psi) ??
            throw new Rejected("PHASE1_CHILD_NOT_STARTED");
        Task<string> stderr=process.StandardError.ReadToEndAsync();
        Append(src,lab,Launched,"real-lab-pid-"+process.Id.ToString(),
            Load(src,lab)[0].TargetSha);
        // OLD completion is observed before C# selects NEW. The Python
        // child cannot write any selector and must await C#'s instruction.
        try
        {
            await ReadBarrier(process,"GF_PHASE1_BARRIER_OLD_DONE:"+token,
                TimeSpan.FromSeconds(105));
            RequireRealSelector(src,installation,tx,signedTarget,"old",1);
            Require(Load(src,lab).All(x=>x.TargetSha==signedTarget),
                "PHASE1_JOURNAL_CHANGED_BEFORE_SWITCH");
            WriteRealSelector(src,installation,tx,signedTarget,"new",2);
            await SendBarrier(process,"GF_PHASE1_CONTINUE_NEW:"+token);
            await ReadBarrier(process,"GF_PHASE1_BARRIER_NEW_DONE:"+token,
                TimeSpan.FromSeconds(105));
            RequireRealSelector(src,installation,tx,signedTarget,"new",2);
            // The NEW child has reported its graceful termination. A real
            // operational controller cannot trust this same-user message;
            // this CI is strictly in a disposable laboratory.
            WriteRealSelector(src,installation,tx,signedTarget,"old",3);
            await SendBarrier(process,"GF_PHASE1_CONTINUE_OLD:"+token);
            // The child writes an extensive JSON diagnostic AFTER the final
            // barrier. Drain stdout while waiting: otherwise its pipe fills,
            // blocking child exit and falsely appearing as recovery timeout.
            Task<string> trailingOutput=process.StandardOutput.ReadToEndAsync();
            await process.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(105));
            string trailing=await trailingOutput;
            Require(trailing.Length<=65536,"PHASE1_CHILD_STDOUT_UNBOUNDED");
        }
        catch(Exception ex) when(ex is TimeoutException or Rejected or IOException)
        {
            if(process.HasExited)
            {
                string earlyErr=await stderr;
                Console.Error.WriteLine("GF_PHASE1_EARLY_CHILD_ERROR "+
                    earlyErr[..Math.Min(earlyErr.Length,1600)]);
            }
            // No fictitious success after a missing handshake, PID still
            // alive, or unobserved late effects. No blind rollback here.
            L16Boundary.MarkUncertain(src);
            RememberUnknown(src,lab,"real-gf-child-or-selector-unknown");
            throw;
        }
        Append(src,lab,Exited,process.ExitCode==0?"exit-code-zero":
            "exit-code-nonzero",Load(src,lab)[0].TargetSha);
        string standardErr=await stderr;
        RequireRealSelector(src,installation,tx,signedTarget,"old",3);
        if(process.ExitCode!=0)
        {
            RememberUnknown(src,lab,"real-gf-lab-child-failed");
            Console.Error.WriteLine("GF_PHASE1_CHILD_DIAGNOSTIC "+
                standardErr[..Math.Min(standardErr.Length,1800)]);
            throw new Rejected("PHASE1_CHILD_FAILED_EXIT_"+process.ExitCode);
        }
        // A successful subordinate report is independently required, not
        // inferred from a persisted UNKNOWN alone.
        Require(File.Exists(report),"PHASE1_RESULT_MISSING");
        Console.WriteLine("GF_PHASE1_CHILD_COMPLETED_VERIFIED_EXIT_ZERO");
        // A JSON report from this same-user child is NOT a trusted READY
        // witness and does not constitute a signed packet-to-journal binding.
        Require(File.Exists(report),"PHASE1_RESULT_MISSING");
        Require(new FileInfo(report).Length is >0 and <=32768,"PHASE1_REPORT_SIZE");
        using(JsonDocument doc=JsonDocument.Parse(File.ReadAllBytes(report)))
        {
            var root=doc.RootElement;
            Require(root.GetProperty("status").GetString()==
                "blocked_until_real_ready_and_independent_review" &&
                !root.GetProperty("ready_verified").GetBoolean() &&
                !root.GetProperty("phase1_complete").GetBoolean() &&
                root.GetProperty("authenticated_real_package").GetBoolean() &&
                root.GetProperty("minimum_readonly_health_contract_observed").GetBoolean() &&
                root.GetProperty("real_manifest_bound_to_preflight").GetBoolean() &&
                root.GetProperty("selector_controlled_by_l18_only").GetBoolean() &&
                root.GetProperty("executable_path_resolved_from_l18_selector").GetBoolean() &&
                root.GetProperty("restored_old_booted_after_recovery").GetBoolean() &&
                root.GetProperty("selector_python_write_attempts").GetInt32()==0 &&
                root.GetProperty("coordinator_selected_new_then_restored_old").GetBoolean() &&
                root.GetProperty("authorization_installation").GetString()==installation &&
                root.GetProperty("authorization_transaction").GetString()==tx &&
                root.GetProperty("signed_test_metadata").GetProperty("manifest_sha256").GetString()==signedTarget &&
                root.GetProperty("old_and_canary_preserved").GetBoolean() &&
                root.GetProperty("rolled_back_due_to_no_ready").GetBoolean() &&
                root.GetProperty("operational_gates_closed").GetInt32()==0,
                "PHASE1_FALSE_SUCCESS_OR_INCOMPLETE_CHILD_REPORT");
            var oldProc=root.GetProperty("old_process");
            var newProc=root.GetProperty("new_process");
            Require(oldProc.GetProperty("graceful_shutdown_verified_for_owned_process").GetBoolean()
                && newProc.GetProperty("graceful_shutdown_verified_for_owned_process").GetBoolean()
                && root.GetProperty("restored_old_process").GetProperty(
                    "graceful_shutdown_verified_for_owned_process").GetBoolean()
                && oldProc.GetProperty("read_only_ui_controls")
                    .GetProperty("functional_inventory_ui_observed").GetBoolean()
                && newProc.GetProperty("read_only_ui_controls")
                    .GetProperty("functional_inventory_ui_observed").GetBoolean(),
                "PHASE1_SHUTDOWN_OR_UI_NOT_DEMONSTRATED");
        }

        // The signed REAL target is persisted across all L18 phases,
        // including UNKNOWN, before and after the child observes NEW.
        Require(Load(src,lab).All(x=>x.TargetSha==signedTarget),
            "PHASE1_REAL_TARGET_JOURNAL_DRIFT");
        // Do NOT use L14 fixture selector to classify the REAL GF.
        // The real selector is now singular and C#-owned. Because no
        // operational independent READY exists, mark the same L18 journal
        // UNKNOWN rather than asserting success based on a same-user receipt.
        RequireRealSelector(src,installation,tx,signedTarget,"old",3);
        RememberUnknown(src,lab,"real-gf-no-independent-operational-ready");
        Require(Load(src,lab)[^1].Phase==Unknown,
            "PHASE1_UNKNOWN_NOT_DURABLE");
        return L18Decision.UnknownBlocked;
    }
}

[System.Runtime.Versioning.SupportedOSPlatform("windows")]
internal static class Phase1BridgeTests
{
    public static async Task<int> Main(string[] args)
    {
        if(!OperatingSystem.IsWindows())return 2;
        // <report-bridge> <fixtures> <bootstrap> <repository> <python>
        // <script> <old> <candidate> <sandbox> <report-python>
        if(args.Length!=11 || args[0]!="--run")return 2;
        L16Tests.fixtures=Path.GetFullPath(args[2]);
        L16Tests.bootstrap=Path.GetFullPath(args[3]);
        L16Tests.suite=Path.Combine(Path.GetTempPath(),
            "gf-u2-phase1-csharp-"+Guid.NewGuid().ToString("N"));
        Directory.CreateDirectory(L16Tests.suite);
        string evidence=args[1];
        try
        {
            using var scenario=await L16Tests.Prepared();
            var decision=await L18Coordinator.ExecuteRealLabInertOnly(
                scenario.Source.Root,scenario.Lab,L16Tests.bootstrap,
                args[5],args[6],args[4],args[7],args[8],args[9],args[10]);
            Require(decision==L18Decision.UnknownBlocked,
                "PHASE1_UNKNOWN_REQUIRED");
            Require(File.Exists(args[10]),"PHASE1_CHILD_REPORT_MISSING_AFTER_BRIDGE");
            Require(new FileInfo(args[10]).Length is >0 and <=32768,"PHASE1_CHILD_REPORT_SIZE");
            using(var real=JsonDocument.Parse(File.ReadAllBytes(args[10])))
            {
                var actual=real.RootElement;
                Require(actual.GetProperty("old_and_canary_preserved").GetBoolean() &&
                    actual.GetProperty("authenticated_real_package").GetBoolean() &&
                    actual.GetProperty("rolled_back_due_to_no_ready").GetBoolean() &&
                    !actual.GetProperty("ready_verified").GetBoolean(),
                    "PHASE1_REAL_REPORT_INCOMPLETE");
            }
            using var finalReal=JsonDocument.Parse(File.ReadAllBytes(args[10]));
            string expectedRealTarget=finalReal.RootElement.GetProperty("signed_test_metadata")
                .GetProperty("manifest_sha256").GetString()!;
            Require(Hex(expectedRealTarget,64),"PHASE1_REAL_TARGET_MISSING");
            // Presence of an append-only UNKNOWN event is required: mere
            // returned UnknownBlocked without durable state is not enough.
            var events=Directory.GetFiles(scenario.Source.Root,
                "l18-phase-*.fixture.json");
            Require(events.Length>=8,"PHASE1_JOURNAL_TOO_SHORT");
            var serialized=events.Select(File.ReadAllText).ToArray();
            Require(serialized.Any(e=>e.Contains("\"Phase\":\"unknown\"",
                StringComparison.OrdinalIgnoreCase)), "PHASE1_UNKNOWN_NOT_DURABLE");
            Require(serialized.All(e=>e.Contains("\"TargetSha\":\""+expectedRealTarget+"\"",
                StringComparison.OrdinalIgnoreCase)),
                "PHASE1_AUTHENTICATED_REAL_TARGET_MISSING_FROM_L18_JOURNAL");
            scenario.Preserved();
            var result=new{
                schema="gf-phase1-csharp-l18-real-lab-bridge/1",
                coordinator_l18_journal_persisted=true,
                real_gf_child_supervised_under_coordinator_gate=true,
                coordinator_decision=decision.ToString(),
                packet_authentication_is_still_external_lab_python=true,
                fixture_target_is_not_real_gf_packet=true,
                real_manifest_sha_in_l18_target=true,
                real_manifest_scope_binds_installation_and_tx=true,
                prepared_tuf_metadata_reverified_before_new=true,
                minimum_readonly_functional_search_verified=true,
                one_real_selector_coordinator_owned=true,
                real_selector_new_then_old_observed=true,
                no_python_selector_writes=true,
                real_packet_not_bound_to_l18_target=false,
                production_tuf_authority=false,
                verified_operational_ready=false,
                phase1_complete=false,
                operational_gates_closed=0
            };
            File.WriteAllText(evidence,JsonSerializer.Serialize(result));
            Console.WriteLine("GF_PHASE1_CSHARP_BRIDGE "+JsonSerializer.Serialize(result));
            return 0;
        }
        finally
        {
            Directory.Delete(L16Tests.suite,true);
        }
    }
